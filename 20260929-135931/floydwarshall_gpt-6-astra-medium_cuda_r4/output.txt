// Index calculation for flattened 2D array
inline constexpr size_t idx2(const size_t i, const size_t j, const size_t n) noexcept {
    return j * n + i;
}

void initializeDistanceMatrix(std::vector<unsigned int>& dist, const size_t numNodes, 
                              const unsigned int rangeMin, const unsigned int rangeMax) {
    unsigned int seed = 42;
    const double range = static_cast<double>(rangeMax - rangeMin) + 1.0;
    
    for (size_t i = 0; i < numNodes * numNodes; ++i) {
        dist[i] = rangeMin + (unsigned int)(range * rand_r(&seed) / (double)RAND_MAX);
    }
    
    // Set diagonal to zero (distance from node to itself is 0)
    for (size_t i = 0; i < numNodes; ++i) {
        dist[idx2(i, i, numNodes)] = 0;
    }
}

void initializePathMatrix(std::vector<unsigned int>& path, const size_t numNodes) {
    for (size_t j = 0; j < numNodes; ++j) {
        for (size_t i = 0; i < numNodes; ++i) {
            path[idx2(i, j, numNodes)] = j;
            path[idx2(j, i, numNodes)] = i;
        }
        path[idx2(j, j, numNodes)] = j;
    }
}

void floydWarshall(std::vector<unsigned int>& dist, 
                   std::vector<unsigned int>& path, 
                   const size_t numNodes) {
    // Classic Floyd-Warshall algorithm
    // For each intermediate node k
    for (size_t k = 0; k < numNodes; ++k) {
        // For each source node i
        for (size_t i = 0; i < numNodes; ++i) {
            // For each destination node j
            for (size_t j = 0; j < numNodes; ++j) {
                const unsigned int distIJ = dist[idx2(j, i, numNodes)];
                const unsigned int distIK = dist[idx2(k, i, numNodes)];
                const unsigned int distKJ = dist[idx2(j, k, numNodes)];
                
                const unsigned int newDist = distIK + distKJ;
                
                if (newDist < distIJ) {
                    dist[idx2(j, i, numNodes)] = newDist;
                    path[idx2(j, i, numNodes)] = k;
                }
            }
        }
    }
}

bool validateResult(const std::vector<unsigned int>& dist, const size_t numNodes) {
    // Basic sanity checks
    
    // 1. Diagonal should be zero
    for (size_t i = 0; i < numNodes; ++i) {
        if (dist[idx2(i, i, numNodes)] != 0) {
            printf("Validation failed: diagonal element [%zu,%zu] is not zero\n", i, i);
            return false;
        }
    }
    
    // 2. Triangle inequality: dist[i][k] + dist[k][j] >= dist[i][j]
    // Check a sample of paths to avoid O(n^3) validation time
    for (size_t i = 0; i < std::min(numNodes, static_cast<size_t>(10)); ++i) {
        for (size_t j = 0; j < std::min(numNodes, static_cast<size_t>(10)); ++j) {
            for (size_t k = 0; k < numNodes; ++k) {
                const unsigned int distIJ = dist[idx2(j, i, numNodes)];
                const unsigned int distIK = dist[idx2(k, i, numNodes)];
                const unsigned int distKJ = dist[idx2(j, k, numNodes)];
                
                // Check for overflow before addition
                if (distIK < INF && distKJ < INF) {

                        printf("Validation failed: triangle inequality violated at [%zu,%zu,%zu]\n", 
                               i, j, k);

                    }
                }
            }
        }
    }
    
    return true;
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

namespace {
void checkCuda(cudaError_t status, const char* operation) {
    if (status != cudaSuccess) {
        fprintf(stderr, "CUDA %s failed: %s\n", operation, cudaGetErrorString(status));
        std::exit(EXIT_FAILURE);
    }
}

// Each warp accesses consecutive columns. Four rows per thread reuse the pivot
// row value and amortize indexing without reducing the number of resident warps.
constexpr unsigned int RowsPerThread = 4;
__global__ void relaxDistances(unsigned int* dist, unsigned int* path,
                               size_t n, size_t stride, size_t k) {
    const size_t j = size_t(blockIdx.x) * blockDim.x + threadIdx.x;
    const size_t firstRow = size_t(blockIdx.y) * blockDim.y * RowsPerThread
                          + threadIdx.y;
    if (j >= n || j == k) return;
    const unsigned int distKJ = dist[k * stride + j];
    #pragma unroll
    for (unsigned int r = 0; r < RowsPerThread; ++r) {
        const size_t i = firstRow + r * blockDim.y;
        if (i < n && i != k) {
            const size_t offset = i * stride + j;
            const unsigned int candidate = dist[i * stride + k] + distKJ;
            if (candidate < dist[offset]) {
                dist[offset] = candidate;
                path[offset] = static_cast<unsigned int>(k);
            }
        }
    }
}
} // namespace

void floydWarshall(std::vector<unsigned int>& dist,
                   std::vector<unsigned int>& path,
                   const size_t numNodes) {
    // Require CUDA even for an empty graph; there is no CPU fallback.
    checkCuda(cudaFree(nullptr), "initialization");
    if (numNodes == 0) return;

    // Align every row for coalesced accesses, including non-warp-sized graphs.
    const size_t stride = (numNodes + 31) / 32 * 32;
    const size_t pitch = stride * sizeof(unsigned int);
    const size_t width = numNodes * sizeof(unsigned int);
    const size_t bytes = pitch * numNodes;
    unsigned int* deviceDist = nullptr;
    unsigned int* devicePath = nullptr;
    checkCuda(cudaMalloc(&deviceDist, bytes), "distance allocation");
    checkCuda(cudaMalloc(&devicePath, bytes), "path allocation");
    checkCuda(cudaMemcpy2D(deviceDist, pitch, dist.data(), width, width, numNodes,
                           cudaMemcpyHostToDevice), "distance upload");
    checkCuda(cudaMemcpy2D(devicePath, pitch, path.data(), width, width, numNodes,
                           cudaMemcpyHostToDevice), "path upload");

    const dim3 block(32, 8);
    const dim3 grid((numNodes + block.x - 1) / block.x,
                    (numNodes + block.y * RowsPerThread - 1)
                    / (block.y * RowsPerThread));
    // Stream ordering supplies the global barrier between successive k values.
    // A zero diagonal and nonnegative weights mean the pivot row/column never
    // change within a step. Skipping their writes also avoids read/write races.
    for (size_t k = 0; k < numNodes; ++k) {
        relaxDistances<<<grid, block>>>(deviceDist, devicePath, numNodes, stride, k);
    }
    checkCuda(cudaGetLastError(), "kernel launch");
    // Blocking downloads include completion in the benchmark's measured time.
    checkCuda(cudaMemcpy2D(dist.data(), width, deviceDist, pitch, width, numNodes,
                           cudaMemcpyDeviceToHost), "distance download");
    checkCuda(cudaMemcpy2D(path.data(), width, devicePath, pitch, width, numNodes,
                           cudaMemcpyDeviceToHost), "path download");
    checkCuda(cudaFree(devicePath), "path release");
    checkCuda(cudaFree(deviceDist), "distance release");
}

bool validateResult(const std::vector<unsigned int>& dist, const size_t numNodes) {
    // Basic sanity checks
    
    // 1. Diagonal should be zero
    for (size_t i = 0; i < numNodes; ++i) {
        if (dist[idx2(i, i, numNodes)] != 0) {
            printf("Validation failed: diagonal element [%zu,%zu] is not zero\n", i, i);
            return false;
        }
    }
    
    // 2. Triangle inequality: dist[i][k] + dist[k][j] >= dist[i][j]
    // Check a sample of paths to avoid O(n^3) validation time
    for (size_t i = 0; i < std::min(numNodes, static_cast<size_t>(10)); ++i) {
        for (size_t j = 0; j < std::min(numNodes, static_cast<size_t>(10)); ++j) {
            for (size_t k = 0; k < numNodes; ++k) {
                const unsigned int distIJ = dist[idx2(j, i, numNodes)];
                const unsigned int distIK = dist[idx2(k, i, numNodes)];
                const unsigned int distKJ = dist[idx2(j, k, numNodes)];
                
                // Check for overflow before addition
                if (distIK < INF && distKJ < INF) {
                    if (distIK + distKJ < distIJ) {
                        printf("Validation failed: triangle inequality violated at [%zu,%zu,%zu]\n", 
                               i, j, k);
                        return false;
                    }
                }
            }
        }
    }
    
    return true;
}
