// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

    std::printf(\"  -r           Print results for external validation\\n\");
    std::printf(\"  -h           Show this help message\\n\");

// Avoid MPI's int count limit when collecting an optional full result.
void transferRows(const double* data, size_t count, int peer, bool send,
                  int rank) {
    size_t offset = 0;
    while (offset < count) {
        const int chunk = static_cast<int>(std::min(count - offset,
                                                   static_cast<size_t>(INT_MAX)));
        if (send) {
            checkMpi(MPI_Send(data + offset, chunk, MPI_DOUBLE, peer, 0,
                              MPI_COMM_WORLD), rank, \"MPI_Send\");
        } else {
            checkMpi(MPI_Recv(const_cast<double*>(data) + offset, chunk,
                              MPI_DOUBLE, peer, 0, MPI_COMM_WORLD,
                              MPI_STATUS_IGNORE), rank, \"MPI_Recv\");
        }
        offset += static_cast<size_t>(chunk);
    }
}
