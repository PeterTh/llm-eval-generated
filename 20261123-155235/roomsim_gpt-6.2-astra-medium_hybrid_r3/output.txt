class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

   325	        const val_t t = (1.0f + std::sqrt(5.0f)) / 2.0f;
   326	
   327	        std::vector<Vec3> vertices = {
   328	            Vec3(-1,  t,  0).normalized() * radius,
   329	            Vec3( 1,  t,  0).normalized() * radius,
   330	            Vec3(-1, -t,  0).normalized() * radius,
   331	            Vec3( 1, -t,  0).normalized() * radius,
   332	            Vec3( 0, -1,  t).normalized() * radius,
   333	            Vec3( 0,  1,  t).normalized() * radius,
   334	            Vec3( 0, -1, -t).normalized() * radius,
   335	            Vec3( 0,  1, -t).normalized() * radius,
   336	            Vec3( t,  0, -1).normalized() * radius,
   337	            Vec3( t,  0,  1).normalized() * radius,
   338	            Vec3(-t,  0, -1).normalized() * radius,
   339	            Vec3(-t,  0,  1).normalized() * radius
   340	        };
   341	
   342	        // Initial icosahedron faces (20 triangles)
   343	        std::vector<std::array<idx_t, 3>> faces = {
   344	            {0, 11, 5}, {0, 5, 1}, {0, 1, 7}, {0, 7, 10}, {0, 10, 11},
   345	            {1, 5, 9}, {5, 11, 4}, {11, 10, 2}, {10, 7, 6}, {7, 1, 8},
   346	            {3, 9, 4}, {3, 4, 2}, {3, 2, 6}, {3, 6, 8}, {3, 8, 9},
   347	            {4, 9, 5}, {2, 4, 11}, {6, 2, 10}, {8, 6, 7}, {9, 8, 1}
   348	        };
   349	
   350	        // Subdivide
   351	        for (int i = 0; i < subdivisions; ++i) {
   352	            std::vector<std::array<idx_t, 3>> newFaces;
   353	            std::map<std::pair<idx_t, idx_t>, idx_t> midpointCache;
   354	
   355	            auto getMidpoint = [&](idx_t i1, idx_t i2) -> idx_t {
   356	                auto key = std::make_pair(std::min(i1, i2), std::max(i1, i2));
   357	                auto it = midpointCache.find(key);
   358	                if (it != midpointCache.end()) return it->second;
   359	
   360	                Vec3 mid = (vertices[i1] + vertices[i2]) / 2.0f;
   361	                mid = mid.normalized() * radius;
   362	                idx_t idx = static_cast<idx_t>(vertices.size());
   363	                vertices.push_back(mid);
   364	                midpointCache[key] = idx;
   365	                return idx;
   366	            };
   367	
   368	            for (const auto& face : faces) {
   369	                idx_t a = getMidpoint(face[0], face[1]);
   370	                idx_t b = getMidpoint(face[1], face[2]);
   371	                idx_t c = getMidpoint(face[2], face[0]);
   372	
   373	                newFaces.push_back({face[0], a, c});
   374	                newFaces.push_back({face[1], b, a});
   375	                newFaces.push_back({face[2], c, b});
   376	                newFaces.push_back({a, b, c});
   377	            }
   378	            faces = std::move(newFaces);
   379	        }
   380	
   381	        // Build triangles (normals pointing inward for a "room")
   382	        triangles.reserve(faces.size());
   383	        for (const auto& face : faces) {
   384	            // Reverse winding to make normals point inward
   385	            triangles.emplace_back(vertices[face[2]], vertices[face[1]], vertices[face[0]]);
   386	        }
   387	    }
   388	};

   390	// ============================================================================
   391	// Random Number Generation
   392	// ============================================================================
   393	
   394	class RandomGenerator {
   395	    std::mt19937 rng;
   396	    std::uniform_real_distribution<val_t> dist;
   397	public:
   398	    explicit RandomGenerator(uint32_t seed = 42) : rng(seed), dist(0.0f, 1.0f) {}
   399	    val_t rand() { return dist(rng); }
   400	};

   402	// Generate a random point inside a triangle using barycentric coordinates
   403	Vec3 randomPointInTriangle(const Triangle& t, RandomGenerator& rng) {
   404	    val_t u = rng.rand();
   405	    val_t v = rng.rand();
   406	    if (u + v > 1.0f) {
   407	        u = 1.0f - u;
   408	        v = 1.0f - v;
   409	    }
   410	    Vec3 ab = t.b - t.a;
   411	    Vec3 ac = t.c - t.a;
   412	    return t.a + ab * u + ac * v;
   413	}

   415	// ============================================================================
   416	// Ray-Triangle Intersection (Möller-Trumbore algorithm)
   417	// ============================================================================
   418	
   419	val_t rayTriangleIntersect(const Vec3& orig, const Vec3& dir,
   420	                            const Vec3& v0, const Vec3& v1, const Vec3& v2) {
   421	    Vec3 e1 = v1 - v0;
   422	    Vec3 e2 = v2 - v0;
   423	    Vec3 pvec = dir.cross(e2);
   424	    val_t det = e1.dot(pvec);
   425	
   426	    if (std::abs(det) < EPSILON) return std::numeric_limits<val_t>::max();
   427	
   428	    val_t invDet = 1.0f / det;
   429	    Vec3 tvec = orig - v0;
   430	    val_t u = tvec.dot(pvec) * invDet;
   431	    if (u < 0.0f || u > 1.0f) return std::numeric_limits<val_t>::max();
   432	
   433	    Vec3 qvec = tvec.cross(e1);
   434	    val_t v = dir.dot(qvec) * invDet;
   435	    if (v < 0.0f || u + v > 1.0f) return std::numeric_limits<val_t>::max();
   436	
   437	    return e2.dot(qvec) * invDet;
   438	}

   440	// ============================================================================
   441	// Visibility Testing (Octree-accelerated)
   442	// ============================================================================
   443	
   444	// Check if a ray between two triangles is blocked by any other triangle
   445	// Uses octree for O(log N) instead of O(N) search
   446	bool isRayBlocked(const Vec3& from, const Vec3& to,
   447	                  const Octree& octree,
   448	                  size_t srcTriIdx, size_t dstTriIdx) {
   449	    Vec3 dir = to - from;
   450	    val_t rayLen = dir.norm();
   451	    if (rayLen < EPSILON) return true;
   452	    Vec3 dirNorm = dir / rayLen;
   453	
   454	    return octree.applyToTris(from, to, [&](size_t idx, const Triangle& tri) {
   455	        if (idx == srcTriIdx || idx == dstTriIdx) return false;
   456	
   457	        val_t dist = rayTriangleIntersect(from, dirNorm, tri.a, tri.b, tri.c);
   458	        if (dist > EPSILON && dist < rayLen - EPSILON) {
   459	            return true;  // Ray is blocked
   460	        }
   461	        return false;
   462	    });
   463	}

   465	// ============================================================================
   466	// Form Factor Computation (Kij)
   467	// ============================================================================
   468	
   469	// Compute the cosine of angle between vector and triangle normal
   470	val_t cosPhi(const Vec3& v, const Vec3& normal) {
   471	    val_t vNorm = v.norm();
   472	    if (vNorm <= EPSILON) return ZERO;
   473	    return std::max(ZERO, v.dot(normal) / vNorm);
   474	}
   475	
   476	// Compute form factor Kij between triangle i (receiver) and triangle j (emitter)
   477	val_t computeKij(size_t idxI, size_t idxJ,
   478	                  const std::vector<Triangle>& triangles,
   479	                  const Octree& octree,
   480	                  RandomGenerator& rng) {
   481	    const Triangle& triI = triangles[idxI];
   482	    const Triangle& triJ = triangles[idxJ];
   483	
   484	    // Cull triangles facing the same direction
   485	    if (triI.normal().dot(triJ.normal()) > 0.99f) return ZERO;
   486	
   487	    val_t kij = ZERO;
   488	
   489	    for (int r = 0; r < NUM_RAYS; ++r) {
   490	        Vec3 pI = randomPointInTriangle(triI, rng);
   491	        Vec3 pJ = randomPointInTriangle(triJ, rng);
   492	
   493	        if (isRayBlocked(pI, pJ, octree, idxI, idxJ)) continue;
   494	
   495	        Vec3 v = pJ - pI;
   496	        val_t distSqr = v.squaredNorm();
   497	        if (distSqr < EPSILON) continue;
   498	
   499	        val_t cosPhiI = cosPhi(v, triI.normal());
   500	        val_t cosPhiJ = cosPhi(-v, triJ.normal());
   501	
   502	        if (cosPhiI <= ZERO || cosPhiJ <= ZERO) continue;
   503	
   504	        kij += (cosPhiI * cosPhiJ) / (PI * distSqr);
   505	    }
   506	
   507	    return kij * INV_NUM_RAYS;
   508	}

   510	// ============================================================================
   511	// Tau (time delay) Computation
   512	// ============================================================================
   513	
   514	int computeTau(const Triangle& triI, const Triangle& triJ) {
   515	    val_t dist = (triI.center() - triJ.center()).norm();
   516	    return static_cast<int>(std::ceil(dist * INV_WAVE_SPEED));
   517	}

   523	struct SimulationState {
   524	    size_t numTriangles;
   525	    size_t numTimesteps;
   526	
   527	    std::vector<Triangle> triangles;
   528	    std::vector<val_t> areas;       // Area of each triangle
   529	    std::vector<val_t> rho;         // Reflectivity (0.0 to 1.0)
   530	    std::vector<val_t> kij;         // Form factors (N x N matrix, row-major)
   531	    std::vector<int> tau;           // Time delays (N x N matrix, row-major)
   532	    std::vector<val_t> radE;        // Emission radiosity (T x N matrix)
   533	    std::vector<val_t> radB;        // Reflected radiosity (T x N matrix)
   534	    std::vector<val_t> distances;   // Computed distances from source
   535	
   536	    Octree octree;                  // Spatial acceleration structure
   537	
   538	    size_t sourceIndex;
   539	
   540	    size_t idx2d(size_t i, size_t j) const { return i * numTriangles + j; }
   541	    size_t idxTN(size_t t, size_t n) const { return t * numTriangles + n; }
   542	};

   544	// ============================================================================
   545	// Initialization
   546	// ============================================================================
   547	
   548	void initializeSimulation(SimulationState& state, int subdivisions, size_t timesteps,
   549	                          size_t sourceIdx, val_t reflectivity) {
   550	    // Generate mesh
   551	    IcosphereMesh mesh(subdivisions, 10.0f);  // Radius 10 units
   552	    state.triangles = std::move(mesh.triangles);
   553	    state.numTriangles = state.triangles.size();
   554	    state.numTimesteps = timesteps;
   555	    state.sourceIndex = sourceIdx % state.numTriangles;
   556	
   557	    printf("Generated icosphere mesh with %zu triangles\n", state.numTriangles);
   558	
   559	    // Build octree for spatial acceleration
   560	    printf("Building octree...\n");
   561	    state.octree.build(state.triangles);
   562	
   563	    // Initialize areas
   564	    state.areas.resize(state.numTriangles);
   565	    for (size_t i = 0; i < state.numTriangles; ++i) {
   566	        state.areas[i] = state.triangles[i].area();
   567	    }
   568	
   569	    // Initialize reflectivity
   570	    state.rho.resize(state.numTriangles, reflectivity);
   571	
   572	    // Initialize matrices
   573	    state.kij.resize(state.numTriangles * state.numTriangles, ZERO);
   574	    state.tau.resize(state.numTriangles * state.numTriangles, 0);
   575	    state.radE.resize(timesteps * state.numTriangles, ZERO);
   576	    state.radB.resize(timesteps * state.numTriangles, ZERO);
   577	    state.distances.resize(state.numTriangles, ZERO);
   578	
   579	    // Set source emission (active for first half of timesteps)
   580	    size_t timeOn = 0;
   581	    size_t timeOff = timesteps / 2;
   582	    for (size_t t = timeOn; t < timeOff; ++t) {
   583	        state.radE[state.idxTN(t, state.sourceIndex)] = 1.0f;
   584	    }
   585	}

   587	// ============================================================================
   588	// Precomputation Phase
   589	// ============================================================================
   590	
   591	void computeFormFactors(SimulationState& state) {
   592	    printf("Computing form factors (Kij)...\n");
   593	    RandomGenerator rng(42);
   594	
   595	    for (size_t i = 0; i < state.numTriangles; ++i) {
   596	        for (size_t j = 0; j < state.numTriangles; ++j) {
   597	            if (i == j) continue;
   598	            state.kij[state.idx2d(i, j)] = computeKij(
   599	                i, j, state.triangles, state.octree, rng);
   600	        }
   601	        if ((i + 1) % 100 == 0 || i + 1 == state.numTriangles) {
   602	            printf("  Progress: %zu/%zu triangles\n", i + 1, state.numTriangles);
   603	        }
   604	    }
   605	}

   607	void computeTimeDelays(SimulationState& state) {
   608	    printf("Computing time delays (Tau)...\n");
   609	
   610	    for (size_t i = 0; i < state.numTriangles; ++i) {
   611	        for (size_t j = 0; j < state.numTriangles; ++j) {
   612	            if (i == j) continue;
   613	            state.tau[state.idx2d(i, j)] = computeTau(
   614	                state.triangles[i], state.triangles[j]);
   615	        }
   616	    }
   617	}

   619	// ============================================================================
   620	// Simulation Phase (Wave Propagation)
   621	// ============================================================================
   622	
   623	void runSimulation(SimulationState& state) {
   624	    printf("Running wave propagation simulation...\n");
   625	
   626	    for (size_t t = 0; t < state.numTimesteps; ++t) {
   627	        for (size_t i = 0; i < state.numTriangles; ++i) {
   628	            val_t sumB = ZERO;
   629	
   630	            for (size_t j = 0; j < state.numTriangles; ++j) {
   631	                if (i == j) continue;
   632	
   633	                int tauij = state.tau[state.idx2d(i, j)];
   634	
   635	                // Skip if wave hasn't yet propagated from j to i
   636	                if (static_cast<int>(t) < tauij) continue;
   637	
   638	                val_t kij = state.kij[state.idx2d(i, j)];
   639	                if (kij <= ZERO) continue;
   640	
   641	                // Get radiosity from source triangle at time when emission occurred
   642	                size_t srcTime = t - static_cast<size_t>(tauij);
   643	                val_t radJ = state.radB[state.idxTN(srcTime, j)];
   644	                if (radJ <= ZERO) continue;
   645	
   646	                // Accumulate contribution: form factor * area * source radiosity
   647	                sumB += std::min(kij * state.areas[j], ONE) * radJ;
   648	            }
   649	
   650	            // Update radiosity: reflection + emission
   651	            state.radB[state.idxTN(t, i)] = state.rho[i] * sumB + state.radE[state.idxTN(t, i)];
   652	        }
   653	
   654	        if ((t + 1) % 10 == 0 || t + 1 == state.numTimesteps) {
   655	            printf("  Timestep %zu/%zu\n", t + 1, state.numTimesteps);
   656	        }
   657	    }
   658	}

   660	// ============================================================================
   661	// Distance Computation (Cross-Correlation)
   662	// ============================================================================
   663	
   664	void computeDistances(SimulationState& state) {
   665	    printf("Computing distances via cross-correlation...\n");
   666	
   667	    for (size_t i = 0; i < state.numTriangles; ++i) {
   668	        val_t maxCorr = ZERO;
   669	        int bestT = 0;
   670	
   671	        // Discrete cross-correlation to find time delay
   672	        for (size_t t = 0; t < state.numTimesteps; ++t) {
   673	            val_t sum = ZERO;
   674	
   675	            for (size_t tt = t; tt < state.numTimesteps; ++tt) {
   676	                val_t pB = state.radB[state.idxTN(tt, i)];
   677	                val_t pS = state.radB[state.idxTN(tt - t, state.sourceIndex)];
   678	                sum += pS * pB;
   679	            }
   680	
   681	            if (sum > maxCorr) {
   682	                maxCorr = sum;
   683	                bestT = static_cast<int>(t);
   684	            }
   685	        }
   686	
   687	        state.distances[i] = WAVE_SPEED * static_cast<val_t>(bestT);
   688	    }
   689	}

   691	// ============================================================================
   692	// Validation
   693	// ============================================================================
   694	
   695	bool validateResults(const SimulationState& state) {
   696	    printf("\nValidation:\n");
   697	
   698	    // Check that distances are non-negative
   699	    bool allNonNegative = true;
   700	    val_t minDist = std::numeric_limits<val_t>::max();
   701	    val_t maxDist = std::numeric_limits<val_t>::lowest();
   702	    val_t sumDist = ZERO;
   703	    int nonZeroCount = 0;
   704	
   705	    for (size_t i = 0; i < state.numTriangles; ++i) {
   706	        val_t d = state.distances[i];
   707	        if (d < 0) {
   708	            allNonNegative = false;
   709	            printf("  ERROR: Negative distance at triangle %zu: %f\n", i, d);
   710	        }
   711	        if (!std::isfinite(d)) {
   712	            printf("  ERROR: Non-finite distance at triangle %zu: %f\n", i, d);
   713	            return false;
   714	        }
   715	        minDist = std::min(minDist, d);
   716	        maxDist = std::max(maxDist, d);
   717	        sumDist += d;
   718	        if (d > EPSILON) nonZeroCount++;
   719	    }
   720	
   721	    printf("  Distance range: [%.4f, %.4f]\n", minDist, maxDist);
   722	    printf("  Average distance: %.4f\n", sumDist / static_cast<val_t>(state.numTriangles));
   723	    printf("  Non-zero distances: %d/%zu\n", nonZeroCount, state.numTriangles);
   724	
   725	    // Check source distance is zero or very small
   726	    val_t srcDist = state.distances[state.sourceIndex];
   727	    if (srcDist > WAVE_SPEED * 2) {
   728	        printf("  WARNING: Source triangle distance is non-zero: %.4f\n", srcDist);
   729	    }
   730	
   731	    // Check radiosity propagation (some triangles should have received energy)
   732	    int receivedEnergy = 0;
   733	    for (size_t i = 0; i < state.numTriangles; ++i) {
   734	        for (size_t t = 0; t < state.numTimesteps; ++t) {
   735	            if (state.radB[state.idxTN(t, i)] > EPSILON) {
   736	                receivedEnergy++;
   737	                break;
   738	            }
   739	        }
   740	    }
   741	
   742	    printf("  Triangles receiving energy: %d/%zu\n", receivedEnergy, state.numTriangles);
   743	
   744	    if (receivedEnergy == 0) {
   745	        printf("  ERROR: No triangles received energy - simulation failed\n");
   746	        return false;
   747	    }
   748	
   749	    // Check Kij matrix (should have some non-zero entries)
   750	    int nonZeroKij = 0;
   751	    for (size_t i = 0; i < state.numTriangles * state.numTriangles; ++i) {
   752	        if (state.kij[i] > EPSILON) nonZeroKij++;
   753	    }
   754	    printf("  Non-zero form factors: %d/%zu (%.2f%%)\n",
   755	           nonZeroKij, state.numTriangles * state.numTriangles,
   756	           100.0f * nonZeroKij / static_cast<val_t>(state.numTriangles * state.numTriangles));
   757	
   758	    if (nonZeroKij == 0) {
   759	        printf("  ERROR: All form factors are zero - visibility computation failed\n");
   760	        return false;
   761	    }
   762	
   763	    if (!allNonNegative) {
   764	        return false;
   765	    }
   766	
   767	    printf("  Validation: PASSED\n");
   768	    return true;
   769	}

   773	// ============================================================================
   774	
   775	uint64_t computeHash(const SimulationState& state) {
   776	    uint64_t hash = 0;
   777	    for (size_t i = 0; i < state.numTriangles; ++i) {
   778	        const uint32_t* ptr = reinterpret_cast<const uint32_t*>(&state.distances[i]);
   779	        hash ^= (static_cast<uint64_t>(*ptr) + i) * 0x9e3779b97f4a7c15ULL;
   780	    }
   781	    return hash;
   782	}

// ============================================================================
// Random Number Generation
// ============================================================================

class RandomGenerator {
    std::mt19937 rng;
    std::uniform_real_distribution<val_t> dist;
public:
    explicit RandomGenerator(uint32_t seed = 42) : rng(seed), dist(0.0f, 1.0f) {}
    val_t rand() { return dist(rng); }
};

// Generate a random point inside a triangle using barycentric coordinates
Vec3 randomPointInTriangle(const Triangle& t, RandomGenerator& rng) {
    val_t u = rng.rand();
    val_t v = rng.rand();
    if (u + v > 1.0f) {
        u = 1.0f - u;
        v = 1.0f - v;
    }
    Vec3 ab = t.b - t.a;
    Vec3 ac = t.c - t.a;
    return t.a + ab * u + ac * v;
}

// ============================================================================
// Ray-Triangle Intersection (Möller-Trumbore algorithm)
// ============================================================================

val_t rayTriangleIntersect(const Vec3& orig, const Vec3& dir,
                            const Vec3& v0, const Vec3& v1, const Vec3& v2) {
    Vec3 e1 = v1 - v0;
    Vec3 e2 = v2 - v0;
    Vec3 pvec = dir.cross(e2);
    val_t det = e1.dot(pvec);

    if (std::abs(det) < EPSILON) return std::numeric_limits<val_t>::max();

    val_t invDet = 1.0f / det;
    Vec3 tvec = orig - v0;
    val_t u = tvec.dot(pvec) * invDet;
    if (u < 0.0f || u > 1.0f) return std::numeric_limits<val_t>::max();

    Vec3 qvec = tvec.cross(e1);
    val_t v = dir.dot(qvec) * invDet;
    if (v < 0.0f || u + v > 1.0f) return std::numeric_limits<val_t>::max();

    return e2.dot(qvec) * invDet;
}

// ============================================================================
// Visibility Testing (Octree-accelerated)
// ============================================================================

// Check if a ray between two triangles is blocked by any other triangle
// Uses octree for O(log N) instead of O(N) search
bool isRayBlocked(const Vec3& from, const Vec3& to,
                  const Octree& octree,
                  size_t srcTriIdx, size_t dstTriIdx) {
    Vec3 dir = to - from;
    val_t rayLen = dir.norm();
    if (rayLen < EPSILON) return true;
    Vec3 dirNorm = dir / rayLen;

    return octree.applyToTris(from, to, [&](size_t idx, const Triangle& tri) {
        if (idx == srcTriIdx || idx == dstTriIdx) return false;

        val_t dist = rayTriangleIntersect(from, dirNorm, tri.a, tri.b, tri.c);
        if (dist > EPSILON && dist < rayLen - EPSILON) {
            return true;  // Ray is blocked
        }
        return false;
    });
}

// ============================================================================
// Form Factor Computation (Kij)
// ============================================================================

// Compute the cosine of angle between vector and triangle normal
val_t cosPhi(const Vec3& v, const Vec3& normal) {
    val_t vNorm = v.norm();
    if (vNorm <= EPSILON) return ZERO;
    return std::max(ZERO, v.dot(normal) / vNorm);
}

// Compute form factor Kij between triangle i (receiver) and triangle j (emitter)
val_t computeKij(size_t idxI, size_t idxJ,
                  const std::vector<Triangle>& triangles,
                  const Octree& octree,
                  RandomGenerator& rng) {
    const Triangle& triI = triangles[idxI];
    const Triangle& triJ = triangles[idxJ];

    // Cull triangles facing the same direction
    if (triI.normal().dot(triJ.normal()) > 0.99f) return ZERO;

    val_t kij = ZERO;

    for (int r = 0; r < NUM_RAYS; ++r) {
        Vec3 pI = randomPointInTriangle(triI, rng);
        Vec3 pJ = randomPointInTriangle(triJ, rng);

        if (isRayBlocked(pI, pJ, octree, idxI, idxJ)) continue;

        Vec3 v = pJ - pI;
        val_t distSqr = v.squaredNorm();
        if (distSqr < EPSILON) continue;

        val_t cosPhiI = cosPhi(v, triI.normal());
        val_t cosPhiJ = cosPhi(-v, triJ.normal());

        if (cosPhiI <= ZERO || cosPhiJ <= ZERO) continue;

        kij += (cosPhiI * cosPhiJ) / (PI * distSqr);
    }

    return kij * INV_NUM_RAYS;
}

// Keep the serial MT19937 stream position in draw units. Each rank can seek
// directly to its first draw even when preceding ranks own billions of draws.
class RandomGenerator {
    static constexpr int words = 624;
    static constexpr int lag = 397;
    static constexpr int degree = 19937;
    using State = std::array<uint32_t, words>;
    State state{};
    int cursor = 0;

    static uint32_t twistWord(uint32_t a, uint32_t b, uint32_t c) {
        uint32_t y = (a & 0x80000000U) | (b & 0x7fffffffU);
        return c ^ (y >> 1) ^ ((0U - (y & 1U)) & 0x9908b0dfU);
    }

    void twist() {
        for (int i = 0; i < words - lag; ++i)
            state[i] = twistWord(state[i], state[i + 1], state[i + lag]);
        for (int i = words - lag; i < words - 1; ++i)
            state[i] = twistWord(state[i], state[i + 1], state[i + lag - words]);
        state.back() = twistWord(state.back(), state.front(), state[lag - 1]);
    }

    // Compute x^offset modulo the characteristic polynomial of the one-word
    // MT19937 transition over GF(2), using binary exponentiation. The listed
    // exponents are its nonzero coefficients below the leading x^19937.
    static constexpr int coefficients[] = {
        0,1189,1416,1585,1643,1870,2493,2773,3000,3227,3454,3681,3908,
        4135,4362,4753,5661,6337,6569,7129,7477,7525,7583,7752,7979,
        8206,9505,9901,9969,10128,10693,10761,10920,11089,11147,11157,
        11215,11321,11374,11384,11485,11611,11712,11717,11838,11881,
        11944,11997,12277,12335,12393,12504,12509,12620,12673,12731,
        12736,12789,12905,12958,12963,13137,13185,13190,13243,13301,
        13412,13528,13533,13639,13697,13760,13813,13866,14093,14151,
        14209,14320,14325,14436,14547,14552,14605,14721,14774,14779,
        14953,15001,15006,15059,15117,15228,15344,15349,15455,15513,
        15576,15629,15682,15909,15967,16025,16136,16141,16252,16363,
        16368,16421,16537,16590,16595,16817,16822,16875,16933,17044,
        17160,17271,17329,17445,17498,17725,17783,17841,17952,18068,
        18179,18237,18406,18633,18691,18860,19087,19314
    };
    using Polynomial = std::array<uint64_t, (degree + 63) / 64>;

    static Polynomial jumpPolynomial(uint64_t offset) {
        Polynomial poly{};
        poly[0] = 1;
        for (int bit = 63 - __builtin_clzll(offset); bit >= 0; --bit) {
            std::array<uint64_t, 2 * Polynomial{}.size() + 1> square{};
            int extra = (offset >> bit) & 1U;
            for (int i = 0; i < degree; ++i) {
                if ((poly[i / 64] >> (i % 64)) & 1U) {
                    int exponent = 2 * i + extra;
                    square[exponent / 64] |= uint64_t{1} << (exponent % 64);
                }
            }
            // Reduce 64 consecutive high coefficients together. The second
            // highest term lies 623 positions below the leading coefficient.
            for (int base = ((degree - 1) / 64) * 64; base >= 0; base -= 64) {
                int word = (degree + base) / 64;
                constexpr int shift = degree % 64;
                uint64_t high = (square[word] >> shift) |
                                (square[word + 1] << (64 - shift));
                if (!high) continue;
                for (int term : coefficients) {
                    int position = (base + term) / 64;
                    int bits = term % 64;
                    square[position] ^= high << bits;
                    if (bits) square[position + 1] ^= high >> (64 - bits);
                }
            }
            std::copy_n(square.begin(), poly.size(), poly.begin());
            poly.back() &= (uint64_t{1} << (degree % 64)) - 1;
        }
        return poly;
    }

    void jump(uint64_t offset) {
        Polynomial poly = jumpPolynomial(offset);
        // Duplicate the circular state so every polynomial term can be XORed
        // in logical stream order with a single contiguous vector loop.
        std::array<uint32_t, 2 * words> ring;
        std::copy(state.begin(), state.end(), ring.begin());
        std::copy(state.begin(), state.end(), ring.begin() + words);
        state.fill(0);
        int head = 0;
        for (int i = 0; i < degree; ++i) {
            if ((poly[i / 64] >> (i % 64)) & 1U) {
                for (int j = 0; j < words; ++j) state[j] ^= ring[head + j];
            }
            uint32_t next = twistWord(ring[head], ring[head + 1], ring[head + lag]);
            ring[head] = ring[head + words] = next;
            if (++head == words) head = 0;
        }
    }

public:
    explicit RandomGenerator(uint64_t firstDraw) {
        state.front() = 42U;
        for (int i = 1; i < words; ++i)
            state[i] = 1812433253U * (state[i-1] ^ (state[i-1] >> 30)) + i;
        twist();
        if (firstDraw > (1U << 20)) {
            jump(firstDraw);
        } else {
            for (uint64_t i = 0; i < firstDraw / words; ++i) twist();
            cursor = firstDraw % words;
        }
    }

    void fill(float* output, size_t count) {
        while (count) {
            if (cursor == words) {
                twist();
                cursor = 0;
            }
            int n = static_cast<int>(std::min(count, size_t(words - cursor)));
            #pragma omp simd
            for (int i = 0; i < n; ++i) {
                uint32_t y = state[cursor + i];
                y ^= y >> 11;
                y ^= (y << 7) & 0x9d2c5680U;
                y ^= (y << 15) & 0xefc60000U;
                y ^= y >> 18;
                // std::uniform_real_distribution<float>(0,1) on libstdc++.
                output[i] = std::min(static_cast<float>(y) * 0x1p-32f, 0x1.fffffep-1f);
            }
            output += n;
            count -= n;
            cursor += n;
        }
    }
};

// ============================================================================
// Ray-Triangle Intersection (Möller-Trumbore algorithm)
// ============================================================================

__device__ val_t rayTriangleIntersect(const Vec3& orig, const Vec3& dir,
                            const Vec3& v0, const Vec3& v1, const Vec3& v2) {
    Vec3 e1 = v1 - v0;
    Vec3 e2 = v2 - v0;
    Vec3 pvec = dir.cross(e2);
    val_t det = e1.dot(pvec);

    if (std::abs(det) < EPSILON) return std::numeric_limits<val_t>::max();

    val_t invDet = 1.0f / det;
    Vec3 tvec = orig - v0;
    val_t u = tvec.dot(pvec) * invDet;
    if (u < 0.0f || u > 1.0f) return std::numeric_limits<val_t>::max();

    Vec3 qvec = tvec.cross(e1);
    val_t v = dir.dot(qvec) * invDet;
    if (v < 0.0f || u + v > 1.0f) return std::numeric_limits<val_t>::max();

    return e2.dot(qvec) * invDet;
}

// A preorder layout lets a GPU ray skip a missed subtree without a stack.
struct FlatNode {
    Vec3 center, halfExtent;
    int end, firstTriangle, count;
};

void flatten(const Octree& tree, std::vector<FlatNode>& nodes, std::vector<int>& indices) {
    const size_t start = nodes.size();
    nodes.push_back({tree.center, tree.halfExtent, 0, static_cast<int>(indices.size()),
                     static_cast<int>(tree.triangleIndices.size())});
    indices.insert(indices.end(), tree.triangleIndices.begin(), tree.triangleIndices.end());
    for (const auto& child : tree.children) {
        if (child) flatten(*child, nodes, indices);
    }
    nodes[start].end = static_cast<int>(nodes.size());
}

__device__ bool intersectsBox(const FlatNode& node, Vec3 midpoint, Vec3 d, Vec3 ad) {
    const Vec3 c = midpoint - node.center;
    const Vec3 h = node.halfExtent;
    return fabsf(c.x) <= h.x + ad.x && fabsf(c.y) <= h.y + ad.y &&
           fabsf(c.z) <= h.z + ad.z &&
           fabsf(d.y * c.z - d.z * c.y) <= h.y * ad.z + h.z * ad.y + EPSILON &&
           fabsf(d.z * c.x - d.x * c.z) <= h.z * ad.x + h.x * ad.z + EPSILON &&
           fabsf(d.x * c.y - d.y * c.x) <= h.x * ad.y + h.y * ad.x + EPSILON;
}

__device__ bool isRayBlocked(Vec3 from, Vec3 dir, float len, int i, int j,
                             const Triangle* triangles, const FlatNode* nodes, const int* indices) {
    const Vec3 norm = dir / len;
    const Vec3 d = dir * 0.5f;
    const Vec3 midpoint = from + d;
    const Vec3 ad{fabsf(d.x), fabsf(d.y), fabsf(d.z)};
    const int end = nodes[0].end;
    for (int k = 0; k < end; ) {
        const FlatNode node = nodes[k];
        if (k && !intersectsBox(node, midpoint, d, ad)) {
            k = node.end;
            continue;
        }
        for (int q = 0; q < node.count; ++q) {
            const int tri = indices[node.firstTriangle + q];
            if (tri == i || tri == j) continue;
            const Triangle& t = triangles[tri];
            const float dist = rayTriangleIntersect(from, norm, t.a, t.b, t.c);
            if (dist > EPSILON && dist < len - EPSILON) return true;
        }
        ++k;
    }
    return false;
}

__device__ Vec3 pointOnTriangle(const Triangle& t, float u, float v) {
    if (u + v > 1.0f) {
        u = 1.0f - u;
        v = 1.0f - v;
    }
    return t.a + (t.b - t.a) * u + (t.c - t.a) * v;
}

// Each bit represents one source in a row. Only set bits consume RNG draws.
struct PairGroup {
    unsigned long long firstSample;
    uint32_t mask;
};

constexpr int FORM_THREADS = 128;
__global__ void computePairBatch(size_t firstGroup, size_t groupCount, int groupsPerRow,
                                 int rows, int firstRow, int n, const PairGroup* groups,
                                 unsigned long long sampleBase, const float4* samples,
                                 const Triangle* triangles, const FlatNode* nodes,
                                 const int* indices, const float* areas, float* weights,
                                 unsigned char* delays, unsigned long long* nonZero) {
    __shared__ float contributions[FORM_THREADS];
    const int ray = threadIdx.x % NUM_RAYS;
    const unsigned pair = (blockIdx.x * blockDim.x + threadIdx.x) / NUM_RAYS;
    if (pair / 32 >= groupCount) return;
    const size_t groupId = firstGroup + pair / 32;
    const int local = groupId / groupsPerRow;
    const int i = firstRow + local;
    const int bit = pair % 32;
    const int j = (groupId % groupsPerRow) * 32 + bit;
    if (j >= n) return;
    const PairGroup group = groups[groupId];
    const uint32_t flag = 1u << bit;
    float contribution = 0.0f;
    if (group.mask & flag) {
        const size_t sample = group.firstSample - sampleBase + __popc(group.mask & (flag - 1));
        const float4 r = samples[sample * NUM_RAYS + ray];
        const Vec3 from = pointOnTriangle(triangles[i], r.x, r.y);
        const Vec3 to = pointOnTriangle(triangles[j], r.z, r.w);
        const Vec3 dir = to - from;
        const float sqr = dir.squaredNorm();
        const float len = sqrtf(sqr);
        if (sqr >= EPSILON) {
            const float ci = dir.dot(triangles[i].normal()) / len;
            const float cj = (-dir).dot(triangles[j].normal()) / len;
            if (ci > ZERO && cj > ZERO && !isRayBlocked(from, dir, len, i, j, triangles, nodes, indices))
                contribution = ci * cj / (PI * sqr);
        }
    }
    contributions[threadIdx.x] = contribution;
    // A half-warp shares one triangle pair. Keep the serial ray sum order.
    const unsigned mask = (threadIdx.x & NUM_RAYS) ? 0xffff0000u : 0x0000ffffu;
    __syncwarp(mask);
    if (ray == 0) {
        float kij = ZERO;
        for (int k = 0; k < NUM_RAYS; ++k) kij += contributions[threadIdx.x + k];
        kij *= INV_NUM_RAYS;
        const size_t dst = static_cast<size_t>(j) * rows + local;
        weights[dst] = fminf(kij * areas[j], ONE);
        delays[dst] = static_cast<unsigned char>(
            ceilf((triangles[i].center() - triangles[j].center()).norm() * INV_WAVE_SPEED));
        if (nonZero && kij > EPSILON) atomicAdd(nonZero, 1ULL);
    }
}

struct SimulationState {

void computeFormFactors(SimulationState& state) {
    const int n = static_cast<int>(state.numTriangles);
    const int groupsPerRow = (n + 31) / 32;
    const size_t groupCount = static_cast<size_t>(state.rows) * groupsPerRow;
    std::vector<PairGroup> groups(groupCount);
    std::vector<unsigned long long> rowCounts(state.rows + 1, 0);
    #pragma omp parallel for schedule(static)
    for (int local = 0; local < state.rows; ++local) {
        const int i = state.first + local;
        const Vec3 normal = state.triangles[i].normal();
        unsigned long long count = 0;
        for (int g = 0; g < groupsPerRow; ++g) {
            uint32_t mask = 0;
            const int start = g * 32;
            const int width = std::min(32, n - start);
            for (int k = 0; k < width; ++k) {
                const int j = start + k;
                if (i != j && normal.dot(state.triangles[j].normal()) <= 0.99f)
                    mask |= 1u << k;
            }
            groups[static_cast<size_t>(local) * groupsPerRow + g] = {count, mask};
            count += __builtin_popcount(mask);
        }
        rowCounts[local + 1] = count;
    }
    for (int i = 1; i <= state.rows; ++i) rowCounts[i] += rowCounts[i - 1];
    #pragma omp parallel for schedule(static)
    for (int local = 0; local < state.rows; ++local) {
        for (int g = 0; g < groupsPerRow; ++g)
            groups[static_cast<size_t>(local) * groupsPerRow + g].firstSample += rowCounts[local];
    }
    unsigned long long firstSample = 0;
    MPI_Exscan(&rowCounts.back(), &firstSample, 1, MPI_UNSIGNED_LONG_LONG, MPI_SUM, MPI_COMM_WORLD);
    int rank;
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    if (rank == 0) firstSample = 0;  // MPI_Exscan leaves the first result undefined.
    if (!state.rows) return;
    RandomGenerator rng(firstSample * (4 * NUM_RAYS));

    std::vector<FlatNode> nodes;
    std::vector<int> indices;
    flatten(state.octree, nodes, indices);
    DeviceBuffer<FlatNode> gpuNodes(nodes.size());
    DeviceBuffer<int> gpuIndices(indices.size());
    DeviceBuffer<Triangle> gpuTriangles(n);
    DeviceBuffer<float> gpuAreas(n);
    DeviceBuffer<PairGroup> gpuGroups(groupCount);
    CUDA_CHECK(cudaMemcpy(gpuNodes.data(), nodes.data(), nodes.size() * sizeof(FlatNode), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(gpuIndices.data(), indices.data(), indices.size() * sizeof(int), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(gpuTriangles.data(), state.triangles.data(), n * sizeof(Triangle), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(gpuAreas.data(), state.areas.data(), n * sizeof(float), cudaMemcpyHostToDevice));
    CUDA_CHECK(cudaMemcpy(gpuGroups.data(), groups.data(), groupCount * sizeof(PairGroup), cudaMemcpyHostToDevice));

    DeviceBuffer<unsigned long long> nonZero(state.validate ? 1 : 0);
    if (state.validate) CUDA_CHECK(cudaMemset(nonZero.data(), 0, sizeof(unsigned long long)));

    // While the GPU traces a batch, fill the pinned input for the next one.
    constexpr size_t groupsPerBatch = 512;
    const size_t capacity = std::min(groupCount, groupsPerBatch) * 32 * NUM_RAYS * 4;
    DeviceBuffer<float> gpuSamples(capacity);
    float* hostSamples;
    CUDA_CHECK(cudaMallocHost(&hostSamples, capacity * sizeof(float)));
    cudaEvent_t copied;
    CUDA_CHECK(cudaEventCreateWithFlags(&copied, cudaEventDisableTiming));
    for (size_t first = 0; first < groupCount; first += groupsPerBatch) {
        const size_t last = std::min(groupCount, first + groupsPerBatch);
        const auto base = groups[first].firstSample;
        const auto end = last == groupCount ? rowCounts.back() : groups[last].firstSample;
        const size_t draws = static_cast<size_t>(end - base) * NUM_RAYS * 4;
        if (first) CUDA_CHECK(cudaEventSynchronize(copied));
        rng.fill(hostSamples, draws);
        CUDA_CHECK(cudaMemcpyAsync(gpuSamples.data(), hostSamples, draws * sizeof(float), cudaMemcpyHostToDevice));
        CUDA_CHECK(cudaEventRecord(copied));
        const int blocks = static_cast<int>((last - first) * 32 * NUM_RAYS / FORM_THREADS);
        computePairBatch<<<blocks, FORM_THREADS>>>(
            first, last - first, groupsPerRow, state.rows, state.first, n, gpuGroups.data(), base,
            reinterpret_cast<const float4*>(gpuSamples.data()), gpuTriangles.data(), gpuNodes.data(),
            gpuIndices.data(), gpuAreas.data(), state.weights.data(), state.delays.data(), nonZero.data());
        CUDA_CHECK(cudaGetLastError());
    }
    CUDA_CHECK(cudaDeviceSynchronize());
    if (state.validate)
        CUDA_CHECK(cudaMemcpy(&state.nonZeroKij, nonZero.data(), sizeof(state.nonZeroKij), cudaMemcpyDeviceToHost));
    CUDA_CHECK(cudaEventDestroy(copied));
    CUDA_CHECK(cudaFreeHost(hostSamples));
}

__global__ void findMinimumDelay(const unsigned char* delays, const float* weights,
                                 size_t pairs, int* minimum) {
    int best = MAX_TIME_BATCH;
    for (size_t p = blockIdx.x * blockDim.x + threadIdx.x; p < pairs;
         p += gridDim.x * blockDim.x) {
        if (weights[p] > ZERO) best = min(best, int(delays[p]));

}

template<bool useRing>
__global__ void propagate(const float* __restrict__ weights, const unsigned char* delays,
                          const float* previous, float* history, int first, int rows,
                          int n, int t0, int emissionEnd, int source, float rho) {
    int row = blockIdx.x * blockDim.x + threadIdx.x;
    if (row >= rows) return;
    int t = t0 + blockIdx.y;
    float sum = ZERO;
    // Threads owning adjacent receivers read adjacent matrix entries. Visit
    // emitters in mesh order to retain the serial floating-point accumulation.
    for (int j = 0; j < n; ++j) {
        size_t pair = size_t(j) * rows + row;
        int delay = delays[pair];
        if (delay > t) continue;
        float w = weights[pair];
        if (w <= ZERO) continue;
        int slot = useRing ? ((t - delay) & (HISTORY_SLOTS - 1)) : t - delay;
        float incoming = previous[size_t(slot) * n + j];
        if (incoming <= ZERO) continue;
        sum += w * incoming;
    }
    float emission = first + row == source && t < emissionEnd ? ONE : ZERO;
    history[size_t(t) * rows + row] = rho * sum + emission;
}

// MPI concatenates each rank's time-major batch. Transpose just the rank/time
// dimensions into time-major global snapshots for the next propagation batch.
__global__ void unpackSnapshots(const float* packed, const int2* owners, float* ring,
                                float* sourceSignal, int source, int n, int t0, int batch) {
    int j = blockIdx.x * blockDim.x + threadIdx.x;
    if (j >= n) return;
    int t = t0 + blockIdx.y;
    int2 part = owners[j]; // first global triangle, local count
    float value = packed[size_t(part.x) * batch + blockIdx.y * part.y + j - part.x];
    ring[size_t(t & (HISTORY_SLOTS - 1)) * n + j] = value;
    if (j == source) sourceSignal[t] = value;
}

void runSimulation(SimulationState& state) {
    if (state.rank == 0) printf("Running wave propagation simulation...\n");
    int n = int(state.numTriangles);
    int rows = state.rows;
    int timesteps = int(state.numTimesteps);
    state.history.allocate(size_t(timesteps) * rows);
    if (timesteps == 0) return;

    // Every positive form factor couples distinct triangle centers, so its
    // delay is positive. Timesteps within a minimum-delay window are independent.
    size_t pairs = size_t(rows) * n;
    int batchLimit = MAX_TIME_BATCH;
    if (pairs) {
        DeviceBuffer<int> minimum(1);
        CUDA_CHECK(cudaMemcpy(minimum.data(), &batchLimit, sizeof(int), cudaMemcpyHostToDevice));
        int blocks = int(std::min<size_t>((pairs + 255) / 256, 1024));
        findMinimumDelay<<<blocks, 256>>>(state.delays.data(), state.weights.data(), pairs,
                                           minimum.data());
        CUDA_CHECK(cudaGetLastError());
        CUDA_CHECK(cudaMemcpy(&batchLimit, minimum.data(), sizeof(int), cudaMemcpyDeviceToHost));
    }

    bool distributed = state.ranks > 1;
    DeviceBuffer<float> ring, packed;
    DeviceBuffer<int2> deviceOwners;
    float* exchange = nullptr;
    std::vector<int> counts(state.ranks), offsets(state.ranks);
    if (distributed) {
        MPI_Allreduce(MPI_IN_PLACE, &batchLimit, 1, MPI_INT, MPI_MIN, MPI_COMM_WORLD);
        batchLimit = std::min(timesteps, batchLimit);
        size_t capacity = size_t(n) * batchLimit;
        ring.allocate(size_t(n) * std::min(timesteps, HISTORY_SLOTS));
        packed.allocate(capacity);
        state.sourceSignal.allocate(timesteps);
        CUDA_CHECK(cudaMallocHost(&exchange, capacity * sizeof(float)));
        std::vector<int2> owners(n);
        for (int r = 0; r < state.ranks; ++r) {
            int first = state.offsets[r], count = state.counts[r];
            std::fill_n(owners.begin() + first, count, make_int2(first, count));
        }
        deviceOwners.allocate(n);
        CUDA_CHECK(cudaMemcpy(deviceOwners.data(), owners.data(), n * sizeof(int2),
                              cudaMemcpyHostToDevice));
    }

    for (int t = 0; t < timesteps;) {
        int batch = std::min(batchLimit, timesteps - t);
        if (rows) {
            dim3 grid((rows + 127) / 128, batch);
            if (distributed) {
                propagate<true><<<grid, 128>>>(state.weights.data(), state.delays.data(),
                    ring.data(), state.history.data(), state.first, rows, n, t,
                    timesteps / 2, state.sourceIndex, state.reflectivity);
            } else {
                propagate<false><<<grid, 128>>>(state.weights.data(), state.delays.data(),
                    state.history.data(), state.history.data(), state.first, rows, n, t,
                    timesteps / 2, state.sourceIndex, state.reflectivity);
            }
            CUDA_CHECK(cudaGetLastError());
        }
        if (distributed) {
            for (int r = 0; r < state.ranks; ++r) {
                counts[r] = batch * state.counts[r];
                offsets[r] = batch * state.offsets[r];
            }
            if (rows) {
                CUDA_CHECK(cudaMemcpy(exchange + offsets[state.rank],
                    state.history.data() + size_t(t) * rows, size_t(batch) * rows * sizeof(float),
                    cudaMemcpyDeviceToHost));
            }
            MPI_Allgatherv(MPI_IN_PLACE, 0, MPI_FLOAT, exchange, counts.data(), offsets.data(),
                           MPI_FLOAT, MPI_COMM_WORLD);
            CUDA_CHECK(cudaMemcpy(packed.data(), exchange, size_t(batch) * n * sizeof(float),
                                  cudaMemcpyHostToDevice));
            dim3 grid((n + 127) / 128, batch);
            unpackSnapshots<<<grid, 128>>>(packed.data(), deviceOwners.data(), ring.data(),
                state.sourceSignal.data(), state.sourceIndex, n, t, batch);
            CUDA_CHECK(cudaGetLastError());
        }
        t += batch;
    }
    CUDA_CHECK(cudaDeviceSynchronize());
    if (exchange) CUDA_CHECK(cudaFreeHost(exchange));
}

__global__ void correlate(const float* history, const float* signal, int signalStride,
                          int rows, int timesteps, float* distances, int* receivedEnergy) {
    constexpr int lagGroups = 8;
    __shared__ float bestValues[lagGroups][32];
    __shared__ int bestLags[lagGroups][32];
    int x = threadIdx.x, y = threadIdx.y;
    int row = blockIdx.x * 32 + x;
    float best = ZERO;
    int bestLag = 0;
    bool energized = false;
    if (row < rows) {
        for (int lag = y; lag < timesteps; lag += lagGroups) {
            float sum = ZERO;
            for (int k = 0; k < timesteps - lag; ++k) {
                float value = history[size_t(lag + k) * rows + row];
                sum += signal[size_t(k) * signalStride] * value;
                if (receivedEnergy && lag == 0 && value > EPSILON) energized = true;
            }
            if (sum > best) {
                best = sum;
                bestLag = lag;
            }
        }
    }
    bestValues[y][x] = best;
    bestLags[y][x] = bestLag;
    __syncthreads();
    if (y == 0) {
        if (receivedEnergy) {
            int count = __popc(__ballot_sync(0xffffffff, energized));
            if (x == 0 && count) atomicAdd(receivedEnergy, count);
        }
        if (row < rows) {
            for (int g = 1; g < lagGroups; ++g) {
                float other = bestValues[g][x];
                int lag = bestLags[g][x];
                if (other > best || (other == best && lag < bestLag)) {
                    best = other;
                    bestLag = lag;
                }
            }
            distances[row] = WAVE_SPEED * float(bestLag);
        }
    }
}

class Octree {
public:
    Vec3 minBound, maxBound;
    Vec3 halfExtent, center;
    std::unique_ptr<Octree> children[8];
    std::vector<size_t> triangleIndices;  // Indices into the global triangle list
    const std::vector<Triangle>* allTriangles;  // Pointer to all triangles

    Octree() : allTriangles(nullptr) {}

    void build(const std::vector<Triangle>& triangles) {
        allTriangles = &triangles;
        if (triangles.empty()) return;

        // Compute bounding box
        minBound = triangles[0].a;
        maxBound = triangles[0].a;
        for (const auto& tri : triangles) {
            for (const auto* v : {&tri.a, &tri.b, &tri.c}) {
                minBound.x = std::min(minBound.x, v->x);
                minBound.y = std::min(minBound.y, v->y);
                minBound.z = std::min(minBound.z, v->z);
                maxBound.x = std::max(maxBound.x, v->x);
                maxBound.y = std::max(maxBound.y, v->y);
                maxBound.z = std::max(maxBound.z, v->z);
            }
        }

        // Collect all indices
        std::vector<size_t> allIndices(triangles.size());
        for (size_t i = 0; i < triangles.size(); ++i) allIndices[i] = i;

        buildNode(allIndices, minBound, maxBound);
    }

private:
    void buildNode(const std::vector<size_t>& indices, const Vec3& nodeMin, const Vec3& nodeMax) {
        minBound = nodeMin;
        maxBound = nodeMax;
        halfExtent = (maxBound - minBound) * 0.5f;
        center = (minBound + maxBound) * 0.5f;

        // If few enough triangles or too small, make this a leaf
        if (indices.size() <= MAX_OCTREE_TRIS ||
            (maxBound - minBound).norm() < MAX_OCTREE_LEAF_SIZE) {
            triangleIndices = indices;
            return;
        }

        // Subdivide into 8 children
        Vec3 childHalfSize = halfExtent * 0.5f;
        std::vector<size_t> childIndices[8];

        for (size_t idx : indices) {
            const Triangle& tri = (*allTriangles)[idx];

            // Check which children this triangle overlaps
            for (int i = 0; i < 8; ++i) {
                Vec3 childCenter = center;
                childCenter.x += (i & 1) ? childHalfSize.x : -childHalfSize.x;
                childCenter.y += (i & 2) ? childHalfSize.y : -childHalfSize.y;
                childCenter.z += (i & 4) ? childHalfSize.z : -childHalfSize.z;

                if (triangleBoxOverlap(childCenter, childHalfSize, tri)) {
                    childIndices[i].push_back(idx);
                }
            }
        }

        // Check if we can't split further (all triangles in one child)
        bool canSplit = false;
        for (int i = 0; i < 8; ++i) {
            if (!childIndices[i].empty() && childIndices[i].size() < indices.size()) {
                canSplit = true;
                break;
            }
        }

        if (!canSplit) {
            triangleIndices = indices;
            return;
        }

        // Build children
        for (int i = 0; i < 8; ++i) {
            if (!childIndices[i].empty()) {
                Vec3 childMin = center;
                Vec3 childMax = center;
                childMin.x = (i & 1) ? center.x : minBound.x;
                childMax.x = (i & 1) ? maxBound.x : center.x;
                childMin.y = (i & 2) ? center.y : minBound.y;
                childMax.y = (i & 2) ? maxBound.y : center.y;
                childMin.z = (i & 4) ? center.z : minBound.z;
                childMax.z = (i & 4) ? maxBound.z : center.z;

                children[i] = std::make_unique<Octree>();
                children[i]->allTriangles = allTriangles;
                children[i]->buildNode(childIndices[i], childMin, childMax);
            }
        }
    }
