// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// Print results for floating-point arrays
template <typename T>
void print_results(const std::vector<T>& data, const char* name = "Result") {
    if (data.empty()) {
        printf("=== RESULTS ===\n");
        printf("Name: %s\n", name);
        printf("Elements: 0\n");
        printf("=== END RESULTS ===\n");
        return;
    }

    const size_t N = data.size();
    
    // Compute statistics
    T sum = kahan_sum(data);
    T min_val = data[0];
    T max_val = data[0];
    for (const auto& val : data) {
        min_val = std::min(min_val, val);
        max_val = std::max(max_val, val);
    }
    
    // Compute hash
    hash h;
    for (const auto& val : data) {
        h.add(val);
    }
    
    // Sample indices: 0, N/4, N/2, 3N/4, N-1
    size_t indices[5] = {0, N/4, N/2, 3*N/4, N-1};
    
    printf("=== RESULTS ===\n");
    printf("Name: %s\n", name);
    printf("Elements: %zu\n", N);
    printf("Sum: %.17e\n", static_cast<double>(sum));
    printf("Min: %.17e\n", static_cast<double>(min_val));
    printf("Max: %.17e\n", static_cast<double>(max_val));
    for (int i = 0; i < 5; ++i) {
        printf("Sample[%zu]: %.17e\n", indices[i], static_cast<double>(data[indices[i]]));
    }
    printf("Hash: %016" PRIx64 "\n", h.get());
    printf("=== END RESULTS ===\n");
}

    87	        for (int step = 0; step < steps; ++step) {
    88	            #pragma omp for schedule(static)
    89	            for (size_t block = 0; block < blocks; ++block) {
    90	                const size_t begin = block * blockSize;
    91	                const size_t count = std::min(blockSize, n - begin);
    92	                const double* x = current->x.data();
    93	                const double* y = current->y.data();
    94	                const double* z = current->z.data();
    95	                double fx[blockSize] = {}, fy[blockSize] = {}, fz[blockSize] = {};
    96	
    97	                for (size_t j = 0; j < n; ++j) {
    98	                    const double sourceX = x[j], sourceY = y[j], sourceZ = z[j];
    99	                    // Vectorize across targets to preserve each body's sum order.
   100	                    #pragma omp simd
   101	                    for (size_t lane = 0; lane < blockSize; ++lane) {
   102	                        const double dx = sourceX - x[begin + lane];
   103	                        const double dy = sourceY - y[begin + lane];
   104	                        const double dz = sourceZ - z[begin + lane];
   105	                        const double distSqr = dx * dx + dy * dy + dz * dz + SOFTENING;
   106	                        const double invDist = 1.0 / std::sqrt(distSqr);
   107	                        const double invDist3 = invDist * invDist * invDist;
   108	
   109	                        fx[lane] += dx * invDist3;
   110	                        fy[lane] += dy * invDist3;
   111	                        fz[lane] += dz * invDist3;
   112	                    }
   113	                }
   114	
   115	                for (size_t lane = 0; lane < count; ++lane) {
   116	                    const size_t i = begin + lane;
   117	                    auto& vel = bodies[i].vel;
   118	                    vel.x = updateVelocity(vel.x, fx[lane]);
   119	                    vel.y = updateVelocity(vel.y, fy[lane]);
   120	                    vel.z = updateVelocity(vel.z, fz[lane]);
   121	                    next->x[i] = x[i] + vel.x * DT;
   122	                    next->y[i] = y[i] + vel.y * DT;
   123	                    next->z[i] = z[i] + vel.z * DT;
   124	                }
   125	            }
   126	            current = next;
   127	            next = next == &first ? &second : &first;
   128	        }
   129	
   130	        #pragma omp for schedule(static) nowait
   131	        for (size_t i = 0; i < n; ++i) {
   132	            bodies[i].pos = {current->x[i], current->y[i], current->z[i]};
   133	        }
   134	    }
   135	}

   137	double computeTotalEnergy(const std::vector<Body>& bodies) {
   138	    double energy = 0.0;
   139	    const size_t n = bodies.size();
   140	    
   141	    // Kinetic energy (assuming unit mass)
   142	    for (const auto& body : bodies) {
   143	        energy += 0.5 * (body.vel.x * body.vel.x + 
   144	                        body.vel.y * body.vel.y + 
   145	                        body.vel.z * body.vel.z);
   146	    }
   147	    if (n < 2) {
   148	        return energy;
   149	    }
   150	
   151	    // Number the pairs in the same order as the original triangular loop.
   152	    const auto rowStart = [n](std::uint64_t i) { return i * (2 * std::uint64_t{n} - i - 1) / 2; };
   153	    constexpr size_t pairsPerBlock = 4096;
   154	    const std::uint64_t pairs = rowStart(n - 1);
   155	    const std::uint64_t blocks = (pairs + pairsPerBlock - 1) / pairsPerBlock;
   156	    // The ordered sum limits how much parallel pair evaluation can help.
   157	    const int threads = std::min({std::uint64_t{8}, static_cast<std::uint64_t>(omp_get_max_threads()), blocks});
   158	    Positions positions(n);
   159	    const double* x = positions.x.data();
   160	    const double* y = positions.y.data();
   161	    const double* z = positions.z.data();
   162	
   163	    #pragma omp parallel num_threads(threads)
   164	    {
   165	        #pragma omp for schedule(static)
   166	        for (size_t i = 0; i < n; ++i) {
   167	            positions.set(i, bodies[i].pos);
   168	        }
   169	
   170	        double potential[pairsPerBlock];
   171	        #pragma omp for ordered schedule(static, 1)
   172	        for (std::uint64_t block = 0; block < blocks; ++block) {
   173	            const std::uint64_t begin = block * pairsPerBlock;
   174	            const size_t count = std::min<std::uint64_t>(pairsPerBlock, pairs - begin);
   175	
   176	            size_t i = 0, upper = n - 1;
   177	            while (i + 1 < upper) {
   178	                const size_t mid = i + (upper - i) / 2;
   179	                if (rowStart(mid) <= begin) {
   180	                    i = mid;
   181	                } else {
   182	                    upper = mid;
   183	                }
   184	            }
   185	
   186	            size_t j = i + 1 + (begin - rowStart(i));
   187	            for (size_t offset = 0; offset < count; ++i, j = i + 1) {
   188	                const double targetX = x[i], targetY = y[i], targetZ = z[i];
   189	                const size_t run = std::min(n - j, count - offset);
   190	                #pragma omp simd
   191	                for (size_t lane = 0; lane < run; ++lane) {
   192	                    const double dx = x[j + lane] - targetX;
   193	                    const double dy = y[j + lane] - targetY;
   194	                    const double dz = z[j + lane] - targetZ;
   195	                    const double dist = std::sqrt(dx * dx + dy * dy + dz * dz + SOFTENING);
   196	                    potential[offset + lane] = 1.0 / dist;
   197	                }
   198	                offset += run;
   199	            }
   200	
   201	            #pragma omp ordered
   202	            {
   203	                for (size_t k = 0; k < count; ++k) {
   204	                    energy -= potential[k];
   205	                }
   206	            }
   207	        }
   208	    }
   209	    
   210	    return energy;
   211	}
