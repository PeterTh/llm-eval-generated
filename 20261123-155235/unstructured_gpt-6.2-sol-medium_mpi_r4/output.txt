// Types to represent unstructured mesh elements
using idx_t = uint64_t;
using val_t = double;

// Maximum number of connections per element (for a 2D grid: 4 neighbors)
constexpr int MAX_CONNECTIONS = 8;

// Material properties for energy transfer
struct Material {
    val_t transfer_coeff;  // Energy transfer coefficient
    val_t external_flow;   // External energy source/sink
};

// Static connectivity information for each element
struct ElementStatic {
    idx_t material_idx;
    idx_t num_connections;
    idx_t connected_idx[MAX_CONNECTIONS];     // Indices of connected elements
    val_t connected_flux[MAX_CONNECTIONS];    // Flux coefficients for each connection
};

// Dynamic state for each element
struct ElementDynamic {
    val_t current_energy;
    val_t total_flux;
};

// World state
struct World {
    std::vector<Material> materials;
    std::vector<ElementStatic> elements_static;
    std::vector<ElementDynamic> elements_dynamic;
    std::vector<ElementDynamic> elements_dynamic_swap;
};

// Material type IDs
constexpr idx_t DEFAULT_MAT_ID = 0;
constexpr idx_t INFLOW_MAT_ID = 1;
constexpr idx_t OUTFLOW_MAT_ID = 2;

// Build a 2D square grid as an unstructured mesh
// This represents computation on arbitrarily-shaped geometries
void buildSquare2D(World& world, const int n_elems_root) {
    const int n_elems = n_elems_root * n_elems_root;
    
    // Initialize materials
    world.materials.emplace_back(Material{0.8, 0.0});    // Default material
    world.materials.emplace_back(Material{0.8, 0.5});    // Inflow material
    world.materials.emplace_back(Material{0.8, -0.5});   // Outflow material
    
    // Allocate elements
    world.elements_static.resize(n_elems);
    world.elements_dynamic.resize(n_elems);
    world.elements_dynamic_swap.resize(n_elems);
    
    // Initialize all elements with default material and zero energy
    for (int i = 0; i < n_elems; ++i) {
        world.elements_static[i].material_idx = DEFAULT_MAT_ID;
        world.elements_static[i].num_connections = 0;
        world.elements_dynamic[i].current_energy = 0.0;
        world.elements_dynamic[i].total_flux = 0.0;
    }
    
    // Build connectivity: each element connects to its neighbors in 2D grid
    for (int x = 0; x < n_elems_root; ++x) {
        for (int y = 0; y < n_elems_root; ++y) {
            const int idx = x * n_elems_root + y;
            ElementStatic& elem = world.elements_static[idx];
            
            // Connect to neighbors (up, down, left, right)
            const int offsets[4][2] = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}};
            
            for (int n = 0; n < 4; ++n) {
                const int nx = x + offsets[n][0];
                const int ny = y + offsets[n][1];
                
                // Check if neighbor is within bounds
                if (nx >= 0 && nx < n_elems_root && ny >= 0 && ny < n_elems_root) {
                    const int neighbor_idx = nx * n_elems_root + ny;
                    elem.connected_idx[elem.num_connections] = neighbor_idx;
                    elem.connected_flux[elem.num_connections] = 1.0;
                    elem.num_connections++;
                }
            }
        }
    }
    
    // Set corner elements as inflow/outflow to create interesting dynamics
    const int last = n_elems_root - 1;
    world.elements_static[0 * n_elems_root + 0].material_idx = INFLOW_MAT_ID;
    world.elements_static[0 * n_elems_root + last].material_idx = OUTFLOW_MAT_ID;
    world.elements_static[last * n_elems_root + 0].material_idx = OUTFLOW_MAT_ID;
    world.elements_static[last * n_elems_root + last].material_idx = INFLOW_MAT_ID;
}

// Compute energy flux between two elements
inline val_t computeFlux(const Material& mat, const ElementDynamic& this_elem,
                        val_t connection_flux, const ElementDynamic& other_elem) {
    return (other_elem.current_energy - this_elem.current_energy) * 
           mat.transfer_coeff * connection_flux * 0.25;
}

// Run simulation for n_iters iterations
void runSimulation(World& world, const int n_iters) {
    const size_t n_elems = world.elements_static.size();
    
    for (int iter = 0; iter < n_iters; ++iter) {
        // Update all elements
        for (size_t i = 0; i < n_elems; ++i) {
            const ElementStatic& elem_static = world.elements_static[i];
            const ElementDynamic& elem_dyn = world.elements_dynamic[i];
            const Material& mat = world.materials[elem_static.material_idx];
            
            // Start with external flow
            val_t total_flux = mat.external_flow;
            
            // Add flux from all connected elements
            for (idx_t j = 0; j < elem_static.num_connections; ++j) {
                const idx_t neighbor_idx = elem_static.connected_idx[j];
                const ElementDynamic& neighbor_dyn = world.elements_dynamic[neighbor_idx];
                total_flux += computeFlux(mat, elem_dyn, elem_static.connected_flux[j], neighbor_dyn);
            }
            
            // Update element state
            ElementDynamic& elem_write = world.elements_dynamic_swap[i];
            elem_write.current_energy = elem_dyn.current_energy + total_flux;
            elem_write.total_flux = elem_dyn.total_flux + std::abs(total_flux);
        }
        
        // Swap buffers
        std::swap(world.elements_dynamic, world.elements_dynamic_swap);
    }
}

// Validate simulation results
bool validateResults(const World& world) {
    val_t energy_sum = 0.0;
    val_t flux_sum = 0.0;
    val_t energy_max = std::numeric_limits<val_t>::lowest();
    val_t energy_min = std::numeric_limits<val_t>::max();
    
    for (const auto& elem : world.elements_dynamic) {
        energy_sum += elem.current_energy;
        flux_sum += elem.total_flux;
        energy_max = std::max(elem.current_energy, energy_max);
        energy_min = std::min(elem.current_energy, energy_min);
    }
    
    printf("Validation results:\n");
    printf("  Energy sum: %.12f\n", energy_sum);
    printf("  Flux sum: %.2f\n", flux_sum);
    printf("  Energy range: [%.6f, %.6f]\n", energy_min, energy_max);
    
    // Check for numerical issues
    constexpr val_t energy_epsilon = 1e-8;
    
    if (!std::isfinite(energy_sum)) {
        printf("  ERROR: Energy sum is not finite\n");
        return false;
    }
    
    if (std::abs(energy_sum) > energy_epsilon) {
        printf("  WARNING: Energy sum diverged from 0 (expected conservation)\n");
        // Don't fail validation as this can happen with external flows
    }
    
    if (!std::isfinite(flux_sum)) {
        printf("  ERROR: Flux sum is not finite\n");
        return false;
    }
    
    if (!std::isfinite(energy_max) || !std::isfinite(energy_min)) {
        printf("  ERROR: Energy extrema are not finite\n");
        return false;
    }
    
    printf("  Validation: PASSED\n");
    return true;
}

    uint64_t hash = 0;
    for (size_t i = 0; i < elements.size(); ++i) {
        // Simple hash combining energy and flux values
        const uint64_t* e_ptr = reinterpret_cast<const uint64_t*>(&elements[i].current_energy);
        const uint64_t* f_ptr = reinterpret_cast<const uint64_t*>(&elements[i].total_flux);
        hash ^= (*e_ptr + i) * 0x9e3779b97f4a7c15ULL;
        hash ^= (*f_ptr + i) * 0xbf58476d1ce4e5b9ULL;
    }
    return hash;
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

template <typename T>

    if (data.empty()) {

        printf("Elements: 0\n");
        printf("=== END RESULTS ===\n");

    T sum = kahan_sum(data);
    T min_val = data[0];
    T max_val = data[0];

        min_val = std::min(min_val, val);

    for (const auto& val : data) {

void buildSquare2D(World& world, const Partition& partition, int rank) {
    if (rank >= partition.active_ranks) {
        return;
    }

    world.tile = partition.tile(rank);
    const auto [first_row, first_col, rows, cols] = world.tile;
    const int grid_size = partition.grid_size;
    const int end_row = first_row + rows;
    const int end_col = first_col + cols;
    const idx_t owned = world.tile.size();

    idx_t buffer_size = owned;
    const idx_t down = buffer_size;
    buffer_size += end_row < grid_size ? cols : 0;
    const idx_t up = buffer_size;
    buffer_size += first_row > 0 ? cols : 0;
    const idx_t right = buffer_size;
    buffer_size += end_col < grid_size ? rows : 0;
    const idx_t left = buffer_size;
    buffer_size += first_col > 0 ? rows : 0;

    world.elements.resize(owned);
    world.total_flux.resize(owned);
    for (auto& buffer : world.energy) {
        buffer.resize(buffer_size);
    }

    // Retain the serial connection order for floating-point accumulation.
    for (int row = 0; row < rows; ++row) {
        for (int col = 0; col < cols; ++col) {
            const idx_t index = static_cast<idx_t>(row) * cols + col;
            ElementStatic& elem = world.elements[index];
            const auto connect = [&](idx_t neighbor) {
                const int connection = elem.num_connections++;
                elem.connected_idx[connection] = neighbor;
                elem.connected_flux[connection] = 1.0;
            };

            const int x = first_row + row;
            const int y = first_col + col;
            if (x + 1 < grid_size) {
                connect(row + 1 < rows ? index + cols : down + col);
            }
            if (x > 0) {
                connect(row > 0 ? index - cols : up + col);
            }
            if (y + 1 < grid_size) {
                connect(col + 1 < cols ? index + 1 : right + row);
            }
            if (y > 0) {
                connect(col > 0 ? index - 1 : left + row);
            }

            const int last = grid_size - 1;
            if ((x == 0 && y == 0) || (x == last && y == last)) {
                elem.material_idx = INFLOW_MAT_ID;
            } else if ((x == 0 && y == last) || (x == last && y == 0)) {
                elem.material_idx = OUTFLOW_MAT_ID;

        MPI_Datatype send_type = MPI_DOUBLE;
        int send_count = count;
        if (stride > 1) {
            MPI_Type_vector(count, 1, stride, MPI_DOUBLE, &send_type);
            MPI_Type_commit(&send_type);
            send_count = 1;

                          MPI_COMM_WORLD, &send_request);
            requests.push_back(recv_request);
            requests.push_back(send_request);
        }

        if (stride > 1) {

        }
    };

    const int band_index = splitOwner(partition.active_ranks, partition.bands, rank);
    const auto addRowHalos = [&](int neighbor_band, idx_t send_offset, idx_t recv_offset) {
        const Band neighbor = partition.band(neighbor_band);

    }
    if (first_col > 0) {
        addHalo(rank - 1, 0, left, rows, cols);
    }
}

inline void updateElements(const ElementStatic* elements, const val_t* current,
                           val_t* next, val_t* total_flux, idx_t first, idx_t end) {
    for (idx_t i = first; i < end; ++i) {
        const ElementStatic& elem = elements[i];
        const Material& mat = materials[elem.material_idx];
        const val_t energy = current[i];
        val_t flux = mat.external_flow;
        for (int j = 0; j < elem.num_connections; ++j) {
            flux += (current[elem.connected_idx[j]] - energy) *
                    mat.transfer_coeff * elem.connected_flux[j] * 0.25;
        }
        next[i] = energy + flux;
        total_flux[i] += std::abs(flux);
    }
}

void runSimulation(World& world, int grid_size, int n_iters) {
    const auto [first_row, first_col, rows, cols] = world.tile;
    const idx_t owned = world.tile.size();
    if (owned == 0) {
        return;
    }

    const int inner_row = first_row > 0 ? 1 : 0;
    const int inner_col = first_col > 0 ? 1 : 0;
    const int end_inner_row = std::max(inner_row, rows - (first_row + rows < grid_size));
    const int end_inner_col = std::max(inner_col, cols - (first_col + cols < grid_size));

    for (int iter = 0; iter < n_iters; ++iter) {
        const int buffer = world.current_buffer;
        auto& requests = world.halo_requests[buffer];
        const auto update = [&](idx_t begin, idx_t end) {
            updateElements(world.elements.data(), world.energy[buffer].data(),
                           world.energy[1 - buffer].data(), world.total_flux.data(), begin, end);
        };

        if (iter == 0 || requests.empty()) {
            // The initial energies, including all ghost values, are zero.
            update(0, owned);
        } else {
            const int count = static_cast<int>(requests.size());
            MPI_Startall(count, requests.data());
            for (int row = inner_row; row < end_inner_row; ++row) {
                const idx_t start = static_cast<idx_t>(row) * cols;
                update(start + inner_col, start + end_inner_col);
            }

            MPI_Waitall(count, requests.data(), MPI_STATUSES_IGNORE);
            update(0, static_cast<idx_t>(inner_row) * cols);
            for (int row = inner_row; row < end_inner_row; ++row) {
                const idx_t start = static_cast<idx_t>(row) * cols;
                update(start, start + inner_col);
                update(start + end_inner_col, start + cols);
            }
            update(static_cast<idx_t>(end_inner_row) * cols, owned);
        }
        world.current_buffer = 1 - buffer;
    }
}

   145	            const auto connect = [&](idx_t neighbor) {
   146	                elem.connected_idx[elem.num_connections++] = neighbor;
   147	            };

   149	            const int x = first_row + row;
   150	            const int y = first_col + col;
   151	            if (x + 1 < grid_size) {

   153	            }

   155	                connect(row > 0 ? index - cols : up + col);
   156	            }
   157	            if (y + 1 < grid_size) {

   159	            }

   161	                connect(col > 0 ? index - 1 : left + row);

   165	            if ((x == 0 && y == 0) || (x == last && y == last)) {
   166	                elem.material_idx = INFLOW_MAT_ID;
   167	            } else if ((x == 0 && y == last) || (x == last && y == 0)) {
   168	                elem.material_idx = OUTFLOW_MAT_ID;
   169	            } else {
   170	                elem.material_idx = DEFAULT_MAT_ID;
   171	            }

   173	    }

   175	    const auto addHalo = [&](int peer, idx_t send_offset, idx_t recv_offset,
   176	                             int count, int stride) {
   177	        MPI_Datatype send_type = MPI_DOUBLE;
   178	        int send_count = count;
   179	        if (stride > 1) {
   180	            MPI_Type_vector(count, 1, stride, MPI_DOUBLE, &send_type);
   181	            MPI_Type_commit(&send_type);
   182	            send_count = 1;
   183	        }
   184	
   185	        for (int buffer = 0; buffer < 2; ++buffer) {
   186	            auto* energy = world.energy[buffer].data();
   187	            auto& requests = world.halo_requests[buffer];
   188	            MPI_Request recv_request, send_request;
   189	            MPI_Recv_init(energy + recv_offset, count, MPI_DOUBLE, peer, 0,
   190	                          MPI_COMM_WORLD, &recv_request);
   191	            MPI_Send_init(energy + send_offset, send_count, send_type, peer, 0,
   192	                          MPI_COMM_WORLD, &send_request);
   193	            requests.push_back(recv_request);
   194	            requests.push_back(send_request);
   195	        }
   196	
   197	        if (stride > 1) {
   198	            MPI_Type_free(&send_type);
   199	        }
   200	    };

   202	    // Adjacent bands can have different tile widths. Exchange each shared segment.
   203	    const int band_index = splitOwner(partition.active_ranks, partition.bands, rank);
   204	    const auto addRowHalos = [&](int neighbor_band, idx_t send_offset, idx_t recv_offset) {
   205	        const Band neighbor = partition.band(neighbor_band);
   206	        int col = first_col;
   207	        int peer_col = splitOwner(grid_size, neighbor.rank_count, col);
   208	        while (col < end_col) {
   209	            const int next = std::min(end_col,
   210	                                      splitStart(grid_size, neighbor.rank_count, peer_col + 1));
   211	            const int offset = col - first_col;
   212	            addHalo(neighbor.first_rank + peer_col, send_offset + offset,
   213	                    recv_offset + offset, next - col, 1);
   214	            col = next;
   215	            ++peer_col;
   216	        }
   217	    };
   218	
   219	    if (end_row < grid_size) {
   220	        addRowHalos(band_index + 1, owned - cols, down);
   221	    }
   222	    if (first_row > 0) {
   223	        addRowHalos(band_index - 1, 0, up);
   224	    }
   225	    if (end_col < grid_size) {
   226	        addHalo(rank + 1, cols - 1, right, rows, cols);
   227	    }
   228	    if (first_col > 0) {
   229	        addHalo(rank - 1, 0, left, rows, cols);
   230	    }
   231	}

   233	inline void updateElements(const ElementStatic* elements, const val_t* current,
   234	                           val_t* next, val_t* total_flux, idx_t first, idx_t end) {
   235	    for (idx_t i = first; i < end; ++i) {
   236	        const ElementStatic& elem = elements[i];
   237	        const Material& mat = materials[elem.material_idx];
   238	        const val_t energy = current[i];
   239	        val_t flux = mat.external_flow;
   240	        for (int j = 0; j < MAX_CONNECTIONS; ++j) {
   241	            if (j == elem.num_connections) {
   242	                break;
   243	            }
   244	            flux += (current[elem.connected_idx[j]] - energy) *
   245	                    mat.transfer_coeff * CONNECTION_FLUX * 0.25;
   246	        }
   247	        next[i] = energy + flux;
   248	        total_flux[i] += std::abs(flux);
   249	    }
   250	}

   252	void runSimulation(World& world, int grid_size, int n_iters) {
   253	    const auto [first_row, first_col, rows, cols] = world.tile;
   254	    const idx_t owned = world.tile.size();
   255	    if (owned == 0) {
   256	        return;
   257	    }
   258	
   259	    const int inner_row = first_row > 0 ? 1 : 0;
   260	    const int inner_col = first_col > 0 ? 1 : 0;
   261	    const int end_inner_row = std::max(inner_row, rows - (first_row + rows < grid_size));
   262	    const int end_inner_col = std::max(inner_col, cols - (first_col + cols < grid_size));
   263	
   264	    for (int iter = 0; iter < n_iters; ++iter) {
   265	        const int buffer = world.current_buffer;
   266	        auto& requests = world.halo_requests[buffer];
   267	        const auto update = [&](idx_t begin, idx_t end) {
   268	            updateElements(world.elements.data(), world.energy[buffer].data(),
   269	                           world.energy[1 - buffer].data(), world.total_flux.data(), begin, end);
   270	        };
   271	
   272	        if (iter == 0 || requests.empty()) {
   273	            // The initial energies, including all ghost values, are zero.
   274	            update(0, owned);
   275	        } else {
   276	            const int count = static_cast<int>(requests.size());
   277	            MPI_Startall(count, requests.data());
   278	            for (int row = inner_row; row < end_inner_row; ++row) {
   279	                const idx_t start = static_cast<idx_t>(row) * cols;
   280	                update(start + inner_col, start + end_inner_col);
   281	            }
   282	
   283	            MPI_Waitall(count, requests.data(), MPI_STATUSES_IGNORE);
   284	            update(0, static_cast<idx_t>(inner_row) * cols);
   285	            for (int row = inner_row; row < end_inner_row; ++row) {
   286	                const idx_t start = static_cast<idx_t>(row) * cols;
   287	                update(start, start + inner_col);
   288	                update(start + end_inner_col, start + cols);
   289	            }
   290	            update(static_cast<idx_t>(end_inner_row) * cols, owned);
   291	        }
   292	        world.current_buffer = 1 - buffer;
   293	    }
   294	}

   296	uint64_t computeHash(const World& world, int grid_size) {
   297	    const auto [first_row, first_col, rows, cols] = world.tile;
   298	    const auto& energy = world.energy[world.current_buffer];
   299	    uint64_t hash = 0;
   300	    for (int row = 0; row < rows; ++row) {
   301	        const idx_t local_start = static_cast<idx_t>(row) * cols;
   302	        const uint64_t global_start = static_cast<uint64_t>(first_row + row) * grid_size + first_col;
   303	        for (int col = 0; col < cols; ++col) {
   304	            const idx_t local = local_start + col;
   305	            const uint64_t global = global_start + col;
   306	            hash ^= (std::bit_cast<uint64_t>(energy[local]) + global) * 0x9e3779b97f4a7c15ULL;
   307	            hash ^= (std::bit_cast<uint64_t>(world.total_flux[local]) + global) * 0xbf58476d1ce4e5b9ULL;
   308	        }
   309	    }
   310	    return hash;
   311	}
