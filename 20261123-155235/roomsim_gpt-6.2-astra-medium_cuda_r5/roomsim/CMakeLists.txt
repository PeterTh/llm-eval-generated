cmake_minimum_required(VERSION 3.24)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND "$ENV{CUDAARCHS}" STREQUAL "")
    set(roomsim_detect_gpu ON)
endif()
project(roomsim_benchmark LANGUAGES CXX CUDA)

if(roomsim_detect_gpu AND CMAKE_CUDA_ARCHITECTURES_NATIVE)
    string(REPLACE "-real" "" roomsim_archs "${CMAKE_CUDA_ARCHITECTURES_NATIVE}")
    set(CMAKE_CUDA_ARCHITECTURES "${roomsim_archs}" CACHE STRING "CUDA architectures to build" FORCE)
endif()

set_source_files_properties(roomsim.cpp PROPERTIES LANGUAGE CUDA)
add_executable(roomsim roomsim.cpp)
set_target_properties(roomsim PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_compile_options(roomsim PRIVATE -O3 -Xcompiler=-march=native)
