   105	// Separating axis test for triangle-box overlap
   106	bool triangleBoxOverlap(const Vec3& boxCenter, const Vec3& boxHalfSize, const Triangle& tri) {
   107	    // Translate triangle to box center
   108	    Vec3 v0 = tri.a - boxCenter;
   109	    Vec3 v1 = tri.b - boxCenter;
   110	    Vec3 v2 = tri.c - boxCenter;
   111	
   112	    // Triangle edges
   113	    Vec3 e0 = v1 - v0;
   114	    Vec3 e1 = v2 - v1;
   115	    Vec3 e2 = v0 - v2;
   116	
   117	    // Test box normals (AABB axes)
   118	    auto minMax3 = [](val_t a, val_t b, val_t c) {
   119	        return std::make_pair(std::min({a, b, c}), std::max({a, b, c}));
   120	    };
   121	
   122	    auto [minX, maxX] = minMax3(v0.x, v1.x, v2.x);
   123	    if (minX > boxHalfSize.x || maxX < -boxHalfSize.x) return false;
   124	
   125	    auto [minY, maxY] = minMax3(v0.y, v1.y, v2.y);
   126	    if (minY > boxHalfSize.y || maxY < -boxHalfSize.y) return false;
   127	
   128	    auto [minZ, maxZ] = minMax3(v0.z, v1.z, v2.z);
   129	    if (minZ > boxHalfSize.z || maxZ < -boxHalfSize.z) return false;
   130	
   131	    // Test triangle normal
   132	    Vec3 triNormal = e0.cross(e1);
   133	    val_t d = triNormal.dot(v0);
   134	    val_t r = boxHalfSize.x * std::abs(triNormal.x) +
   135	              boxHalfSize.y * std::abs(triNormal.y) +
   136	              boxHalfSize.z * std::abs(triNormal.z);
   137	    if (std::abs(d) > r) return false;
   138	
   139	    // Test 9 edge cross products
   140	    auto testAxis = [&](const Vec3& axis) {
   141	        val_t p0 = axis.dot(v0);
   142	        val_t p1 = axis.dot(v1);
   143	        val_t p2 = axis.dot(v2);
   144	        val_t r = boxHalfSize.x * std::abs(axis.x) +
   145	                  boxHalfSize.y * std::abs(axis.y) +
   146	                  boxHalfSize.z * std::abs(axis.z);
   147	        auto [minP, maxP] = minMax3(p0, p1, p2);
   148	        return !(minP > r || maxP < -r);
   149	    };
   150	
   151	    Vec3 axes[3] = {{1,0,0}, {0,1,0}, {0,0,1}};
   152	    Vec3 edges[3] = {e0, e1, e2};
   153	    for (const auto& axis : axes) {
   154	        for (const auto& edge : edges) {
   155	            Vec3 crossAxis = axis.cross(edge);
   156	            if (crossAxis.squaredNorm() > EPSILON) {
   157	                if (!testAxis(crossAxis)) return false;
   158	            }
   159	        }
   160	    }
   161	
   162	    return true;
   163	}

   209	        minBound = nodeMin;
   210	        maxBound = nodeMax;
   211	        halfExtent = (maxBound - minBound) * 0.5f;
   212	        center = (minBound + maxBound) * 0.5f;
   213	
   214	        // If few enough triangles or too small, make this a leaf
   215	        if (indices.size() <= MAX_OCTREE_TRIS ||

   217	            triangleIndices = indices;
   218	            return;
   219	        }

   231	                childCenter.x += (i & 1) ? childHalfSize.x : -childHalfSize.x;
   232	                childCenter.y += (i & 2) ? childHalfSize.y : -childHalfSize.y;

   237	                }
   238	            }
   239	        }
   240	
   241	        // Check if we can't split further (all triangles in one child)
   242	        bool canSplit = false;
   243	        for (int i = 0; i < 8; ++i) {
   244	            if (!childIndices[i].empty() && childIndices[i].size() < indices.size()) {
   245	                canSplit = true;
   246	                break;
   247	            }
   248	        }
   249	
   250	        if (!canSplit) {
   251	            triangleIndices = indices;
   252	            return;
   253	        }
   254	
   255	        // Build children
   256	        for (int i = 0; i < 8; ++i) {
   257	            if (!childIndices[i].empty()) {
   258	                Vec3 childMin = center;
   259	                Vec3 childMax = center;
   260	                childMin.x = (i & 1) ? center.x : minBound.x;
   261	                childMax.x = (i & 1) ? maxBound.x : center.x;
   262	                childMin.y = (i & 2) ? center.y : minBound.y;
   263	                childMax.y = (i & 2) ? maxBound.y : center.y;
   264	                childMin.z = (i & 4) ? center.z : minBound.z;
   265	                childMax.z = (i & 4) ? maxBound.z : center.z;
   266	
   267	                children[i] = std::make_unique<Octree>();
   268	                children[i]->allTriangles = allTriangles;
   269	                children[i]->buildNode(childIndices[i], childMin, childMax);
   270	            }
   271	        }
   272	    }

   275	    // Check if a ray intersects this node's bounding box
   276	    bool rayIntersectsBox(const Vec3& p1, const Vec3& p2) const {
   277	        Vec3 d = (p2 - p1) * 0.5f;
   278	        Vec3 c = p1 + d - center;
   279	        Vec3 ad = {std::abs(d.x), std::abs(d.y), std::abs(d.z)};
   280	
   281	        if (std::abs(c.x) > halfExtent.x + ad.x) return false;
   282	        if (std::abs(c.y) > halfExtent.y + ad.y) return false;
   283	        if (std::abs(c.z) > halfExtent.z + ad.z) return false;
   284	
   285	        if (std::abs(d.y * c.z - d.z * c.y) > halfExtent.y * ad.z + halfExtent.z * ad.y + EPSILON) return false;
   286	        if (std::abs(d.z * c.x - d.x * c.z) > halfExtent.z * ad.x + halfExtent.x * ad.z + EPSILON) return false;
   287	        if (std::abs(d.x * c.y - d.y * c.x) > halfExtent.x * ad.y + halfExtent.y * ad.x + EPSILON) return false;
   288	
   289	        return true;
   290	    }

   292	    // Apply a function to all triangles potentially intersecting the ray
   293	    // Returns true if the function returns true for any triangle
   294	    template<typename Func>

   296	        // If leaf node, check triangles directly

   319	class IcosphereMesh {
   320	public:
   321	    std::vector<Triangle> triangles;
   322	
   323	    IcosphereMesh(int subdivisions, val_t radius) {
   324	        // Initial icosahedron vertices
   325	        const val_t t = (1.0f + std::sqrt(5.0f)) / 2.0f;
   326	
   327	        std::vector<Vec3> vertices = {
   328	            Vec3(-1,  t,  0).normalized() * radius,
   329	            Vec3( 1,  t,  0).normalized() * radius,
   330	            Vec3(-1, -t,  0).normalized() * radius,
   331	            Vec3( 1, -t,  0).normalized() * radius,
   332	            Vec3( 0, -1,  t).normalized() * radius,
   333	            Vec3( 0,  1,  t).normalized() * radius,
   334	            Vec3( 0, -1, -t).normalized() * radius,
   335	            Vec3( 0,  1, -t).normalized() * radius,
   336	            Vec3( t,  0, -1).normalized() * radius,
   337	            Vec3( t,  0,  1).normalized() * radius,
   338	            Vec3(-t,  0, -1).normalized() * radius,
   339	            Vec3(-t,  0,  1).normalized() * radius
   340	        };
   341	
   342	        // Initial icosahedron faces (20 triangles)
   343	        std::vector<std::array<idx_t, 3>> faces = {
   344	            {0, 11, 5}, {0, 5, 1}, {0, 1, 7}, {0, 7, 10}, {0, 10, 11},
   345	            {1, 5, 9}, {5, 11, 4}, {11, 10, 2}, {10, 7, 6}, {7, 1, 8},
   346	            {3, 9, 4}, {3, 4, 2}, {3, 2, 6}, {3, 6, 8}, {3, 8, 9},
   347	            {4, 9, 5}, {2, 4, 11}, {6, 2, 10}, {8, 6, 7}, {9, 8, 1}
   348	        };
   349	
   350	        // Subdivide
   351	        for (int i = 0; i < subdivisions; ++i) {
   352	            std::vector<std::array<idx_t, 3>> newFaces;
   353	            std::map<std::pair<idx_t, idx_t>, idx_t> midpointCache;
   354	
   355	            auto getMidpoint = [&](idx_t i1, idx_t i2) -> idx_t {
   356	                auto key = std::make_pair(std::min(i1, i2), std::max(i1, i2));
   357	                auto it = midpointCache.find(key);
   358	                if (it != midpointCache.end()) return it->second;
   359	
   360	                Vec3 mid = (vertices[i1] + vertices[i2]) / 2.0f;
   361	                mid = mid.normalized() * radius;
   362	                idx_t idx = static_cast<idx_t>(vertices.size());
   363	                vertices.push_back(mid);
   364	                midpointCache[key] = idx;
   365	                return idx;
   366	            };
   367	
   368	            for (const auto& face : faces) {
   369	                idx_t a = getMidpoint(face[0], face[1]);
   370	                idx_t b = getMidpoint(face[1], face[2]);
   371	                idx_t c = getMidpoint(face[2], face[0]);
   372	
   373	                newFaces.push_back({face[0], a, c});
   374	                newFaces.push_back({face[1], b, a});
   375	                newFaces.push_back({face[2], c, b});
   376	                newFaces.push_back({a, b, c});
   377	            }
   378	            faces = std::move(newFaces);
   379	        }
   380	
   381	        // Build triangles (normals pointing inward for a "room")
   382	        triangles.reserve(faces.size());
   383	        for (const auto& face : faces) {
   384	            // Reverse winding to make normals point inward
   385	            triangles.emplace_back(vertices[face[2]], vertices[face[1]], vertices[face[0]]);
   386	        }
   387	    }
   388	};

   402	// Generate a random point inside a triangle using barycentric coordinates
   403	Vec3 randomPointInTriangle(const Triangle& t, RandomGenerator& rng) {
   404	    val_t u = rng.rand();
   405	    val_t v = rng.rand();
   406	    if (u + v > 1.0f) {
   407	        u = 1.0f - u;
   408	        v = 1.0f - v;
   409	    }
   410	    Vec3 ab = t.b - t.a;
   411	    Vec3 ac = t.c - t.a;
   412	    return t.a + ab * u + ac * v;

   415	// ============================================================================
   416	// Ray-Triangle Intersection (Möller-Trumbore algorithm)
   417	// ============================================================================
   418	
   419	val_t rayTriangleIntersect(const Vec3& orig, const Vec3& dir,
   420	                            const Vec3& v0, const Vec3& v1, const Vec3& v2) {
   421	    Vec3 e1 = v1 - v0;
   422	    Vec3 e2 = v2 - v0;
   423	    Vec3 pvec = dir.cross(e2);
   424	    val_t det = e1.dot(pvec);
   425	
   426	    if (std::abs(det) < EPSILON) return std::numeric_limits<val_t>::max();
   427	
   428	    val_t invDet = 1.0f / det;
   429	    Vec3 tvec = orig - v0;
   430	    val_t u = tvec.dot(pvec) * invDet;
   431	    if (u < 0.0f || u > 1.0f) return std::numeric_limits<val_t>::max();
   432	
   433	    Vec3 qvec = tvec.cross(e1);
   434	    val_t v = dir.dot(qvec) * invDet;
   435	    if (v < 0.0f || u + v > 1.0f) return std::numeric_limits<val_t>::max();
   436	
   437	    return e2.dot(qvec) * invDet;
   438	}

   469	// Compute the cosine of angle between vector and triangle normal
   470	val_t cosPhi(const Vec3& v, const Vec3& normal) {
   471	    val_t vNorm = v.norm();

   473	    return std::max(ZERO, v.dot(normal) / vNorm);

   476	// Compute form factor Kij between triangle i (receiver) and triangle j (emitter)
   477	val_t computeKij(size_t idxI, size_t idxJ,
   478	                  const std::vector<Triangle>& triangles,
   479	                  const Octree& octree,
   480	                  RandomGenerator& rng) {
   481	    const Triangle& triI = triangles[idxI];
   482	    const Triangle& triJ = triangles[idxJ];
   483	
   484	    // Cull triangles facing the same direction
   485	    if (triI.normal().dot(triJ.normal()) > 0.99f) return ZERO;
   486	
   487	    val_t kij = ZERO;
   488	
   489	    for (int r = 0; r < NUM_RAYS; ++r) {
   490	        Vec3 pI = randomPointInTriangle(triI, rng);
   491	        Vec3 pJ = randomPointInTriangle(triJ, rng);
   492	
   493	        if (isRayBlocked(pI, pJ, octree, idxI, idxJ)) continue;
   494	
   495	        Vec3 v = pJ - pI;
   496	        val_t distSqr = v.squaredNorm();
   497	        if (distSqr < EPSILON) continue;
   498	
   499	        val_t cosPhiI = cosPhi(v, triI.normal());
   500	        val_t cosPhiJ = cosPhi(-v, triJ.normal());
   501	
   502	        if (cosPhiI <= ZERO || cosPhiJ <= ZERO) continue;
   503	
   504	        kij += (cosPhiI * cosPhiJ) / (PI * distSqr);
   505	    }
   506	
   507	    return kij * INV_NUM_RAYS;
   508	}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// Print results for floating-point arrays
template <typename T>
void print_results(const std::vector<T>& data, const char* name = "Result") {
    if (data.empty()) {
        printf("=== RESULTS ===\n");
        printf("Name: %s\n", name);
        printf("Elements: 0\n");
        printf("=== END RESULTS ===\n");
        return;
    }

    const size_t N = data.size();
    
    // Compute statistics
    T sum = kahan_sum(data);
    T min_val = data[0];
    T max_val = data[0];
    for (const auto& val : data) {
        min_val = std::min(min_val, val);
        max_val = std::max(max_val, val);
    }
    
    // Compute hash
    hash h;
    for (const auto& val : data) {
        h.add(val);
    }
    
    // Sample indices: 0, N/4, N/2, 3N/4, N-1
    size_t indices[5] = {0, N/4, N/2, 3*N/4, N-1};
    
    printf("=== RESULTS ===\n");
    printf("Name: %s\n", name);
    printf("Elements: %zu\n", N);
    printf("Sum: %.17e\n", static_cast<double>(sum));
    printf("Min: %.17e\n", static_cast<double>(min_val));
    printf("Max: %.17e\n", static_cast<double>(max_val));
    for (int i = 0; i < 5; ++i) {
        printf("Sample[%zu]: %.17e\n", indices[i], static_cast<double>(data[indices[i]]));
    }
    printf("Hash: %016" PRIx64 "\n", h.get());
    printf("=== END RESULTS ===\n");
}

// Generate an icosphere by subdividing an icosahedron
class IcosphereMesh {
public:
    std::vector<Triangle> triangles;

    IcosphereMesh(int subdivisions, val_t radius) {
        // Initial icosahedron vertices
        const val_t t = (1.0f + std::sqrt(5.0f)) / 2.0f;

        std::vector<Vec3> vertices = {
            Vec3(-1,  t,  0).normalized() * radius,
            Vec3( 1,  t,  0).normalized() * radius,
            Vec3(-1, -t,  0).normalized() * radius,
            Vec3( 1, -t,  0).normalized() * radius,
            Vec3( 0, -1,  t).normalized() * radius,
            Vec3( 0,  1,  t).normalized() * radius,
            Vec3( 0, -1, -t).normalized() * radius,
            Vec3( 0,  1, -t).normalized() * radius,
            Vec3( t,  0, -1).normalized() * radius,
            Vec3( t,  0,  1).normalized() * radius,
            Vec3(-t,  0, -1).normalized() * radius,
            Vec3(-t,  0,  1).normalized() * radius
        };

        // Initial icosahedron faces (20 triangles)
        std::vector<std::array<idx_t, 3>> faces = {
            {0, 11, 5}, {0, 5, 1}, {0, 1, 7}, {0, 7, 10}, {0, 10, 11},
            {1, 5, 9}, {5, 11, 4}, {11, 10, 2}, {10, 7, 6}, {7, 1, 8},
            {3, 9, 4}, {3, 4, 2}, {3, 2, 6}, {3, 6, 8}, {3, 8, 9},
            {4, 9, 5}, {2, 4, 11}, {6, 2, 10}, {8, 6, 7}, {9, 8, 1}
        };

        // Subdivide
        for (int i = 0; i < subdivisions; ++i) {
            std::vector<std::array<idx_t, 3>> newFaces;
            std::map<std::pair<idx_t, idx_t>, idx_t> midpointCache;

            auto getMidpoint = [&](idx_t i1, idx_t i2) -> idx_t {
                auto key = std::make_pair(std::min(i1, i2), std::max(i1, i2));
                auto it = midpointCache.find(key);
                if (it != midpointCache.end()) return it->second;

                Vec3 mid = (vertices[i1] + vertices[i2]) / 2.0f;
                mid = mid.normalized() * radius;
                idx_t idx = static_cast<idx_t>(vertices.size());
                vertices.push_back(mid);
                midpointCache[key] = idx;
                return idx;
            };

            for (const auto& face : faces) {
                idx_t a = getMidpoint(face[0], face[1]);
                idx_t b = getMidpoint(face[1], face[2]);
                idx_t c = getMidpoint(face[2], face[0]);

                newFaces.push_back({face[0], a, c});
                newFaces.push_back({face[1], b, a});
                newFaces.push_back({face[2], c, b});
                newFaces.push_back({a, b, c});
            }
            faces = std::move(newFaces);
        }

        // Build triangles (normals pointing inward for a "room")
        triangles.reserve(faces.size());
        for (const auto& face : faces) {
            // Reverse winding to make normals point inward
            triangles.emplace_back(vertices[face[2]], vertices[face[1]], vertices[face[0]]);
        }
    }
};

class RandomGenerator {
    std::mt19937 rng;
    std::uniform_real_distribution<val_t> dist;
public:
    explicit RandomGenerator(uint32_t seed = 42) : rng(seed), dist(0.0f, 1.0f) {}
    val_t rand() { return dist(rng); }
};

// Generate a random point inside a triangle using barycentric coordinates
Vec3 randomPointInTriangle(const Triangle& t, RandomGenerator& rng) {
    val_t u = rng.rand();
    val_t v = rng.rand();
    if (u + v > 1.0f) {
        u = 1.0f - u;
        v = 1.0f - v;
    }
    Vec3 ab = t.b - t.a;
    Vec3 ac = t.c - t.a;
    return t.a + ab * u + ac * v;
}

// ============================================================================
// Ray-Triangle Intersection (Möller-Trumbore algorithm)
// ============================================================================

val_t rayTriangleIntersect(const Vec3& orig, const Vec3& dir,
                            const Vec3& v0, const Vec3& v1, const Vec3& v2) {
    Vec3 e1 = v1 - v0;
    Vec3 e2 = v2 - v0;
    Vec3 pvec = dir.cross(e2);
    val_t det = e1.dot(pvec);

    if (std::abs(det) < EPSILON) return std::numeric_limits<val_t>::max();

    val_t invDet = 1.0f / det;
    Vec3 tvec = orig - v0;
    val_t u = tvec.dot(pvec) * invDet;
    if (u < 0.0f || u > 1.0f) return std::numeric_limits<val_t>::max();

    Vec3 qvec = tvec.cross(e1);
    val_t v = dir.dot(qvec) * invDet;
    if (v < 0.0f || u + v > 1.0f) return std::numeric_limits<val_t>::max();

    return e2.dot(qvec) * invDet;
}

// Check if a ray between two triangles is blocked by any other triangle
// Uses octree for O(log N) instead of O(N) search
bool isRayBlocked(const Vec3& from, const Vec3& to,
                  const Octree& octree,
                  size_t srcTriIdx, size_t dstTriIdx) {
    Vec3 dir = to - from;
    val_t rayLen = dir.norm();
    if (rayLen < EPSILON) return true;
    Vec3 dirNorm = dir / rayLen;

    return octree.applyToTris(from, to, [&](size_t idx, const Triangle& tri) {
        if (idx == srcTriIdx || idx == dstTriIdx) return false;

        val_t dist = rayTriangleIntersect(from, dirNorm, tri.a, tri.b, tri.c);
        if (dist > EPSILON && dist < rayLen - EPSILON) {
            return true;  // Ray is blocked
        }
        return false;
    });
}

// Compute the cosine of angle between vector and triangle normal
val_t cosPhi(const Vec3& v, const Vec3& normal) {
    val_t vNorm = v.norm();
    if (vNorm <= EPSILON) return ZERO;
    return std::max(ZERO, v.dot(normal) / vNorm);
}

// Compute form factor Kij between triangle i (receiver) and triangle j (emitter)
val_t computeKij(size_t idxI, size_t idxJ,
                  const std::vector<Triangle>& triangles,
                  const Octree& octree,
                  RandomGenerator& rng) {
    const Triangle& triI = triangles[idxI];
    const Triangle& triJ = triangles[idxJ];

    // Cull triangles facing the same direction
    if (triI.normal().dot(triJ.normal()) > 0.99f) return ZERO;

    val_t kij = ZERO;

    for (int r = 0; r < NUM_RAYS; ++r) {
        Vec3 pI = randomPointInTriangle(triI, rng);
        Vec3 pJ = randomPointInTriangle(triJ, rng);

        if (isRayBlocked(pI, pJ, octree, idxI, idxJ)) continue;

        Vec3 v = pJ - pI;
        val_t distSqr = v.squaredNorm();
        if (distSqr < EPSILON) continue;

        val_t cosPhiI = cosPhi(v, triI.normal());
        val_t cosPhiJ = cosPhi(-v, triJ.normal());

        if (cosPhiI <= ZERO || cosPhiJ <= ZERO) continue;

        kij += (cosPhiI * cosPhiJ) / (PI * distSqr);
    }

    return kij * INV_NUM_RAYS;
}

// ============================================================================
// Tau (time delay) Computation
// ============================================================================

int computeTau(const Triangle& triI, const Triangle& triJ) {
    val_t dist = (triI.center() - triJ.center()).norm();
    return static_cast<int>(std::ceil(dist * INV_WAVE_SPEED));
}
