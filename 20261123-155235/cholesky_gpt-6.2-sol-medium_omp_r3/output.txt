// Simple Cholesky decomposition (sequential, unblocked algorithm)
// Decomposes positive definite matrix A into L * L^T where L is lower triangular

bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
    // A is stored in row-major order
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j <= i; ++j) {
            double sum = 0.0;
            
            if (i == j) {
                // Diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[j * n + k] * A[j * n + k];
                }
                const double val = A[j * n + j] - sum;
                if (val <= 0.0) {
                    // Matrix is not positive definite
                    printf("Error: Matrix is not positive definite at diagonal element %zu\n", j);
                    return false;
                }
                A[j * n + j] = sqrt(val);
            } else {
                // Off-diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[i * n + k] * A[j * n + k];
                }
                A[i * n + j] = (A[i * n + j] - sum) / A[j * n + j];
            }
        }
        
        // Zero out upper triangular part
        for (size_t j = i + 1; j < n; ++j) {
            A[i * n + j] = 0.0;
        }
    }
    
    return true;
}

void generatePositiveDefiniteMatrix(std::vector<double>& A, const size_t n) {

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

   150	void updateTile(double* matrix, size_t n, size_t rowBegin, size_t rowEnd,
   151	                size_t colBegin, size_t colEnd, size_t panelBegin, size_t panelEnd) {
   152	    constexpr size_t rowsPerBlock = 4;
   153	    const bool diagonal = rowBegin == colBegin;
   154	    size_t i = rowBegin;
   155	    for (; i + rowsPerBlock <= rowEnd; i += rowsPerBlock) {
   156	        updateRows<rowsPerBlock>(matrix, n, i, colBegin, colEnd,
   157	                                 diagonal ? i + rowsPerBlock : colEnd, panelBegin, panelEnd);
   158	    }
   159	    for (; i < rowEnd; ++i) {
   160	        updateRows<1>(matrix, n, i, colBegin, colEnd, diagonal ? i + 1 : colEnd, panelBegin, panelEnd);
   161	    }

   165	// Factor the lower triangle in place as A = L * L^T.
   166	bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
   167	    double* matrix = A.data();
   168	    std::atomic<size_t> failedPivot{n};
   169	
   170	    #pragma omp parallel num_threads(teamSize(n))
   171	    {
   172	        #pragma omp single
   173	        {
   174	            // The first element identifies each tile, including a panel tile's transpose.
   175	            for (size_t k = 0; k < n; k += tileSize) {
   176	                const size_t kEnd = std::min(k + tileSize, n);
   177	                const size_t diagonal = k * n + k;
   178	                #pragma omp task depend(inout: matrix[diagonal])
   179	                {
   180	                    if (failedPivot.load(std::memory_order_relaxed) == n) {
   181	                        const size_t pivot = factorDiagonal(matrix, n, k, kEnd);
   182	                        if (pivot != n) {
   183	                            failedPivot.store(pivot, std::memory_order_relaxed);
   184	                        }
   185	                    }
   186	                }
   187	
   188	                for (size_t i = kEnd; i < n; i += tileSize) {
   189	                    const size_t iEnd = std::min(i + tileSize, n);
   190	                    const size_t panel = i * n + k;
   191	                    #pragma omp task depend(in: matrix[diagonal]) depend(inout: matrix[panel])
   192	                    {
   193	                        if (failedPivot.load(std::memory_order_relaxed) == n) {
   194	                            solvePanel(matrix, n, i, iEnd, k, kEnd);
   195	                        }
   196	                    }
   197	                }
   198	
   199	                for (size_t i = kEnd; i < n; i += tileSize) {
   200	                    const size_t iEnd = std::min(i + tileSize, n);
   201	                    const size_t left = i * n + k;
   202	                    #pragma omp task depend(in: matrix[left]) depend(inout: matrix[i * n + i])
   203	                    {
   204	                        if (failedPivot.load(std::memory_order_relaxed) == n) {
   205	                            updateTile(matrix, n, i, iEnd, i, iEnd, k, kEnd);
   206	                        }
   207	                    }
   208	                    for (size_t j = kEnd; j < i; j += tileSize) {
   209	                        const size_t jEnd = std::min(j + tileSize, n);
   210	                        const size_t right = j * n + k;
   211	                        const size_t target = i * n + j;
   212	                        #pragma omp task depend(in: matrix[left], matrix[right]) depend(inout: matrix[target])
   213	                        {
   214	                            if (failedPivot.load(std::memory_order_relaxed) == n) {
   215	                                updateTile(matrix, n, i, iEnd, j, jEnd, k, kEnd);
   216	                            }
   217	                        }
   218	                    }
   219	                }
   220	            }
   221	        } // The barrier also waits for all tile tasks.
   222	
   223	        #pragma omp for schedule(static, 1)
   224	        for (size_t i = 0; i < n; ++i) {
   225	            std::fill(matrix + i * n + i + 1, matrix + (i + 1) * n, 0.0);
   226	        }
   227	    }
   228	
   229	    const size_t pivot = failedPivot.load(std::memory_order_relaxed);
   230	    if (pivot != n) {
   231	        printf("Error: Matrix is not positive definite at diagonal element %zu\n", pivot);
   232	        return false;
   233	    }
   234	    return true;
   235	}

   237	void generatePositiveDefiniteMatrix(std::vector<double>& A, const size_t n) {
   238	    std::vector<double> B(n * n);
   239	    unsigned int seed = 42;
   240	    for (double& value : B) {
   241	        value = (rand_r(&seed) / (double)RAND_MAX) - 0.5;
   242	    }
   243	
   244	    // Compute one triangle of B * B^T, then mirror it.
   245	    #pragma omp parallel for num_threads(teamSize(n)) schedule(static, 1)
   246	    for (size_t ii = 0; ii < n; ii += gramTileSize) {
   247	        const size_t iEnd = std::min(ii + gramTileSize, n);
   248	        for (size_t jj = 0; jj <= ii; jj += gramTileSize) {
   249	            const size_t jEnd = std::min(jj + gramTileSize, n);
   250	            for (size_t i = ii; i < iEnd; ++i) {
   251	                for (size_t j = jj; j < std::min(jEnd, i); ++j) {
   252	                    const double value = gramDotProduct(B.data() + i * n, B.data() + j * n, n);
   253	                    A[i * n + j] = value;
   254	                    A[j * n + i] = value;
   255	                }
   256	            }
   257	        }
   258	        for (size_t i = ii; i < iEnd; ++i) {
   259	            const double* row = B.data() + i * n;
   260	            A[i * n + i] = gramDotProduct(row, row, n) + n;
   261	        }
   262	    }
   263	}
