// Simple Cholesky decomposition (sequential, unblocked algorithm)
// Decomposes positive definite matrix A into L * L^T where L is lower triangular

bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
    // A is stored in row-major order
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j <= i; ++j) {
            double sum = 0.0;
            
            if (i == j) {
                // Diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[j * n + k] * A[j * n + k];
                }
                const double val = A[j * n + j] - sum;
                if (val <= 0.0) {
                    // Matrix is not positive definite
                    printf("Error: Matrix is not positive definite at diagonal element %zu\n", j);
                    return false;
                }
                A[j * n + j] = sqrt(val);
            } else {
                // Off-diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[i * n + k] * A[j * n + k];
                }
                A[i * n + j] = (A[i * n + j] - sum) / A[j * n + j];
            }
        }
        
        // Zero out upper triangular part
        for (size_t j = i + 1; j < n; ++j) {
            A[i * n + j] = 0.0;
        }
    }
    
    return true;
}

// Generate a symmetric positive definite matrix
void generatePositiveDefiniteMatrix(std::vector<double>& A, const size_t n) {
    // Method: Create A = B * B^T where B is random
    // This guarantees A is positive semi-definite
    // Then add identity to make it strictly positive definite
    
    std::vector<double> B(n * n);
    unsigned int seed = 42;
    
    // Generate random matrix B
    for (size_t i = 0; i < n * n; ++i) {
        B[i] = (rand_r(&seed) / (double)RAND_MAX) - 0.5;
    }
    
    // Compute A = B * B^T
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < n; ++k) {
                sum += B[i * n + k] * B[j * n + k];
            }
            A[i * n + j] = sum;
        }
    }
    
    // Add diagonal dominance to ensure positive definiteness
    for (size_t i = 0; i < n; ++i) {
        A[i * n + i] += n;
    }
}

bool validateCholesky(const std::vector<double>& L, const std::vector<double>& A_orig, const size_t n) {
    // Validate by computing L * L^T and comparing with original matrix
    
    std::vector<double> reconstructed(n * n);
    
    // Compute L * L^T
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < n; ++k) {
                sum += L[i * n + k] * L[j * n + k];
            }
            reconstructed[i * n + j] = sum;
        }
    }
    
    // Compare with original
    double maxError = 0.0;
    double relError = 0.0;
    
    for (size_t i = 0; i < n * n; ++i) {
        const double error = fabs(reconstructed[i] - A_orig[i]);
        maxError = std::max(maxError, error);
        
        const double rel = error / (fabs(A_orig[i]) + 1e-10);
        relError = std::max(relError, rel);
    }
    
    printf("Max absolute error: %.10e\n", maxError);
    printf("Max relative error: %.10e\n", relError);
    
    // Check if error is within tolerance
    if (relError > 1e-6) {
        printf("Validation failed: relative error too large\n");
        return false;
    }
    
    return true;
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// FNV-1a hash, 64 bit length
class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:

__global__ void shiftDiagonal(double* matrix, int n) {
    const size_t i = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;

// cuBLAS and cuSOLVER see the transpose of the row-major matrix.
constexpr auto triangle = CUBLAS_FILL_MODE_UPPER;

void generatePositiveDefiniteMatrix(cublasHandle_t blas, double* matrix, int n) {
    const size_t count = static_cast<size_t>(n) * n;
    DeviceBuffer<double> input(count);
    std::vector<double> random(count);
    unsigned int seed = 42;
    for (double& value : random) {
        value = rand_r(&seed) / static_cast<double>(RAND_MAX) - 0.5;
    }

    const double zero = 0.0;

                         &one, input.get(), n, &zero, matrix, n));

    checkCuda(cudaGetLastError());
}

bool choleskyDecomposition(CudaHandles& cuda, double* matrix, int n, float& milliseconds) {

    checkCuda(cudaEventCreate(&cuda.stop));

    int workspaceSize = 0;

    DeviceBuffer<double> workspace(workspaceSize);
    DeviceBuffer<int> deviceInfo(1);

    checkCuda(cudaEventRecord(cuda.start));
    checkSolver(cusolverDnDpotrf(cuda.solver, triangle, n, matrix, n,
                               workspace.get(), workspaceSize, deviceInfo.get()));
    checkCuda(cudaEventRecord(cuda.stop));

    checkCuda(cudaEventElapsedTime(&milliseconds, cuda.start, cuda.stop));

__global__ void zeroUpperTriangle(double* matrix, int n, size_t count) {
    const size_t start = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    const size_t stride = static_cast<size_t>(gridDim.x) * blockDim.x;
    for (size_t i = start; i < count; i += stride) {
        if (i % n > i / n) {
            matrix[i] = 0.0;
        }
    }

struct Errors {
    double absolute;
    double relative;
};

__global__ void reconstructionErrors(const double* reconstructed, const double* original,
                                     int n, size_t count, Errors* blockErrors) {
    const size_t start = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    const size_t stride = static_cast<size_t>(gridDim.x) * blockDim.x;
    Errors worst{};
    for (size_t i = start; i < count; i += stride) {
        if (i % n <= i / n) {
            const double expected = original[i];
            const double error = fabs(reconstructed[i] - expected);
            worst.absolute = fmax(worst.absolute, error);
            worst.relative = fmax(worst.relative, error / (fabs(expected) + 1e-10));
        }
    }

    __shared__ Errors shared[blockSize];
    const int lane = threadIdx.x;
    shared[lane] = worst;
    __syncthreads();
    for (int offset = blockSize / 2; offset > 0; offset /= 2) {
        if (lane < offset) {
            shared[lane].absolute = fmax(shared[lane].absolute, shared[lane + offset].absolute);
            shared[lane].relative = fmax(shared[lane].relative, shared[lane + offset].relative);
        }
        __syncthreads();
    }
    if (lane == 0) {
        blockErrors[blockIdx.x] = shared[0];
    }
}

Errors validateCholesky(cublasHandle_t blas, const double* factor, const double* original, int n) {
    const size_t count = static_cast<size_t>(n) * n;
    DeviceBuffer<double> reconstructed(count);
    const int blocks = matrixBlocks(count);
    DeviceBuffer<Errors> deviceErrors(blocks);
    const double one = 1.0;
    const double zero = 0.0;
    checkBlas(cublasDsyrk(blas, triangle, CUBLAS_OP_T, n, n,
                         &one, factor, n, &zero, reconstructed.get(), n));

                                                deviceErrors.get());

                         cudaMemcpyDeviceToHost));

        worst.absolute = std::max(worst.absolute, errors.absolute);
        worst.relative = std::max(worst.relative, errors.relative);
    }
    return worst;
}

   151	                                     int n, Errors* blockErrors) {
   152	    Errors worst{};
   153	    for (int row = blockIdx.x; row < n; row += gridDim.x) {
   154	        const size_t offset = static_cast<size_t>(row) * n;
   155	        for (int col = threadIdx.x; col <= row; col += blockDim.x) {
   156	            const double expected = original[offset + col];
   157	            const double error = fabs(reconstructed[offset + col] - expected);
   158	            if (!isfinite(error)) {
   159	                worst = {CUDART_INF, CUDART_INF};
   160	                continue;
   161	            }
   162	            worst.absolute = fmax(worst.absolute, error);
   163	            worst.relative = fmax(worst.relative, error / (fabs(expected) + 1e-10));
   164	        }
   165	    }

   168	    const int lane = threadIdx.x;

   172	        if (lane < offset) {

   174	            shared[lane].relative = fmax(shared[lane].relative, shared[lane + offset].relative);
   175	        }

   177	    }
