constexpr unsigned int INF = 1000000000;
constexpr unsigned int MAX_DISTANCE = 200;

// Index calculation for flattened 2D array
inline constexpr size_t idx2(const size_t i, const size_t j, const size_t n) noexcept {
    return j * n + i;
}

void initializeDistanceMatrix(std::vector<unsigned int>& dist, const size_t numNodes, 
                              const unsigned int rangeMin, const unsigned int rangeMax) {
    unsigned int seed = 42;
    const double range = static_cast<double>(rangeMax - rangeMin) + 1.0;
    
    for (size_t i = 0; i < numNodes * numNodes; ++i) {
        dist[i] = rangeMin + (unsigned int)(range * rand_r(&seed) / (double)RAND_MAX);
    }
    
    // Set diagonal to zero (distance from node to itself is 0)
    for (size_t i = 0; i < numNodes; ++i) {
        dist[idx2(i, i, numNodes)] = 0;
    }
}

void initializePathMatrix(std::vector<unsigned int>& path, const size_t numNodes) {
    for (size_t j = 0; j < numNodes; ++j) {
        for (size_t i = 0; i < numNodes; ++i) {
            path[idx2(i, j, numNodes)] = j;
            path[idx2(j, i, numNodes)] = i;
        }
        path[idx2(j, j, numNodes)] = j;
    }
}

void floydWarshall(std::vector<unsigned int>& dist, 
                   std::vector<unsigned int>& path, 
                   const size_t numNodes) {
    // Classic Floyd-Warshall algorithm
    // For each intermediate node k
    for (size_t k = 0; k < numNodes; ++k) {
        // For each source node i
        for (size_t i = 0; i < numNodes; ++i) {
            // For each destination node j
            for (size_t j = 0; j < numNodes; ++j) {
                const unsigned int distIJ = dist[idx2(j, i, numNodes)];
                const unsigned int distIK = dist[idx2(k, i, numNodes)];
                const unsigned int distKJ = dist[idx2(j, k, numNodes)];
                
                const unsigned int newDist = distIK + distKJ;
                
                if (newDist < distIJ) {
                    dist[idx2(j, i, numNodes)] = newDist;
                    path[idx2(j, i, numNodes)] = k;
                }
            }
        }
    }
}

bool validateResult(const std::vector<unsigned int>& dist, const size_t numNodes) {
    // Basic sanity checks
    
    // 1. Diagonal should be zero
    for (size_t i = 0; i < numNodes; ++i) {
        if (dist[idx2(i, i, numNodes)] != 0) {
            printf("Validation failed: diagonal element [%zu,%zu] is not zero\n", i, i);
            return false;
        }
    }
    
    // 2. Triangle inequality: dist[i][k] + dist[k][j] >= dist[i][j]
    // Check a sample of paths to avoid O(n^3) validation time
    for (size_t i = 0; i < std::min(numNodes, static_cast<size_t>(10)); ++i) {
        for (size_t j = 0; j < std::min(numNodes, static_cast<size_t>(10)); ++j) {
            for (size_t k = 0; k < numNodes; ++k) {
                const unsigned int distIJ = dist[idx2(j, i, numNodes)];
                const unsigned int distIK = dist[idx2(k, i, numNodes)];
                const unsigned int distKJ = dist[idx2(j, k, numNodes)];
                
                // Check for overflow before addition
                if (distIK < INF && distKJ < INF) {
                    if (distIK + distKJ < distIJ) {
                        printf("Validation failed: triangle inequality violated at [%zu,%zu,%zu]\n", 
                               i, j, k);
                        return false;
                    }
                }
            }
        }
    }
    
    return true;
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

template <typename T>
void print_results(const std::vector<T>& data, const char* name = "Result") {
    if (data.empty()) {

        printf("Elements: 0\n");
        printf("=== END RESULTS ===\n");
        return;
    }

    const size_t N = data.size();
    
    // Compute statistics
    T sum = kahan_sum(data);
    T min_val = data[0];
    T max_val = data[0];

        min_val = std::min(min_val, val);
        max_val = std::max(max_val, val);
    }
    
    // Compute hash
    hash h;
    for (const auto& val : data) {
        h.add(val);

// FNV-1a hash, 64 bit length
class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

    60	template <int TileSize>
    61	__global__ void packDistances(unsigned int* matrix, int numNodes, int stride) {
    62	    const int firstCol = blockIdx.x * TileSize + threadIdx.x;
    63	    const int firstRow = blockIdx.y * TileSize + threadIdx.y;
    64	
    65	    #pragma unroll
    66	    for (int r = 0; r < TileSize; r += blockRows) {
    67	        const int row = firstRow + r;
    68	        #pragma unroll
    69	        for (int c = 0; c < TileSize; c += blockCols) {
    70	            const int col = firstCol + c;
    71	            auto& entry = matrix[static_cast<size_t>(row) * stride + col];
    72	            const unsigned int distance =
    73	                row < numNodes && col < numNodes ? entry : paddingDistance;
    74	            entry = distance << distanceShift;
    75	        }
    76	    }
    77	}

    79	// A shortest path is labeled by its largest intermediate index plus one.
    80	// Minimizing that label on distance ties gives the first scalar round that
    81	// reaches the final distance. The label is zero for an original edge.
    82	__device__ __forceinline__ unsigned int through(unsigned int left, unsigned int right,
    83	                                                unsigned int pivot) {
    84	    const unsigned int distance = (left & distanceMask) + (right & distanceMask);
    85	    const unsigned int witness = max(pivot, max(left & witnessMask, right & witnessMask));
    86	    return distance | witness;
    87	}

    89	template <int TileSize>
    90	__global__ void closePivotTile(unsigned int* matrix, int stride, int pivotTile) {
    91	    __shared__ unsigned int tile[TileSize][TileSize];
    92	    const int x = threadIdx.x;
    93	    const int y = threadIdx.y;
    94	    const int first = pivotTile * TileSize;
    95	    auto* const block = matrix + static_cast<size_t>(first) * stride + first;
    96	
    97	    #pragma unroll
    98	    for (int r = 0; r < TileSize; r += blockRows) {
    99	        const int row = y + r;
   100	        #pragma unroll
   101	        for (int c = 0; c < TileSize; c += blockCols) {
   102	            const int col = x + c;
   103	            tile[row][col] = block[static_cast<size_t>(row) * stride + col];
   104	        }
   105	    }
   106	    __syncthreads();
   107	
   108	    for (int k = 0; k < TileSize; ++k) {
   109	        // The pivot row and column cannot improve through their own vertex.
   110	        // Leaving them untouched also avoids shared-memory read/write races.
   111	        #pragma unroll
   112	        for (int c = 0; c < TileSize; c += blockCols) {
   113	            const int col = x + c;
   114	            if (col != k) {
   115	                const unsigned int right = tile[k][col];
   116	                #pragma unroll
   117	                for (int r = 0; r < TileSize; r += blockRows) {
   118	                    const int row = y + r;
   119	                    if (row != k) {
   120	                        const unsigned int candidate = through(tile[row][k], right, first + k + 1);
   121	                        tile[row][col] = min(tile[row][col], candidate);
   122	                    }
   123	                }
   124	            }
   125	        }
   126	        __syncthreads();
   127	    }
   128	
   129	    #pragma unroll
   130	    for (int r = 0; r < TileSize; r += blockRows) {
   131	        const int row = y + r;
   132	        #pragma unroll
   133	        for (int c = 0; c < TileSize; c += blockCols) {
   134	            const int col = x + c;
   135	            block[static_cast<size_t>(row) * stride + col] = tile[row][col];
   136	        }
   137	    }
   138	}

   141	    return tile + (tile >= pivotTile);

   144	template <int TileSize, bool Border>
   145	__global__ void relaxTiles(unsigned int* matrix, int stride, int pivotTile) {
   146	    __shared__ unsigned int left[TileSize][TileSize];
   147	    __shared__ unsigned int right[TileSize][TileSize];
   148	    const int x = threadIdx.x;
   149	    const int y = threadIdx.y;
   150	    const int pivot = pivotTile * TileSize;
   151	    int col = skipPivot(blockIdx.x, pivotTile) * TileSize;
   152	    int row;
   153	    if constexpr (Border) {
   154	        row = pivot;
   155	        if (blockIdx.y != 0) {
   156	            row = col;
   157	            col = pivot;
   158	        }
   159	    } else {
   160	        row = skipPivot(blockIdx.y, pivotTile) * TileSize;
   161	    }
   162	    auto* const output = matrix + static_cast<size_t>(row) * stride + col;
   163	    const auto* const leftInput = matrix + static_cast<size_t>(row) * stride + pivot;
   164	    const auto* const rightInput = matrix + static_cast<size_t>(pivot) * stride + col;
   165	
   166	    // The closed pivot tile lets each border tile be updated by one product
   167	    // with its original values. The remaining tiles use the updated borders.
   168	    constexpr int rowsPerThread = TileSize / blockRows;
   169	    constexpr int colsPerThread = TileSize / blockCols;
   170	    unsigned int best[rowsPerThread][colsPerThread];
   171	    #pragma unroll
   172	    for (int r = 0; r < rowsPerThread; ++r) {
   173	        const int localRow = y + r * blockRows;
   174	        #pragma unroll
   175	        for (int c = 0; c < colsPerThread; ++c) {
   176	            const int localCol = x + c * blockCols;
   177	            const size_t offset = static_cast<size_t>(localRow) * stride + localCol;
   178	            best[r][c] = output[offset];
   179	            left[localRow][localCol] = leftInput[offset];
   180	            right[localRow][localCol] = rightInput[offset];
   181	        }
   182	    }
   183	    __syncthreads();
   184	
   185	    #pragma unroll 8
   186	    for (int k = 0; k < TileSize; ++k) {
   187	        unsigned int rightDistance[colsPerThread];
   188	        unsigned int rightWitness[colsPerThread];
   189	        #pragma unroll
   190	        for (int c = 0; c < colsPerThread; ++c) {
   191	            const unsigned int entry = right[k][x + c * blockCols];
   192	            rightDistance[c] = entry & distanceMask;
   193	            rightWitness[c] = max(static_cast<unsigned int>(pivot + k + 1), entry & witnessMask);
   194	        }
   195	        #pragma unroll
   196	        for (int r = 0; r < rowsPerThread; ++r) {
   197	            const unsigned int entry = left[y + r * blockRows][k];
   198	            const unsigned int distance = entry & distanceMask;
   199	            const unsigned int witness = entry & witnessMask;
   200	            #pragma unroll
   201	            for (int c = 0; c < colsPerThread; ++c) {
   202	                const unsigned int candidate = (distance + rightDistance[c])
   203	                                             | max(witness, rightWitness[c]);
   204	                best[r][c] = min(best[r][c], candidate);
   205	            }
   206	        }
   207	    }
   208	
   209	    #pragma unroll
   210	    for (int r = 0; r < rowsPerThread; ++r) {
   211	        const size_t rowOffset = static_cast<size_t>(y + r * blockRows) * stride;
   212	        #pragma unroll
   213	        for (int c = 0; c < colsPerThread; ++c) {
   214	            output[rowOffset + x + c * blockCols] = best[r][c];
   215	        }
   216	    }
   217	}

   219	template <int TileSize>
   220	double runCuda(std::vector<unsigned int>& dist, std::vector<unsigned int>& path, size_t numNodes) {
   221	    const int tiles = (numNodes + TileSize - 1) / TileSize;
   222	    const int stride = tiles * TileSize;
   223	    const size_t pitch = static_cast<size_t>(stride) * sizeof(unsigned int);
   224	    const size_t rowBytes = numNodes * sizeof(unsigned int);
   225	    const dim3 threads(blockCols, blockRows);
   226	    unsigned int* deviceMatrix;
   227	    checkCuda(cudaMalloc(&deviceMatrix, stride * pitch), "Allocating CUDA distance matrix");
   228	    checkCuda(cudaMemcpy2D(deviceMatrix, pitch, dist.data(), rowBytes, rowBytes, numNodes,
   229	                           cudaMemcpyHostToDevice), "Uploading distance matrix");
   230	    packDistances<TileSize><<<dim3(tiles, tiles), threads>>>(deviceMatrix, numNodes, stride);
   231	    checkCuda(cudaGetLastError(), "Launching distance packing kernel");
   232	
   233	    cudaEvent_t start;
   234	    cudaEvent_t stop;
   235	    checkCuda(cudaEventCreate(&start), "Creating CUDA start event");
   236	    checkCuda(cudaEventCreate(&stop), "Creating CUDA stop event");
   237	    checkCuda(cudaEventRecord(start), "Recording CUDA start event");
   238	    for (int pivot = 0; pivot < tiles; ++pivot) {
   239	        closePivotTile<TileSize><<<1, threads>>>(deviceMatrix, stride, pivot);
   240	        if (tiles > 1) {
   241	            relaxTiles<TileSize, true><<<dim3(tiles - 1, 2), threads>>>(deviceMatrix, stride, pivot);
   242	            relaxTiles<TileSize, false><<<dim3(tiles - 1, tiles - 1), threads>>>(deviceMatrix, stride, pivot);
   243	        }
   244	    }
   245	    checkCuda(cudaGetLastError(), "Launching Floyd-Warshall kernels");
   246	    checkCuda(cudaEventRecord(stop), "Recording CUDA stop event");
   247	    checkCuda(cudaEventSynchronize(stop), "Running Floyd-Warshall kernels");
   248	
   249	    float milliseconds;
   250	    checkCuda(cudaEventElapsedTime(&milliseconds, start, stop), "Measuring CUDA computation time");
   251	    checkCuda(cudaEventDestroy(start), "Destroying CUDA start event");
   252	    checkCuda(cudaEventDestroy(stop), "Destroying CUDA stop event");
   253	    checkCuda(cudaMemcpy2D(dist.data(), rowBytes, deviceMatrix, pitch, rowBytes, numNodes,
   254	                           cudaMemcpyDeviceToHost), "Downloading shortest paths");
   255	    checkCuda(cudaFree(deviceMatrix), "Freeing CUDA distance matrix");
   256	
   257	    for (size_t i = 0; i < dist.size(); ++i) {
   258	        const unsigned int witness = dist[i] & witnessMask;
   259	        dist[i] >>= distanceShift;
   260	        if (witness != 0) {
   261	            path[i] = witness - 1;
   262	        }
   263	    }
   264	    return milliseconds;
   265	}

   269	double floydWarshall(std::vector<unsigned int>& dist, std::vector<unsigned int>& path,
   270	                     size_t numNodes) {
   271	    if (numNodes == 0) {
   272	        return 0.0;
   273	    }
   274	    // Smaller tiles expose more blocks on small graphs. Larger tiles reuse
   275	    // each shared-memory load across more output entries on large graphs.
   276	    return numNodes < 3072 ? runCuda<smallTile>(dist, path, numNodes)
   277	                           : runCuda<largeTile>(dist, path, numNodes);
   278	}

   280	bool validateResult(const std::vector<unsigned int>& dist, const size_t numNodes) {
   281	    // Basic sanity checks
   282	    
   283	    // 1. Diagonal should be zero
   284	    for (size_t i = 0; i < numNodes; ++i) {
   285	        if (dist[idx2(i, i, numNodes)] != 0) {
   286	            printf("Validation failed: diagonal element [%zu,%zu] is not zero\n", i, i);
   287	            return false;
   288	        }
   289	    }
   290	    
   291	    // 2. Triangle inequality: dist[i][k] + dist[k][j] >= dist[i][j]
   292	    // Check a sample of paths to avoid O(n^3) validation time
   293	    for (size_t i = 0; i < std::min(numNodes, static_cast<size_t>(10)); ++i) {
   294	        for (size_t j = 0; j < std::min(numNodes, static_cast<size_t>(10)); ++j) {
   295	            for (size_t k = 0; k < numNodes; ++k) {
   296	                const unsigned int distIJ = dist[idx2(j, i, numNodes)];
   297	                const unsigned int distIK = dist[idx2(k, i, numNodes)];
   298	                const unsigned int distKJ = dist[idx2(j, k, numNodes)];
   299	                
   300	                // Check for overflow before addition
   301	                if (distIK < INF && distKJ < INF) {
   302	                    if (distIK + distKJ < distIJ) {
   303	                        printf("Validation failed: triangle inequality violated at [%zu,%zu,%zu]\n", 
   304	                               i, j, k);
   305	                        return false;
   306	                    }
   307	                }
   308	            }
   309	        }
   310	    }
   311	    
   312	    return true;
   313	}
