cmake_minimum_required(VERSION 3.20)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES AND "$ENV{CUDAARCHS}" STREQUAL "")
    set(use_default_cuda_architectures TRUE)
endif()
project(black_scholes_benchmark LANGUAGES CXX CUDA)

find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED COMPONENTS CXX)

if(use_default_cuda_architectures)
    set(cuda_architectures 75-real 80-real)
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13)
        list(PREPEND cuda_architectures 60-real 70-real)
    endif()
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 11.1)
        list(APPEND cuda_architectures 86-real)
    endif()
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 11.8)
        list(APPEND cuda_architectures 90-real 90-virtual)
    else()
        list(APPEND cuda_architectures 80-virtual)
    endif()
    set(CMAKE_CUDA_ARCHITECTURES "${cuda_architectures}" CACHE STRING "CUDA architectures" FORCE)
endif()
set_source_files_properties(black_scholes.cpp PROPERTIES LANGUAGE CUDA)
add_executable(black_scholes black_scholes.cpp)
set_target_properties(black_scholes PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
target_compile_options(black_scholes PRIVATE -O3 --fmad=false
    "-Xcompiler=${OpenMP_CXX_FLAGS},-Wall,-Wextra")
target_compile_definitions(black_scholes PRIVATE OMPI_SKIP_MPICXX MPICH_SKIP_MPICXX)
target_link_libraries(black_scholes PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX)
