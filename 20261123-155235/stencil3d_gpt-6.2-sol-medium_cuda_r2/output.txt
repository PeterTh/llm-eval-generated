// 7-point stencil computation
void stencilIteration(const std::vector<Real>& input, 
                      std::vector<Real>& output,
                      const size_t nx, const size_t ny, const size_t nz) {
    // Process interior points (not on boundaries)
    for (size_t z = 1; z < nz - 1; ++z) {
        for (size_t y = 1; y < ny - 1; ++y) {
            for (size_t x = 1; x < nx - 1; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                
                const Real center = input[idx];
                const Real left = input[idx3(x-1, y, z, nx, ny)];
                const Real right = input[idx3(x+1, y, z, nx, ny)];
                const Real front = input[idx3(x, y-1, z, nx, ny)];
                const Real back = input[idx3(x, y+1, z, nx, ny)];
                const Real bottom = input[idx3(x, y, z-1, nx, ny)];
                const Real top = input[idx3(x, y, z+1, nx, ny)];
                
                // Simple averaging stencil
                output[idx] = (center + left + right + front + back + bottom + top) / 7.0;
            }
        }
    }
    
    // Copy boundary values
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                if (x == 0 || x == nx-1 || y == 0 || y == ny-1 || z == 0 || z == nz-1) {
                    const size_t idx = idx3(x, y, z, nx, ny);
                    output[idx] = input[idx];
                }
            }
        }
    }
}

// FNV-1a hash, 64 bit length
class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

__global__ void initializeGrid(Real* first, Real* second, size_t count,
                               size_t nx, size_t ny, size_t nz, bool zeroSecond) {
    const size_t stride = static_cast<size_t>(blockDim.x) * gridDim.x;
    for (size_t index = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
         index < count; index += stride) {
        const Real value = static_cast<Real>(index % 19);
        first[index] = value;
        if (zeroSecond) {
            second[index] = 0.0;
        } else {
            const size_t row = index / nx;
            const size_t x = index % nx;
            const size_t y = row % ny;
            const size_t z = row / ny;
            // Initialize the fixed faces in both buffers so kernels only write interior cells.
            if (x == 0 || x == nx - 1 || y == 0 || y == ny - 1 || z == 0 || z == nz - 1) {
                second[index] = value;
            }
        }
    }
}

constexpr unsigned blockX = 32;
constexpr unsigned blockY = 4;
constexpr unsigned planesPerThread = 4;

template <typename Index>
__global__ void stencilIteration(const Real* __restrict__ input, Real* __restrict__ output,
                                 Index nx, Index ny, Index nz, Index yTiles, Index yzTiles) {
    const Index x = static_cast<Index>(blockIdx.x) * blockX + threadIdx.x;
    if (x == 0 || x >= nx - 1) {
        return;
    }

    const Index plane = nx * ny;
    for (Index tile = blockIdx.y; tile < yzTiles; tile += gridDim.y) {
        const Index y = (tile % yTiles) * blockY + threadIdx.y + 1;
        if (y >= ny - 1) {
            continue;
        }

        const Index z = (tile / yTiles) * planesPerThread + 1;
        Index index = z * plane + y * nx + x;
        Real bottom = input[index - plane];
        Real center = input[index];

        // Reuse the three values along Z as the thread advances through its planes.
        #pragma unroll
        for (unsigned offset = 0; offset < planesPerThread; ++offset) {
            if (z + offset >= nz - 1) {
                break;
            }

            const Real top = input[index + plane];
            const Real sum = center + input[index - 1] + input[index + 1] +
                             input[index - nx] + input[index + nx] + bottom + top;
            output[index] = sum / 7.0;

            bottom = center;
            center = top;
            index += plane;
        }
    }
}

        return false;
    }

    return true;
}
