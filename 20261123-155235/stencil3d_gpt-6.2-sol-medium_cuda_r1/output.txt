// 3D index calculation
inline constexpr size_t idx3(const size_t x, const size_t y, const size_t z, const size_t nx, const size_t ny) noexcept {
    return z * (nx * ny) + y * nx + x;
}

void initializeGrid(std::vector<Real>& grid, const size_t nx, const size_t ny, const size_t nz) {
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);

            }
        }
    }
}

// 7-point stencil computation
void stencilIteration(const std::vector<Real>& input, 
                      std::vector<Real>& output,
                      const size_t nx, const size_t ny, const size_t nz) {
    // Process interior points (not on boundaries)
    for (size_t z = 1; z < nz - 1; ++z) {
        for (size_t y = 1; y < ny - 1; ++y) {
            for (size_t x = 1; x < nx - 1; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                
                const Real center = input[idx];
                const Real left = input[idx3(x-1, y, z, nx, ny)];
                const Real right = input[idx3(x+1, y, z, nx, ny)];
                const Real front = input[idx3(x, y-1, z, nx, ny)];
                const Real back = input[idx3(x, y+1, z, nx, ny)];
                const Real bottom = input[idx3(x, y, z-1, nx, ny)];
                const Real top = input[idx3(x, y, z+1, nx, ny)];
                
                // Simple averaging stencil
                output[idx] = (center + left + right + front + back + bottom + top) / 7.0;
            }
        }
    }
    
    // Copy boundary values
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                if (x == 0 || x == nx-1 || y == 0 || y == ny-1 || z == 0 || z == nz-1) {
                    const size_t idx = idx3(x, y, z, nx, ny);
                    output[idx] = input[idx];
                }
            }
        }
    }
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

__global__ void initializeGrids(Real* first, Real* second, size_t count, bool zero) {
    const size_t stride = static_cast<size_t>(gridDim.x) * blockDim.x;
    for (size_t i = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
         i < count; i += stride) {
        const Real value = zero ? 0.0 : static_cast<Real>(i % 19);
        first[i] = value;
        if (second) {
            second[i] = value;
        }
    }
}

__device__ __forceinline__ Real divideBySeven(Real sum) {
    // Correct the reciprocal product with its exact residual. For binary inputs
    // divided by 7, the quotient is at least 1/14 ulp away from a rounding midpoint.
    constexpr Real reciprocal = 1.0 / 7.0;
    const Real quotient = sum * reciprocal;
    const Real residual = __fma_rn(-7.0, quotient, sum);
    return __fma_rn(reciprocal, residual, quotient);
}

__global__ void stencilIteration(const Real* __restrict__ input, Real* __restrict__ output,
                                 size_t nx, size_t ny, size_t nz) {
    // Adjacent lanes work on adjacent X cells. Each thread reuses its Z neighbors.
    const size_t x = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    if (x == 0 || x >= nx - 1) {
        return;
    }

    const size_t plane = nx * ny;
    const size_t tileStride = static_cast<size_t>(gridDim.z) * planesPerTile;
    const size_t rowStride = static_cast<size_t>(gridDim.y) * blockDim.y;
    for (size_t y = 1 + static_cast<size_t>(blockIdx.y) * blockDim.y + threadIdx.y;
         y < ny - 1; y += rowStride) {
        for (size_t z = 1 + static_cast<size_t>(blockIdx.z) * planesPerTile;
             z < nz - 1; z += tileStride) {
            size_t i = z * plane + y * nx + x;
            Real bottom = input[i - plane];
            Real center = input[i];
            const unsigned planes = min(static_cast<size_t>(planesPerTile), nz - 1 - z);
            #pragma unroll
            for (unsigned offset = 0; offset < planesPerTile; ++offset) {
                if (offset >= planes) {
                    break;
                }
                const Real top = input[i + plane];
                const Real sum = center + input[i - 1] + input[i + 1] + input[i - nx]
                               + input[i + nx] + bottom + top;
                output[i] = divideBySeven(sum);
                bottom = center;
                center = top;
                i += plane;
            }
        }
    }
}

    // Simple sanity checks

            printf("Validation failed: found NaN or Inf value\n");
            return false;
        }
    }
    
    // 2. Values should be reasonable (bounded)
    Real minVal = grid.empty() ? 0.0 : grid.front();
    Real maxVal = minVal;
    for (const auto& val : grid) {
        minVal = std::min(minVal, val);
        maxVal = std::max(maxVal, val);
    }
    
    printf("Value range: [%.6f, %.6f]\n", minVal, maxVal);

    }

}
