cmake_minimum_required(VERSION 3.18)
project(cholesky_benchmark LANGUAGES CXX CUDA)

set(MPI_CXX_SKIP_MPICXX ON)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_package(CUDAToolkit REQUIRED)

set_source_files_properties(cholesky.cpp PROPERTIES LANGUAGE CUDA)
add_executable(cholesky cholesky.cpp)
set_target_properties(cholesky PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
separate_arguments(openmp_flags NATIVE_COMMAND "${OpenMP_CXX_FLAGS}")
if(CMAKE_CUDA_COMPILER_ID STREQUAL "NVIDIA")
    list(JOIN openmp_flags "," openmp_host_flags)
    target_compile_options(cholesky PRIVATE -O3 "-Xcompiler=${openmp_host_flags},-Wall,-Wextra")
else()
    target_compile_options(cholesky PRIVATE -O3 -Wall -Wextra ${openmp_flags})
endif()
target_link_libraries(cholesky PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX
    CUDA::cudart CUDA::cublas CUDA::cusolver)
