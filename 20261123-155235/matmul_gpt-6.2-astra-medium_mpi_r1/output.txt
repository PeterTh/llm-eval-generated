class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

void matrixMultiply(const std::vector<double>& a, const std::vector<double>& b,
                    std::vector<double>& c, size_t n, size_t rows, size_t stride) {
    // B is packed in strips of NR columns. Reuse each strip across local rows
    // while keeping a complete MR x NR output tile in registers.
    for (size_t k0 = 0; k0 < n; k0 += depthBlock) {
        const size_t end = std::min(n, k0 + depthBlock);
        for (size_t j = 0; j < stride; j += NR) {
            for (size_t i = 0; i < rows; i += MR) {
                double sums[MR][NR];
                for (size_t r = 0; r < MR; ++r)
                    std::copy_n(&c[(i + r) * stride + j], NR, sums[r]);
                for (size_t k = k0; k < end; ++k) {
                    const double* strip = &b[j * n + k * NR];
                    for (size_t r = 0; r < MR; ++r) {
                        const double left = a[(i + r) * n + k];
                        for (size_t s = 0; s < NR; ++s)

                }
                for (size_t r = 0; r < MR; ++r)
                    std::copy_n(sums[r], NR, &c[(i + r) * stride + j]);
