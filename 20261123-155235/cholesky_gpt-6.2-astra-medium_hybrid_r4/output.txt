// Simple Cholesky decomposition (sequential, unblocked algorithm)
// Decomposes positive definite matrix A into L * L^T where L is lower triangular

bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
    // A is stored in row-major order
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j <= i; ++j) {
            double sum = 0.0;
            
            if (i == j) {
                // Diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[j * n + k] * A[j * n + k];
                }
                const double val = A[j * n + j] - sum;
                if (val <= 0.0) {
                    // Matrix is not positive definite
                    printf("Error: Matrix is not positive definite at diagonal element %zu\n", j);
                    return false;
                }
                A[j * n + j] = sqrt(val);
            } else {
                // Off-diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[i * n + k] * A[j * n + k];
                }
                A[i * n + j] = (A[i * n + j] - sum) / A[j * n + j];
            }
        }
        
        // Zero out upper triangular part
        for (size_t j = i + 1; j < n; ++j) {
            A[i * n + j] = 0.0;
        }
    }
    
    return true;
}

// Generate a symmetric positive definite matrix
void generatePositiveDefiniteMatrix(std::vector<double>& A, const size_t n) {
    // Method: Create A = B * B^T where B is random
    // This guarantees A is positive semi-definite
    // Then add identity to make it strictly positive definite
    
    std::vector<double> B(n * n);
    unsigned int seed = 42;
    
    // Generate random matrix B
    for (size_t i = 0; i < n * n; ++i) {
        B[i] = (rand_r(&seed) / (double)RAND_MAX) - 0.5;
    }
    
    // Compute A = B * B^T
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < n; ++k) {
                sum += B[i * n + k] * B[j * n + k];
            }
            A[i * n + j] = sum;
        }
    }
    
    // Add diagonal dominance to ensure positive definiteness
    for (size_t i = 0; i < n; ++i) {
        A[i * n + i] += n;
    }
}

class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

    unsigned seed = 42, mul = 1103515245u, add = 12345u;
    for (size_t steps = 3 * offset; steps; steps >>= 1) {
        if (steps & 1) seed = mul * seed + add;
        add *= mul + 1;
        mul *= mul;
    }
    return seed;
}

__global__ void packRows(const double* panel, int ld, double* packed, int cols,
                         int first, int k, int tile, int rank, int ranks) {
    int c = first + blockIdx.x * blockDim.x + threadIdx.x;
    if (c < cols) {
        int row = ((long long)(c/tile)*ranks + rank)*tile + c%tile;
        packed[size_t(blockIdx.y)*cols+c] = panel[size_t(blockIdx.y)*ld+row-k];
    }
}

int factor(Grid g, double* a, double* panel, double* host, double* packed,
           cublasHandle_t blas, cusolverDnHandle_t solver) {
    int block = ranks == 1 ? g.n : g.tile;
    int workSize = 0;
    if (g.cols) CHECK(cusolverDnDpotrf_bufferSize(solver, CUBLAS_FILL_MODE_LOWER,
                                   block, a, g.n, &workSize));
    Device<double> work(workSize);
    Device<int> status(1);
    int first = 0;
    for (int k = 0; k < g.n;) {
        int width = std::min(block, g.n-k), ld = g.n-k, owner = g.owner(k);
        int below = ld-width;
        const double* left = panel;
        if (rank == owner) {
            double* diagonal = a + size_t(first)*g.n+k;
            CHECK(cusolverDnDpotrf(solver, CUBLAS_FILL_MODE_LOWER, width, diagonal,
                                   g.n, work.p, workSize, status.p));
            int info;
            CHECK(cudaMemcpy(&info, status.p, sizeof(info), cudaMemcpyDeviceToHost));
            if (info < 0) check(info, \"potrf argument\");
            if (ranks == 1) return info ? k+info : 0;
            if ("'!info && below) {
                CHECK(cublasDtrsm(blas, CUBLAS_SIDE_RIGHT, CUBLAS_FILL_MODE_LOWER,
                           CUBLAS_OP_T, CUBLAS_DIAG_NON_UNIT, below, width, &one, diagonal,
                           g.n, diagonal+width, g.n));
                CHECK(cudaMemcpy2D(host, ld*sizeof(double), diagonal, g.n*sizeof(double),
                                   ld*sizeof(double), width, cudaMemcpyDeviceToHost));
            }
            // The first pivot is positive. Use its broadcast slot to signal failure.
            if (info) host[0] = -(k+info);
            else if (!below) host[0] = 1;
            first += width;
            left = diagonal;
        }
        broadcast(host, below ? size_t(ld)*width : 1, owner);
        if (host[0] < 0) return int(-host[0]);
        if (first < g.cols) {
            if (rank != owner)
                CHECK(cudaMemcpy(panel, host, size_t(ld)*width*sizeof(double), cudaMemcpyHostToDevice));
            int pitch = rank == owner ? g.n : ld;
            packRows<<<dim3((g.cols-first+255)/256,width),256>>>(left, pitch, packed,
                                       g.cols, first, k, g.tile, rank, ranks);
            CHECK(cudaGetLastError());
            // Group adjacent local tiles for larger GEMMs, skipping rows above the group.
            for (int c = first; c < g.cols;) {
                int nc = std::min(4*g.tile, g.cols-c), row = g.global(c);
                CHECK(cublasDgemm(blas, CUBLAS_OP_N, CUBLAS_OP_T, g.n-row, nc, width,
                      &minusOne, left+row-k, pitch, packed+c, g.cols, &one,
                      a+size_t(c)*g.n+row, g.n));
                c += nc;
            }
        }
        k += width;
    }
    return 0;
}

              double* host, double* panel, double* packed) {

        int width = ranks == 1 ? g.n : std::min(g.tile, g.n-k);
        const double* left = a;
        const double* right = a;
        int rightLd = g.n;
        if (ranks > 1) {
            size_t bytes = size_t(width)*g.n*sizeof(double);
            if (rank == g.owner(k)) {
                left = a+size_t(g.local(k))*g.n;
                CHECK(cudaMemcpy(host, left, bytes, cudaMemcpyDeviceToHost));
            }
            broadcast(host, bytes/sizeof(double), g.owner(k));
            if (rank != g.owner(k)) {
                CHECK(cudaMemcpy(panel, host, bytes, cudaMemcpyHostToDevice));
                left = panel;
            }
            if (g.cols) {
                packRows<<<dim3((g.cols+255)/256,width),256>>>(left, g.n, packed,
                                             g.cols, 0, 0, g.tile, rank, ranks);
                CHECK(cudaGetLastError());
            }
            right = packed;
            rightLd = g.cols;
        }
        if (g.cols) CHECK(cublasDgemm(blas, CUBLAS_OP_N, CUBLAS_OP_T, g.n, g.cols, width,
                 &one, left, g.n, right, rightLd, k ? &one : &zero, recon.p, g.n));
        k += width;
    }
    std::vector<double> values(count);
    if (count) CHECK(cudaMemcpy(values.data(), recon.p, count*sizeof(double), cudaMemcpyDeviceToHost));
    double absolute = 0, relative = 0;
    #pragma omp parallel for reduction(max:absolute,relative)
    for (size_t i = 0; i < count; ++i) {
        double err = std::fabs(values[i]-original[i]);
        if (!std::isfinite(err)) err = std::numeric_limits<double>::infinity();
        absolute = std::max(absolute, err);
        relative = std::max(relative, err/(std::fabs(original[i])+1e-10));
    }
    double errors[] = {absolute, relative};
    CHECK(MPI_Allreduce(MPI_IN_PLACE, errors, 2, MPI_DOUBLE, MPI_MAX, MPI_COMM_WORLD));
    if (!rank) std::printf("Max absolute error: %.10e'"\\nMax relative error: %.10e\\n\", errors[0], errors[1]);
    return errors[1] <= 1e-6;
}

void broadcast(double* p, size_t count, int owner) {
    while (count) {
        int chunk = int(std::min<size_t>(count, INT_MAX));
        CHECK(MPI_Bcast(p, chunk, MPI_DOUBLE, owner, MPI_COMM_WORLD));
        p += chunk;
        count -= chunk;
    }
}

// glibc rand_r advances its 32-bit LCG three times per output. Jump to a row
// boundary, then use the original library function for every sample.
#ifdef __GLIBC__
unsigned rowSeed(size_t offset) {
    unsigned seed = 42, mul = 1103515245u, add = 12345u;
    for (size_t steps = 3 * offset; steps; steps >>= 1) {
        if (steps & 1) seed = mul * seed + add;
        add *= mul + 1;
        mul *= mul;
    }
    return seed;
}
#endif


void generate(Grid g, double* a, double* panel, double* host, cublasHandle_t blas) {
    size_t count = size_t(g.n) * g.cols;
    Device<double> b(count);
    std::vector<double> values(count);
#ifndef __GLIBC__
    std::vector<unsigned> seeds(g.n);
    unsigned state = 42;
    for (int i = 0; i < g.n; ++i) {
        seeds[i] = state;
        for (int j = 0; j < g.n; ++j) rand_r(&state);
    }
#endif
    #pragma omp parallel for schedule(static)
    for (int c = 0; c < g.cols; ++c) {
#ifdef __GLIBC__
        unsigned seed = rowSeed(size_t(g.global(c)) * g.n);
#else
        unsigned seed = seeds[g.global(c)];
#endif
        for (int i = 0; i < g.n; ++i)
            values[size_t(c) * g.n + i] = rand_r(&seed) / double(RAND_MAX) - 0.5;
    }
    if (count) CHECK(cudaMemcpy(b.p, values.data(), count*sizeof(double), cudaMemcpyHostToDevice));
    // B rows are contiguous vectors, interpreted by BLAS as columns of B^T.
    for (int k = 0; k < g.n;) {
        int width = ranks == 1 ? g.n : std::min(g.tile, g.n-k);
        size_t length = size_t(g.n)*width;
        const double* left = b.p;
        if (ranks > 1) {
            if (rank == g.owner(k))
                std::copy_n(values.data() + size_t(g.local(k))*g.n, length, host);
            broadcast(host, length, g.owner(k));
            if (rank == g.owner(k)) left = b.p + size_t(g.local(k))*g.n;
            else {
                CHECK(cudaMemcpy(panel, host, length*sizeof(double), cudaMemcpyHostToDevice));
                left = panel;
            }
        }
        if (g.cols) CHECK(cublasDgemm(blas, CUBLAS_OP_T, CUBLAS_OP_N, width, g.cols, g.n,
                    &one, left, g.n, b.p, g.n, &zero, a+k, g.n));
        k += width;
    }
}

__global__ void shiftDiagonal(double* a, int n, int cols, int tile, int rank, int ranks) {
    int c = blockIdx.x * blockDim.x + threadIdx.x;
    if (c < cols) {
        int i = ((long long)(c/tile)*ranks + rank)*tile + c%tile;
        a[size_t(c)*n+i] += n;
    }
}

// Pack the rows of a broadcast panel that correspond to this rank's columns.
__global__ void packRows(const double* panel, int ld, double* packed, int cols,
                         int first, int k, int tile, int rank, int ranks) {
    int c = first + blockIdx.x * blockDim.x + threadIdx.x;
    if (c < cols) {
        int row = ((long long)(c/tile)*ranks + rank)*tile + c%tile;
        packed[size_t(blockIdx.y)*cols+c] = panel[size_t(blockIdx.y)*ld+row-k];
    }
}

int factor(Grid g, double* a, double* panel, double* host, double* packed,
           cublasHandle_t blas, cusolverDnHandle_t solver) {
    int block = ranks == 1 ? g.n : g.tile;
    int workSize = 0;
    if (g.cols) CHECK(cusolverDnDpotrf_bufferSize(solver, CUBLAS_FILL_MODE_LOWER,
                                   block, a, g.n, &workSize));
    Device<double> work(workSize);
    Device<int> status(1);
    int first = 0;
    for (int k = 0; k < g.n;) {
        int width = std::min(block, g.n-k), ld = g.n-k, owner = g.owner(k);
        int below = ld-width;
        const double* left = panel;
        if (rank == owner) {
            double* diagonal = a + size_t(first)*g.n+k;
            CHECK(cusolverDnDpotrf(solver, CUBLAS_FILL_MODE_LOWER, width, diagonal,
                                   g.n, work.p, workSize, status.p));
            int info;
            CHECK(cudaMemcpy(&info, status.p, sizeof(info), cudaMemcpyDeviceToHost));
            if (info < 0) check(info, "potrf argument");
            if (ranks == 1) return info ? k+info : 0;
            if (!info && below) {
                CHECK(cublasDtrsm(blas, CUBLAS_SIDE_RIGHT, CUBLAS_FILL_MODE_LOWER,
                           CUBLAS_OP_T, CUBLAS_DIAG_NON_UNIT, below, width, &one, diagonal,
                           g.n, diagonal+width, g.n));
                CHECK(cudaMemcpy2D(host, ld*sizeof(double), diagonal, g.n*sizeof(double),
                                   ld*sizeof(double), width, cudaMemcpyDeviceToHost));
            }
            // The first pivot is positive. Use its broadcast slot to signal failure.
            if (info) host[0] = -(k+info);
            else if (!below) host[0] = 1;
            first += width;
            left = diagonal;
        }
        broadcast(host, below ? size_t(ld)*width : 1, owner);
        if (host[0] < 0) return int(-host[0]);
        if (first < g.cols) {
            if (rank != owner)
                CHECK(cudaMemcpy(panel, host, size_t(ld)*width*sizeof(double), cudaMemcpyHostToDevice));
            int pitch = rank == owner ? g.n : ld;
            packRows<<<dim3((g.cols-first+255)/256,width),256>>>(left, pitch, packed,
                                       g.cols, first, k, g.tile, rank, ranks);
            CHECK(cudaGetLastError());
            // Group adjacent local tiles for larger GEMMs, skipping rows above the group.
            for (int c = first; c < g.cols;) {
                int nc = std::min(4*g.tile, g.cols-c), row = g.global(c);
                CHECK(cublasDgemm(blas, CUBLAS_OP_N, CUBLAS_OP_T, g.n-row, nc, width,
                      &minusOne, left+row-k, pitch, packed+c, g.cols, &one,
                      a+size_t(c)*g.n+row, g.n));
                c += nc;
            }
        }
        k += width;
    }
    return 0;
}

__global__ void clearUpper(double* a, int n, int cols, int tile, int rank, int ranks) {
    int c = blockIdx.x;

    for (int i = threadIdx.x; i < global; i += blockDim.x) a[size_t(c)*n+i] = 0;
}
