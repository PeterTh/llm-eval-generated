// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// Simple Cholesky decomposition (sequential, unblocked algorithm)
// Decomposes positive definite matrix A into L * L^T where L is lower triangular

bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
    // A is stored in row-major order
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j <= i; ++j) {
            double sum = 0.0;
            
            if (i == j) {
                // Diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[j * n + k] * A[j * n + k];
                }
                const double val = A[j * n + j] - sum;
                if (val <= 0.0) {
                    // Matrix is not positive definite
                    printf("Error: Matrix is not positive definite at diagonal element %zu\n", j);
                    return false;
                }
                A[j * n + j] = sqrt(val);
            } else {
                // Off-diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[i * n + k] * A[j * n + k];
                }
                A[i * n + j] = (A[i * n + j] - sum) / A[j * n + j];
            }
        }
        
        // Zero out upper triangular part
        for (size_t j = i + 1; j < n; ++j) {
            A[i * n + j] = 0.0;
        }
    }
    
    return true;
}

// Generate a symmetric positive definite matrix
void generatePositiveDefiniteMatrix(std::vector<double>& A, const size_t n) {
    // Method: Create A = B * B^T where B is random
    // This guarantees A is positive semi-definite
    // Then add identity to make it strictly positive definite
    
    std::vector<double> B(n * n);
    unsigned int seed = 42;
    
    // Generate random matrix B
    for (size_t i = 0; i < n * n; ++i) {
        B[i] = (rand_r(&seed) / (double)RAND_MAX) - 0.5;
    }
    
    // Compute A = B * B^T
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < n; ++k) {
                sum += B[i * n + k] * B[j * n + k];
            }
            A[i * n + j] = sum;
        }
    }
    
    // Add diagonal dominance to ensure positive definiteness
    for (size_t i = 0; i < n; ++i) {
        A[i * n + i] += n;
    }
}

bool validateCholesky(const std::vector<double>& L, const std::vector<double>& A_orig, const size_t n) {
    // Validate by computing L * L^T and comparing with original matrix
    
    std::vector<double> reconstructed(n * n);
    
    // Compute L * L^T
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < n; ++k) {
                sum += L[i * n + k] * L[j * n + k];
            }
            reconstructed[i * n + j] = sum;
        }
    }
    
    // Compare with original
    double maxError = 0.0;
    double relError = 0.0;
    
    for (size_t i = 0; i < n * n; ++i) {
        const double error = fabs(reconstructed[i] - A_orig[i]);
        maxError = std::max(maxError, error);
        
        const double rel = error / (fabs(A_orig[i]) + 1e-10);
        relError = std::max(relError, rel);
    }
    
    printf("Max absolute error: %.10e\n", maxError);
    printf("Max relative error: %.10e\n", relError);
    
    // Check if error is within tolerance
    if (relError > 1e-6) {
        printf("Validation failed: relative error too large\n");
        return false;
    }
    
    return true;
}

__global__ void reconstructionError(const double* reconstructed, const double* original,
                                    int n, ErrorMaxima* maxima) {
    const size_t col = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
    const size_t row = blockIdx.y;
    double absolute = 0.0;
    double relative = 0.0;

        const size_t offset = row * n + col;
        absolute = fabs(reconstructed[offset] - original[offset]);
        relative = absolute / (fabs(original[offset]) + 1e-10);
    }

    // Ignore NaNs, matching the original maximum initialized to zero.
    absolute = fmax(absolute, 0.0);
    relative = fmax(relative, 0.0);
    for (int stride = warpSize / 2; stride > 0; stride /= 2) {
        absolute = fmax(absolute, __shfl_down_sync(0xffffffff, absolute, stride));
        relative = fmax(relative, __shfl_down_sync(0xffffffff, relative, stride));
    }

    __shared__ double warpAbsolute[blockSize / 32];
    __shared__ double warpRelative[blockSize / 32];
    if (threadIdx.x % warpSize == 0) {
        warpAbsolute[threadIdx.x / warpSize] = absolute;
        warpRelative[threadIdx.x / warpSize] = relative;
    }
    __syncthreads();

    if (threadIdx.x < warpSize) {
        absolute = threadIdx.x < blockSize / warpSize ? warpAbsolute[threadIdx.x] : 0.0;
        relative = threadIdx.x < blockSize / warpSize ? warpRelative[threadIdx.x] : 0.0;

            relative = fmax(relative, __shfl_down_sync(0xffffffff, relative, stride));

            atomicMax(&maxima->absolute, static_cast<unsigned long long>(__double_as_longlong(absolute)));
            atomicMax(&maxima->relative, static_cast<unsigned long long>(__double_as_longlong(relative)));

    73	__global__ void symmetricProduct(const double* input, double* output, int n, int k, int pitch) {
    74	    if (blockIdx.x > blockIdx.y) {
    75	        return;
    76	    }
    77	    const int x = threadIdx.x;
    78	    const int y = threadIdx.y;
    79	    const int row = blockIdx.y * matrixTile + y;
    80	    const int col = blockIdx.x * matrixTile + x;
    81	    const int tileCol = blockIdx.x * matrixTile + y;
    82	    __shared__ double left[matrixTile][matrixTile + 1];
    83	    __shared__ double right[matrixTile][matrixTile + 1];
    84	    double sum = 0.0;
    85	
    86	    for (int first = 0; first < k; first += matrixTile) {
    87	        const int inner = first + x;
    88	        left[y][x] = row < n && inner < k ? input[static_cast<size_t>(row) * pitch + inner] : 0.0;
    89	        right[y][x] = tileCol < n && inner < k ? input[static_cast<size_t>(tileCol) * pitch + inner] : 0.0;
    90	        __syncthreads();
    91	        #pragma unroll
    92	        for (int i = 0; i < matrixTile; ++i) {
    93	            sum = fma(left[y][i], right[x][i], sum);
    94	        }
    95	        __syncthreads();
    96	    }
    97	
    98	    if (row < n && col <= row) {
    99	        const size_t index = static_cast<size_t>(row) * pitch + col;
   100	        if constexpr (subtract) {
   101	            output[index] -= sum;
   102	        } else {
   103	            output[index] = sum;
   104	        }
   105	    }
   106	}

   109	void updateWithProduct(cublasHandle_t blas, const double* input, double* output,
   110	                       int n, int k, int pitch) {
   111	    if (n <= smallMatrix) {
   112	        const int tiles = (n + matrixTile - 1) / matrixTile;
   113	        symmetricProduct<subtract><<<dim3(tiles, tiles), dim3(matrixTile, matrixTile)>>>(
   114	            input, output, n, k, pitch);
   115	        check(cudaGetLastError(), "Launching the tiled symmetric product");
   116	        return;
   117	    }
   118	
   119	    // The column-major input is the transpose of the row-major input. Its
   120	    // upper-triangle output occupies the row-major lower triangle.
   121	    const double alpha = subtract ? -1.0 : 1.0;
   122	    const double beta = subtract ? 1.0 : 0.0;
   123	    check(cublasDsyrk(blas, CUBLAS_FILL_MODE_UPPER, CUBLAS_OP_T, n, k,
   124	                      &alpha, input, pitch, &beta, output, pitch),
   125	          "Computing the GPU symmetric product");
   126	}

   128	__global__ void factorDiagonal(double* matrix, int n, int first, int width, int* info) {
   129	    if (*info != 0) {
   130	        return;
   131	    }
   132	    __shared__ double tile[panelWidth][panelWidth + 1];
   133	    __shared__ double inverse;
   134	    __shared__ bool validPivot;
   135	    const int col = threadIdx.x;
   136	
   137	    for (int row = threadIdx.y; row < width; row += diagonalRowsPerBlock) {
   138	        if (col <= row) {
   139	            tile[row][col] = matrix[static_cast<size_t>(first + row) * n + first + col];
   140	        }
   141	    }
   142	    __syncthreads();
   143	
   144	    for (int pivot = 0; pivot < width; ++pivot) {
   145	        if (threadIdx.y == 0 && col == 0) {
   146	            const double value = tile[pivot][pivot];
   147	            validPivot = !(value <= 0.0);
   148	            if (validPivot) {
   149	                inverse = rsqrt(value);
   150	                tile[pivot][pivot] = sqrt(value);
   151	            } else {
   152	                *info = first + pivot + 1;
   153	            }
   154	        }
   155	        __syncthreads();
   156	        if (!validPivot) {
   157	            return;
   158	        }
   159	
   160	        if (col == pivot) {
   161	            for (int row = threadIdx.y; row < width; row += diagonalRowsPerBlock) {
   162	                if (row > pivot) {
   163	                    tile[row][pivot] *= inverse;
   164	                }
   165	            }
   166	        }
   167	        __syncthreads();
   168	
   169	        if (col > pivot) {
   170	            for (int row = threadIdx.y; row < width; row += diagonalRowsPerBlock) {
   171	                if (col <= row) {
   172	                    tile[row][col] = fma(-tile[row][pivot], tile[col][pivot], tile[row][col]);
   173	                }
   174	            }
   175	        }
   176	        __syncthreads();
   177	    }
   178	
   179	    for (int row = threadIdx.y; row < width; row += diagonalRowsPerBlock) {
   180	        if (col <= row) {
   181	            matrix[static_cast<size_t>(first + row) * n + first + col] = tile[row][col];
   182	        }
   183	    }
   184	}

   186	__global__ void solvePanel(double* matrix, int n, int first, const int* info) {
   187	    if (*info != 0) {
   188	        return;
   189	    }
   190	    __shared__ double diagonal[panelWidth][panelWidth + 1];
   191	    __shared__ double inverse[panelWidth];
   192	    const int col = threadIdx.x;
   193	    for (int row = threadIdx.y; row < panelWidth; row += solveRowsPerBlock) {
   194	        if (col <= row) {
   195	            diagonal[col][row] = matrix[static_cast<size_t>(first + row) * n + first + col];
   196	        }
   197	    }
   198	    if (threadIdx.y == 0) {
   199	        inverse[col] = 1.0 / matrix[static_cast<size_t>(first + col) * n + first + col];
   200	    }
   201	    __syncthreads();
   202	
   203	    const int row = first + panelWidth + blockIdx.x * solveRowsPerBlock + threadIdx.y;
   204	    if (row < n) {
   205	        const size_t index = static_cast<size_t>(row) * n + first + col;
   206	        double value = matrix[index];
   207	        // Each warp solves one row, broadcasting each completed entry to update the rest.
   208	        for (int pivot = 0; pivot < panelWidth; ++pivot) {
   209	            if (col == pivot) {
   210	                value *= inverse[pivot];
   211	            }
   212	            const double solved = __shfl_sync(0xffffffff, value, pivot);
   213	            if (col > pivot) {
   214	                value = fma(-solved, diagonal[pivot][col], value);
   215	            }
   216	        }
   217	        matrix[index] = value;
   218	    }
   219	}

   221	void factorCholesky(cublasHandle_t blas, double* matrix, int n, int* info) {
   222	    for (int first = 0; first < n; first += panelWidth) {
   223	        const int width = std::min(panelWidth, n - first);
   224	        factorDiagonal<<<1, dim3(panelWidth, diagonalRowsPerBlock)>>>(matrix, n, first, width, info);
   225	        check(cudaGetLastError(), "Launching the diagonal factorization");
   226	
   227	        const int next = first + width;
   228	        const int remaining = n - next;
   229	        if (remaining > 0) {
   230	            const int blocks = (remaining + solveRowsPerBlock - 1) / solveRowsPerBlock;
   231	            solvePanel<<<blocks, dim3(panelWidth, solveRowsPerBlock)>>>(matrix, n, first, info);
   232	            check(cudaGetLastError(), "Launching the panel solve");
   233	
   234	            const double* panel = matrix + static_cast<size_t>(next) * n + first;
   235	            double* trailing = matrix + static_cast<size_t>(next) * n + next;
   236	            updateWithProduct<true>(blas, panel, trailing, remaining, width, n);
   237	        }
   238	    }

   254	__global__ void reconstructionError(const double* reconstructed, const double* original,
   255	                                    int n, ErrorMaxima* maxima) {
   256	    const size_t col = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x;
   257	    const size_t row = blockIdx.y;
   258	    if (static_cast<size_t>(blockIdx.x) * blockDim.x > row) {
   259	        return;
   260	    }
   261	    double absolute = 0.0;
   262	    double relative = 0.0;
   263	    if (col <= row) {
   264	        const size_t offset = row * n + col;
   265	        absolute = fabs(reconstructed[offset] - original[offset]);
   266	        relative = absolute / (fabs(original[offset]) + 1e-10);
   267	    }
   268	
   269	    // Ignore NaNs, matching the original maximum initialized to zero.
   270	    absolute = fmax(absolute, 0.0);
   271	    relative = fmax(relative, 0.0);
   272	    for (int stride = warpSize / 2; stride > 0; stride /= 2) {
   273	        absolute = fmax(absolute, __shfl_down_sync(0xffffffff, absolute, stride));
   274	        relative = fmax(relative, __shfl_down_sync(0xffffffff, relative, stride));
   275	    }
   276	
   277	    __shared__ double warpAbsolute[blockSize / 32];
   278	    __shared__ double warpRelative[blockSize / 32];
   279	    if (threadIdx.x % warpSize == 0) {
   280	        warpAbsolute[threadIdx.x / warpSize] = absolute;
   281	        warpRelative[threadIdx.x / warpSize] = relative;
   282	    }
   283	    __syncthreads();
   284	
   285	    if (threadIdx.x < warpSize) {
   286	        absolute = threadIdx.x < blockSize / warpSize ? warpAbsolute[threadIdx.x] : 0.0;
   287	        relative = threadIdx.x < blockSize / warpSize ? warpRelative[threadIdx.x] : 0.0;
   288	        for (int stride = warpSize / 2; stride > 0; stride /= 2) {
   289	            absolute = fmax(absolute, __shfl_down_sync(0xffffffff, absolute, stride));
   290	            relative = fmax(relative, __shfl_down_sync(0xffffffff, relative, stride));
   291	        }
   292	        if (threadIdx.x == 0) {
   293	            // Unsigned bit ordering agrees with floating-point ordering for nonnegative doubles.
   294	            atomicMax(&maxima->absolute, static_cast<unsigned long long>(__double_as_longlong(absolute)));
   295	            atomicMax(&maxima->relative, static_cast<unsigned long long>(__double_as_longlong(relative)));
   296	        }
   297	    }
   298	}

   322	        printf("Validation failed: relative error too large\n");
