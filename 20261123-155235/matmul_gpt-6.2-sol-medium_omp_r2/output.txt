// Generate pseudo-random values for matrix initialization
constexpr double getPseudoRndValue(const size_t N, const size_t i, const size_t j) noexcept {
    return (((i + 1) * (i + j + 1) * 1299709) % (N * N)) / static_cast<double>(N * N);
}

void initMatrix(std::vector<double>& mat, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            mat[i * N + j] = getPseudoRndValue(N, i, j);
        }
    }
}

void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < N; ++k) {
                sum += A[i * N + k] * B[k * N + j];
            }
            C[i * N + j] = sum;
        }
    }
}

// FNV-1a hash, 64 bit length
class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// Generate pseudo-random values for matrix initialization
constexpr double getPseudoRndValue(const size_t N, const size_t i, const size_t j) noexcept {
    return (((i + 1) * (i + j + 1) * 1299709) % (N * N)) / static_cast<double>(N * N);
}

void initMatrices(double* A, double* B, const size_t N, const int threads) {
    // B stores one contiguous N-by-panelWidth panel for each group of columns.
    #pragma omp parallel for schedule(static) num_threads(threads)
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; j += panelWidth) {
            double* packed = B + j * N + i * panelWidth;
            const size_t width = std::min(panelWidth, N - j);
            for (size_t lane = 0; lane < width; ++lane) {
                const double value = getPseudoRndValue(N, i, j + lane);
                A[i * N + j + lane] = value;
                packed[lane] = value;
            }
            std::fill(packed + width, packed + panelWidth, 0.0);
        }
    }
}

// GCC's original x86 dot product uses separate products and sums for
// pairs of k values, with a scalar fused multiply-add for an odd last value.
#if defined(__GNUC__) && !defined(__clang__) && defined(__SSE2__)
#pragma GCC push_options
#pragma GCC optimize ("fp-contract=off")
constexpr bool fuseOddTail =
#ifdef __FMA__
    true;
#else
    false;
#endif
#else
constexpr bool fuseOddTail = false;
#endif

template <size_t Rows>
void multiplyPanel(const double* A, const double* B, double* C, const size_t N,
                   const size_t width, const size_t begin, const size_t end) {
    double sums[Rows][panelWidth] = {};
    if (begin != 0) {
        for (size_t row = 0; row < Rows; ++row) {
            std::copy_n(C + row * N, width, sums[row]);
        }
    }

    const bool oddTail = fuseOddTail && end == N && N % 2 != 0;
    const size_t bulkEnd = end - oddTail;
    for (size_t k = begin; k < bulkEnd; ++k) {
        const double* right = B + k * panelWidth;
        for (size_t row = 0; row < Rows; ++row) {
            const double left = A[row * N + k];
            #pragma omp simd
            for (size_t lane = 0; lane < panelWidth; ++lane) {
                sums[row][lane] += left * right[lane];
            }
        }
    }

    if (oddTail) {
        const double* right = B + bulkEnd * panelWidth;
        for (size_t row = 0; row < Rows; ++row) {
            const double left = A[row * N + bulkEnd];
            #pragma omp simd
            for (size_t lane = 0; lane < panelWidth; ++lane) {
                sums[row][lane] = std::fma(left, right[lane], sums[row][lane]);
            }
        }
    }
    for (size_t row = 0; row < Rows; ++row) {
        std::copy_n(sums[row], width, C + row * N);
    }
}

#if defined(__GNUC__) && !defined(__clang__) && defined(__SSE2__)
#pragma GCC pop_options
#endif

void matrixMultiply(const double* A, const double* B, double* C, const size_t N,
                    const int threads) {
    #pragma omp parallel for collapse(2) schedule(static) num_threads(threads)
    for (size_t ii = 0; ii < N; ii += tileRows) {
        for (size_t jj = 0; jj < N; jj += tileCols) {
            const size_t iEnd = std::min(ii + tileRows, N);
            const size_t jEnd = std::min(jj + tileCols, N);
            for (size_t kk = 0; kk < N; kk += depthBlock) {
                const size_t kEnd = std::min(kk + depthBlock, N);
                for (size_t j = jj; j < jEnd; j += panelWidth) {
                    const size_t width = std::min(panelWidth, jEnd - j);
                    const double* panel = B + j * N;
                    size_t i = ii;
                    for (; i + kernelRows <= iEnd; i += kernelRows) {
                        multiplyPanel<kernelRows>(A + i * N, panel, C + i * N + j,
                                                  N, width, kk, kEnd);
                    }
                    for (; i < iEnd; ++i) {
                        multiplyPanel<1>(A + i * N, panel, C + i * N + j,
                                         N, width, kk, kEnd);
                    }
                }
            }
        }
    }
}

bool validateResult(const double* A, const double* B, const std::vector<double>& C,
                    const size_t N) {
    if (N == 0) {
        return true;
    }
    // Check a few random positions
    constexpr size_t checkPoints[] = {0, 1, 2, 3, 4};
    
    for (size_t pi = 0; pi < 5; ++pi) {
        for (size_t pj = 0; pj < 5; ++pj) {
            const size_t i = checkPoints[pi] % N;
            const size_t j = checkPoints[pj] % N;
            
            const double* column = B + (j / panelWidth * panelWidth) * N + j % panelWidth;
            double expected = 0.0;
            for (size_t k = 0; k < N; ++k) {

            }

            const double relError = std::abs((actual - expected) / (expected + 1e-10));

            }
        }
    }

}
