// Simple Cholesky decomposition (sequential, unblocked algorithm)
// Decomposes positive definite matrix A into L * L^T where L is lower triangular

bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
    // A is stored in row-major order
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j <= i; ++j) {
            double sum = 0.0;
            
            if (i == j) {
                // Diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[j * n + k] * A[j * n + k];
                }
                const double val = A[j * n + j] - sum;
                if (val <= 0.0) {
                    // Matrix is not positive definite
                    printf("Error: Matrix is not positive definite at diagonal element %zu\n", j);
                    return false;
                }
                A[j * n + j] = sqrt(val);
            } else {
                // Off-diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[i * n + k] * A[j * n + k];
                }
                A[i * n + j] = (A[i * n + j] - sum) / A[j * n + j];
            }
        }
        
        // Zero out upper triangular part
        for (size_t j = i + 1; j < n; ++j) {
            A[i * n + j] = 0.0;
        }
    }
    
    return true;
}

// Generate a symmetric positive definite matrix
void generatePositiveDefiniteMatrix(std::vector<double>& A, const size_t n) {
    // Method: Create A = B * B^T where B is random
    // This guarantees A is positive semi-definite
    // Then add identity to make it strictly positive definite
    
    std::vector<double> B(n * n);
    unsigned int seed = 42;
    
    // Generate random matrix B
    for (size_t i = 0; i < n * n; ++i) {
        B[i] = (rand_r(&seed) / (double)RAND_MAX) - 0.5;
    }
    
    // Compute A = B * B^T
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < n; ++k) {
                sum += B[i * n + k] * B[j * n + k];
            }
            A[i * n + j] = sum;
        }
    }
    
    // Add diagonal dominance to ensure positive definiteness
    for (size_t i = 0; i < n; ++i) {
        A[i * n + i] += n;
    }
}

    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < n; ++k) {
                sum += L[i * n + k] * L[j * n + k];
            }
            reconstructed[i * n + j] = sum;
        }
    }
    
    // Compare with original
    double maxError = 0.0;

class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

struct Grid {

    int block;
    int tiles;
    int rows;
    int cols;

    Grid(int size, int worldRank, int worldSize) : n(size), rank(worldRank) {

        const int shortSide = static_cast<int>(std::sqrt(worldSize));

// Store just the local lower-triangular tiles, in column-major order within a tile.
struct Matrix {
    const Grid& grid;
    std::vector<size_t> offsets;
    std::vector<double> values;

    explicit Matrix(const Grid& g) : grid(g) {
        size_t count = 0;
        for (int j = g.col; j < g.tiles; j += g.cols) {
            offsets.push_back(count);
            count += static_cast<size_t>(ceilDiv(g.tiles - firstRow(j), g.rows)) *
                     g.tileSize();
        }
        values.resize(count);
    }

    int firstRow(int j) const { return firstOwned(j, grid.row, grid.rows); }

    size_t offset(int i, int j) const {
        return offsets[j / grid.cols] +
               static_cast<size_t>((i - firstRow(j)) / grid.rows) * grid.tileSize();
    }

    double* tile(int i, int j) { return values.data() + offset(i, j); }
    const double* tile(int i, int j) const { return values.data() + offset(i, j); }
};

double dot(const double* a, const double* b, int length, double sum = 0.0) {
    for (int k = 0; k < length; ++k) {
        sum += a[k] * b[k];
    }
    return sum;
}

std::vector<unsigned int> matrixSeeds(const Grid& g) {
    std::vector<unsigned int> seeds(g.tiles);
#ifdef __GLIBC__
    // glibc'"'s rand_r advances this 32-bit LCG three times per output.
    // Exponentiating the affine transform lets each owner start at its own tile.
    for (int i = 0; i < g.tiles; ++i) {
        auto steps = 3 * static_cast<uint64_t>(i) * g.block * g.n;
        uint32_t seed = 42;
        uint32_t mul = 1103515245u;
        uint32_t add = 12345u;
        for (; steps "'!= 0; steps >>= 1) {
            if (steps & 1) {
                seed = mul * seed + add;
            }
            add *= mul + 1;
            mul *= mul;
        }
        seeds[i] = seed;
    }
#else
    // Other C libraries may implement rand_r differently.
    if (g.rank == 0) {
        unsigned int seed = 42;
        for (int i = 0; i < g.tiles; ++i) {
            seeds[i] = seed;
            if (i + 1 < g.tiles) {
                for (size_t k = 0; k < static_cast<size_t>(g.block) * g.n; ++k) {
                    rand_r(&seed);
                }
            }
        }
    }
    MPI_Bcast(seeds.data(), g.tiles, MPI_UNSIGNED, 0, g.comm);
#endif
    return seeds;
}

// A rank needs the input rows for its tile rows and its tile columns.
struct RowStrips {
    const Grid& grid;
    size_t stride;
    std::vector<double> rows;
    std::vector<double> cols;

    explicit RowStrips(const Grid& g)
        : grid(g), stride(static_cast<size_t>(g.block) * g.n),
          rows(ceilDiv(g.tiles - g.row, g.rows) * stride),
          cols(ceilDiv(g.tiles - g.col, g.cols) * stride) {}

    double* row(int i) { return rows.data() + i / grid.rows * stride; }
    double* col(int j) { return cols.data() + j / grid.cols * stride; }
};

void generateMatrix(Matrix& a) {
    const Grid& g = a.grid;
    RowStrips b(g);
    auto seeds = matrixSeeds(g);
    for (int i = 0; i < g.tiles; ++i) {
        const int rowOwner = i % g.rows;
        const int colOwner = i % g.cols;
        const int count = g.extent(i) * g.n;
        if (g.row == rowOwner && g.col == colOwner) {
            for (int k = 0; k < count; ++k) {
                b.row(i)[k] = rand_r(&seeds[i]) / static_cast<double>(RAND_MAX) - 0.5;
            }
            std::copy_n(b.row(i), count, b.col(i));
        }
        if (g.row == rowOwner) {
            MPI_Bcast(b.row(i), count, MPI_DOUBLE, colOwner, g.across);
        }
        if (g.col == colOwner) {
            MPI_Bcast(b.col(i), count, MPI_DOUBLE, rowOwner, g.down);
        }
    }

    for (int j = g.col; j < g.tiles; j += g.cols) {
        for (int i = a.firstRow(j); i < g.tiles; i += g.rows) {
            double* tile = a.tile(i, j);
            for (int c = 0; c < g.extent(j); ++c) {
                for (int r = (i == j ? c : 0); r < g.extent(i); ++r) {
                    tile[c * g.block + r] =
                        dot(b.row(i) + static_cast<size_t>(r) * g.n,
                            b.col(j) + static_cast<size_t>(c) * g.n, g.n);
                }
                if (i == j) {
                    tile[c * g.block + c] += g.n;
                }
            }
        }
    }
}

// The prefix contains dot products from completed tiles. Finish each entry
// using the preceding elements within the current diagonal tile.
int factorDiagonal(double* tile, const double* prefix, double* diag, int b, int width) {
    for (int r = 0; r < width; ++r) {
        for (int c = 0; c < r; ++c) {
            const double sum = dot(diag + r * b, diag + c * b, c, prefix[c * b + r]);
            diag[r * b + c] = tile[c * b + r] = (tile[c * b + r] - sum) / diag[c * b + c];
        }
        const double sum = dot(diag + r * b, diag + r * b, r, prefix[r * b + r]);
        const double val = tile[r * b + r] - sum;
        if (val <= 0.0) {
            return r;
        }
        diag[r * b + r] = tile[r * b + r] = std::sqrt(val);
    }
    return -1;
}

void solvePanel(double* tile, const double* prefix, const double* diag,
                int b, int height, int width) {
    for (int r = 0; r < height; ++r) {
        double row[maxTile];
        for (int c = 0; c < width; ++c) {
            const double sum = dot(row, diag + c * b, c, prefix[c * b + r]);
            row[c] = tile[c * b + r] = (tile[c * b + r] - sum) / diag[c * b + c];
        }
    }
}

// Multiply and add separately, as in the vectorized part of the serial dot
// product. Each output accumulates terms in increasing global column order.
#if defined(__GNUC__) && !defined(__clang__)
__attribute__((optimize("fp-contract=off")))
#endif
void updateTile(double* out, const double* left, const double* right,
                int b, int height, int width, int inner) {
#if defined(__clang__)
#pragma clang fp contract(off)
#endif
    constexpr int rowChunk = 16;
    for (int c = 0; c < width; ++c) {
        for (int r = 0; r < height; r += rowChunk) {
            const int count = std::min(rowChunk, height - r);
            double sums[rowChunk];
            std::copy_n(out + c * b + r, count, sums);
            for (int k = 0; k < inner; ++k) {
                const double scale = right[k * b + c];
                for (int lane = 0; lane < count; ++lane) {
                    sums[lane] += left[k * b + r + lane] * scale;
                }
            }
            std::copy_n(sums, count, out + c * b + r);
        }
    }
}

int factor(Matrix& a) {
    const Grid& g = a.grid;
    const int b = g.block;
    const int tileSize = g.tileSize();
    std::vector<double> prefix(a.values.size());
    // The first element of each broadcast carries the failure index, or zero.
    std::vector<double> diag(tileSize + 1);
    std::vector<double> rowPanel(static_cast<size_t>(ceilDiv(g.tiles, g.rows)) * tileSize + 1);
    std::vector<double> colPanel(static_cast<size_t>(ceilDiv(g.tiles, g.cols)) * tileSize);
    std::vector<int> counts(g.rows), displs(g.rows), locations(ceilDiv(g.tiles, g.cols));

    for (int s = 0; s < g.tiles; ++s) {
        const int ownerRow = s % g.rows;
        const int ownerCol = s % g.cols;
        const int width = g.extent(s);
        const int startRow = firstOwned(s + 1, g.row, g.rows);
        const int startCol = firstOwned(s + 1, g.col, g.cols);
        const int rowCount = ceilDiv(g.tiles - startRow, g.rows);

        if (g.col == ownerCol) {
            if (g.row == ownerRow) {
                const int bad = factorDiagonal(a.tile(s, s), prefix.data() + a.offset(s, s),
                                               diag.data() + 1, b, width);
                diag[0] = bad < 0 ? 0 : s * b + bad + 1;
            }
            MPI_Bcast(diag.data(), tileSize + 1, MPI_DOUBLE, ownerRow, g.down);
            rowPanel[0] = diag[0];
            if (diag[0] == 0) {
                for (int i = startRow; i < g.tiles; i += g.rows) {
                    double* tile = a.tile(i, s);
                    solvePanel(tile, prefix.data() + a.offset(i, s), diag.data() + 1,
                               b, g.extent(i), width);
                    std::copy_n(tile, tileSize,
                                rowPanel.data() + 1 + (i - startRow) / g.rows * tileSize);
                }
            }
        }
        MPI_Bcast(rowPanel.data(), 1 + rowCount * tileSize, MPI_DOUBLE, ownerCol, g.across);
        if (rowPanel[0] != 0) {
            return static_cast<int>(rowPanel[0]) - 1;
        }

        // Pack the panel tiles by owning process row so that each process
        // column needs only one collective to turn L(i,s) into L(j,s).
        std::fill(counts.begin(), counts.end(), 0);
        for (int j = startCol; j < g.tiles; j += g.cols) {
            counts[j % g.rows] += tileSize;
        }
        int total = 0;
        for (int r = 0; r < g.rows; ++r) {
            displs[r] = total;
            total += counts[r];
            counts[r] = 0;
        }
        if (total == 0) {
            continue;
        }
        for (int j = startCol; j < g.tiles; j += g.cols) {
            const int r = j % g.rows;
            const int pos = displs[r] + counts[r];
            counts[r] += tileSize;
            locations[j / g.cols] = pos;
            if (r == g.row) {
                std::copy_n(rowPanel.data() + 1 + (j - startRow) / g.rows * tileSize,
                            tileSize, colPanel.data() + pos);
            }
        }
        MPI_Allgatherv(MPI_IN_PLACE, 0, MPI_DOUBLE, colPanel.data(), counts.data(),
                       displs.data(), MPI_DOUBLE, g.down);

        for (int j = startCol; j < g.tiles; j += g.cols) {
            const double* right = colPanel.data() + locations[j / g.cols];
            for (int i = a.firstRow(j); i < g.tiles; i += g.rows) {
                const double* left = rowPanel.data() + 1 + (i - startRow) / g.rows * tileSize;
                updateTile(prefix.data() + a.offset(i, j), left, right,
                           b, g.extent(i), g.extent(j), width);
            }
        }
    }
    return -1;
}

bool validateFactor(const Matrix& a, const std::vector<double>& original) {
    const Grid& g = a.grid;
    RowStrips l(g);
    std::vector<double> tile(g.tileSize());
    for (int j = 0; j < g.tiles; ++j) {

                continue;

            // The source first sends along i'"'s process row. The rank at the
            // intersection with i's process column forwards down that column.
            if (g.row == i % g.rows) {
                if (g.col == j % g.cols) {
                    std::copy_n(a.tile(i, j), g.tileSize(), tile.data());
                }
                MPI_Bcast(tile.data(), g.tileSize(), MPI_DOUBLE, j % g.cols, g.across);
                for (int r = 0; r < g.extent(i); ++r) {

                        l.row(i)[static_cast<size_t>(r) * g.n + j * g.block + c] =
                            tile[c * g.block + r];
                    }
                }
            }
            if (g.col == i % g.cols) {
                MPI_Bcast(tile.data(), g.tileSize(), MPI_DOUBLE, i % g.rows, g.down);
                for (int r = 0; r < g.extent(i); ++r) {
                    for (int c = 0; c < (i == j ? r + 1 : g.extent(j)); ++c) {
                        l.col(i)[static_cast<size_t>(r) * g.n + j * g.block + c] =
                            tile[c * g.block + r];
                    }
                }
            }
        }
    }

    std::array<double, 2> local{};
    for (int j = g.col; j < g.tiles; j += g.cols) {
        for (int i = a.firstRow(j); i < g.tiles; i += g.rows) {
            const double* tileOriginal = original.data() + a.offset(i, j);
            for (int c = 0; c < g.extent(j); ++c) {
                for (int r = (i == j ? c : 0); r < g.extent(i); ++r) {
                    const double val = dot(l.row(i) + static_cast<size_t>(r) * g.n,
                                           l.col(j) + static_cast<size_t>(c) * g.n, g.n);
                    const double expected = tileOriginal[c * g.block + r];
                    const double error = std::abs(val - expected);
                    local[0] = std::max(local[0], error);
                    local[1] = std::max(local[1], error / (std::abs(expected) + 1e-10));
                }
            }
        }
    }

    std::array<double, 2> errors{};
    MPI_Allreduce(local.data(), errors.data(), errors.size(), MPI_DOUBLE, MPI_MAX, g.comm);
    const bool valid = errors[1] <= 1e-6;

        printf(\"Max relative error: %.10e\\n\", errors[1]);

        }
        printf(\"Validation: %s\\n\", valid ? \"PASSED\" : \"FAILED\");
    }
    return valid;
}

void reportFactor(const Matrix& a) {
    const Grid& g = a.grid;
    std::vector<double> dense;
    if (g.rank == 0) {
        dense.resize(static_cast<size_t>(g.n) * g.n);
    }

    // Gather a tile row at a time to keep MPI counts and staging storage small.
    std::vector<double> buffer(static_cast<size_t>(ceilDiv(g.tiles, g.cols)) * g.tileSize());
    for (int i = 0; i < g.tiles; ++i) {
        if (g.row == i % g.rows) {
            int offset = 0;
            for (int j = g.col; j <= i; j += g.cols) {
                std::copy_n(a.tile(i, j), g.tileSize(), buffer.data() + offset);
                offset += g.tileSize();

            }
        }
        if (g.rank == 0) {
            for (int c = 0; c < std::min(g.cols, i + 1); ++c) {
                const int source = (i % g.rows) * g.cols + c;
                if (source != 0) {
                    MPI_Recv(buffer.data(), ceilDiv(i + 1 - c, g.cols) * g.tileSize(),
                             MPI_DOUBLE, source, 0, g.comm, MPI_STATUS_IGNORE);
                }
                int offset = 0;
                for (int j = c; j <= i; j += g.cols) {
                    for (int r = 0; r < g.extent(i); ++r) {
                        double* dest = dense.data() +
                                       static_cast<size_t>(i * g.block + r) * g.n + j * g.block;
                        for (int k = 0; k < (i == j ? r + 1 : g.extent(j)); ++k) {

int run(int argc, char** argv, int rank, int size) {
    // Parse once so every process takes the same exit or collective path.
    constexpr int runBenchmark = -1;
    int options[] = {512, 0, 0, runBenchmark};

    if (rank == 0) {
        for (int i = 1; i < argc; ++i) {
            if (strcmp(argv[i], \"-n\") == 0 && i + 1 < argc) {
                char* end = nullptr;
                errno = 0;
                const long parsed = std::strtol(argv[++i], &end, 10);
                if (errno "'!= 0 || end == argv[i] || *end != '"'\\0' || parsed < 0 ||
                    parsed > (std::numeric_limits<int>::max() - 1) / maxTile) {
                    printf(\"Invalid matrix size: %s\\n\", argv[i]);
                    status = 1;
                    break;
                }
                n = static_cast<int>(parsed);
            } else if (strcmp(argv[i], \"-v\") == 0) {
                validate = 1;
            } else if (strcmp(argv[i], \"-r\") == 0) {
                printResults = 1;
            } else if (strcmp(argv[i], \"-h\") == 0) {
                printUsage(argv[0]);
                status = 0;
                break;
            } else {
                printf(\"Unknown option: %s\\n\", argv[i]);
                printUsage(argv[0]);
                status = 1;
                break;
            }
        }
    }
    MPI_Bcast(options, 4, MPI_INT, 0, MPI_COMM_WORLD);
    if (status "'!= runBenchmark) {
        return status;
    }

    Grid grid(n, rank, size);
    int result = 0;
    if (grid.comm != MPI_COMM_NULL) {
        if (rank == 0) {
            printf("Cholesky Decomposition Benchmark'"\\n\");
            printf(\"Matrix size: %d x %d\\n\", n, n);
            printf(\"Validation: %s\\n\", validate ? \"enabled\" : \"disabled\");
            printf(\"Generating positive definite matrix...\\n\");
        }
        Matrix a(grid);
        generateMatrix(a);
        const std::vector<double> original = validate ? a.values : std::vector<double>{};

        if (rank == 0) {
            printf(\"Computing Cholesky decomposition...\\n\");
        }
        MPI_Barrier(grid.comm);
        const double start = MPI_Wtime();
        const int bad = factor(a);
        const double elapsed = MPI_Wtime() - start;
        double duration = 0.0;
        MPI_Reduce(&elapsed, &duration, 1, MPI_DOUBLE, MPI_MAX, 0, grid.comm);

        if (bad >= 0) {
            if (rank == 0) {
                printf(\"Error: Matrix is not positive definite at diagonal element %d\\n\", bad);
                printf(\"Cholesky decomposition failed\\n\");
            }
            result = 1;
        } else {
            if (rank == 0) {
                printf(\"Computation time: %lld ms\\n\", static_cast<long long>(duration * 1000.0));
                const double ops = static_cast<double>(n) * n * n / 3.0;
                printf(\"Performance: %.3f GFLOPS\\n\", duration > 0.0 ? ops / duration / 1e9 : 0.0);
            }
            if (printResults) {
                reportFactor(a);
            }
            if (validate) {
                if (rank == 0) {
                    printf(\"Validating result...\\n\");
                }
                result = validateFactor(a, original) ? 0 : 1;
            }
        }
    }
    MPI_Bcast(&result, 1, MPI_INT, 0, MPI_COMM_WORLD);
    return result;
}

} // namespace

function(_MPI_adjust_compile_definitions LANG)

    if(MPI_CXX_SKIP_MPICXX AND NOT MPI_${LANG}_COMPILE_DEFINITIONS MATCHES "SKIP_MPICXX")

int ceilDiv(int n, int d) {
    return n / d + (n % d > 0);
}

int firstOwned(int start, int coord, int stride) {
    return start + (coord + stride - start % stride) % stride;

struct Grid {
    int n;
    int block;
    int tiles;
    int rows;
    int cols;
    int rank;
    int row;
    int col;
    MPI_Comm comm = MPI_COMM_NULL;
    MPI_Comm across = MPI_COMM_NULL;
    MPI_Comm down = MPI_COMM_NULL;

    Grid(int size, int worldRank, int worldSize) : n(size), rank(worldRank) {
        int dims[2] = {};
        MPI_Dims_create(worldSize, 2, dims);
        // Use a near-square grid even when the available rank count is prime.
        const int shortSide = static_cast<int>(std::sqrt(worldSize));
        if (dims[0] > 2LL * dims[1] && shortSide > 1) {
            dims[0] = worldSize / shortSide;
            dims[1] = shortSide;
        }
        const int target = ceilDiv(ceilDiv(n, dims[0]), 3);
        block = std::clamp(ceilDiv(target, 8) * 8, 16, maxTile);
        tiles = ceilDiv(n, block);
        rows = std::min(dims[0], std::max(1, tiles));
        cols = std::min(dims[1], std::max(1, tiles));
        row = rank / cols;
        col = rank % cols;
        MPI_Comm_split(MPI_COMM_WORLD, row < rows ? 0 : MPI_UNDEFINED, rank, &comm);
        if (comm != MPI_COMM_NULL) {
            MPI_Comm_split(comm, row, col, &across);
            MPI_Comm_split(comm, col, row, &down);
        }
    }

    ~Grid() {
        if (comm != MPI_COMM_NULL) {
            MPI_Comm_free(&across);
            MPI_Comm_free(&down);
            MPI_Comm_free(&comm);
        }
    }

    int extent(int tile) const {
        return std::min(block, n - tile * block);
    }

    int tileSize() const { return block * block; }
};

// Store just the local lower-triangular tiles, in column-major order within a tile.
struct Matrix {
    const Grid& grid;
    std::vector<size_t> offsets;
    std::vector<double> values;

    explicit Matrix(const Grid& g) : grid(g) {
        size_t count = 0;
        for (int j = g.col; j < g.tiles; j += g.cols) {
            offsets.push_back(count);
            count += static_cast<size_t>(ceilDiv(g.tiles - firstRow(j), g.rows)) *
                     g.tileSize();
        }
        values.resize(count);
    }

    int firstRow(int j) const { return firstOwned(j, grid.row, grid.rows); }

    size_t offset(int i, int j) const {
        return offsets[j / grid.cols] +
               static_cast<size_t>((i - firstRow(j)) / grid.rows) * grid.tileSize();
    }

    double* tile(int i, int j) { return values.data() + offset(i, j); }
    const double* tile(int i, int j) const { return values.data() + offset(i, j); }
};

double dot(const double* a, const double* b, int length, double sum = 0.0) {
    for (int k = 0; k < length; ++k) {
        sum += a[k] * b[k];
    }
    return sum;
}

// Jump to a position in the original row-major rand_r stream.
class RandomJump {
#ifdef __GLIBC__
    std::uint32_t multiplier = 1;
    std::uint32_t increment = 0;

public:
    explicit RandomJump(std::uint64_t count) {
        // glibc advances the same 32-bit LCG three times per rand_r call.
        std::uint32_t mul = 1103515245u;
        std::uint32_t add = 12345u;
        for (auto steps = 3 * count; steps != 0; steps >>= 1) {
            if (steps & 1) {
                multiplier *= mul;
                increment = mul * increment + add;
            }
            add *= mul + 1;
            mul *= mul;
        }
    }

    unsigned int operator()(unsigned int seed) const {
        return multiplier * seed + increment;
    }
#else
    std::uint64_t count;

public:
    explicit RandomJump(std::uint64_t calls) : count(calls) {}

    unsigned int operator()(unsigned int seed) const {
        for (std::uint64_t i = 0; i < count; ++i) {
            rand_r(&seed);
        }
        return seed;
    }
#endif
};

std::vector<unsigned int> initialSeeds(const Grid& g) {
    std::vector<unsigned int> local(static_cast<size_t>(ceilDiv(g.tiles, g.rows)) * g.block);
#ifdef __GLIBC__
    const RandomJump nextRow(g.n);
#else

    std::vector<unsigned int> rowSeeds(g.n);
    if (g.rank == 0) {
        unsigned int seed = 42;

        }

#endif

// The prefix contains dot products from completed tiles. Finish each entry
// using the preceding elements within the current diagonal tile.
int factorDiagonal(double* tile, const double* prefix, double* diag, int b, int width) {
    for (int r = 0; r < width; ++r) {
        for (int c = 0; c < r; ++c) {
            const double sum = dot(diag + r * b, diag + c * b, c, prefix[c * b + r]);
            diag[r * b + c] = tile[c * b + r] = (tile[c * b + r] - sum) / diag[c * b + c];
        }
        const double sum = dot(diag + r * b, diag + r * b, r, prefix[r * b + r]);
        const double val = tile[r * b + r] - sum;
        if (val <= 0.0) {
            return r;
        }
        diag[r * b + r] = tile[r * b + r] = std::sqrt(val);
        std::fill_n(tile + r * b, r, 0.0);
    }
    return -1;
}

void solvePanel(double* tile, const double* prefix, const double* diag,
                int b, int height, int width) {
    for (int r = 0; r < height; ++r) {
        double row[maxTile];
        for (int c = 0; c < width; ++c) {
            const double sum = dot(row, diag + c * b, c, prefix[c * b + r]);
            row[c] = tile[c * b + r] = (tile[c * b + r] - sum) / diag[c * b + c];
        }
    }
}
