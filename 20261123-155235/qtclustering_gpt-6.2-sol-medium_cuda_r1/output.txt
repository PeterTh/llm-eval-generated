// QT Clustering Benchmark - Simplified Sequential Version
// 
// QT (Quality Threshold) clustering is an algorithm that builds clusters
// by starting with a seed point and iteratively adding the closest point
// that maintains the cluster's diameter below a threshold.

#include <algorithm>
#include <chrono>
#include <cmath>
#include <cstdio>
#include <cstdlib>
#include <cstring>

static const double MAX_WIDTH = 20.0;
static const double MAX_HEIGHT = 20.0;

// Structure to represent a point in 2D space
struct Point {
    double x, y;
};

// Structure to represent a cluster
struct Cluster {
    std::vector<int> members;
    int seed_point;
};

// Generate synthetic 2D point data in clusters
void generateSyntheticData(std::vector<Point>& points, const int N, unsigned int seed = 42) {
    auto frand = [&seed]() mutable { return rand_r(&seed) / static_cast<double>(RAND_MAX); };
    
    const double min_dim = std::min(MAX_WIDTH, MAX_HEIGHT);
    int count = 0;
    
    while (count < N) {
        // Create group_cnt points within a circle of radius R
        // around center point (cntr_x, cntr_y)
        const double cntr_x = frand() * MAX_WIDTH;
        const double cntr_y = frand() * MAX_HEIGHT;
        const double R = frand() * min_dim / 2.0;
        int group_cnt = static_cast<int>(frand() * (N / 30.0));
        
        // Make sure we don't make more points than we need
        if (group_cnt > (N - count)) {
            group_cnt = N - count;
        }
        
        while (group_cnt > 0) {
            const double sign = (frand() < 0.5) ? -1.0 : 1.0;
            const double r = frand() * R;
            const double dx = (2.0 * frand() - 1.0) * r;
            const double dy = std::sqrt(r * r - dx * dx) * sign;
            const double x = cntr_x + dx;
            const double y = cntr_y + dy;
            
            if (x < 0 || x > MAX_WIDTH || y < 0 || y > MAX_HEIGHT) {
                continue;
            }
            
            points[count] = {x, y};
            count++;
            group_cnt--;
        }
    }
}

// Calculate Euclidean distance between two points
inline double distance(const Point& p1, const Point& p2) {
    double dx = p1.x - p2.x;
    double dy = p1.y - p2.y;
    return std::sqrt(dx * dx + dy * dy);
}

// Find the closest unclustered point to the current cluster that maintains diameter < threshold
// Returns -1 if no such point exists
int findClosestPoint(const std::vector<int>& cluster_members,
                     const std::vector<bool>& clustered,
                     const std::vector<bool>& in_cluster,
                     const std::vector<Point>& points,
                     const double threshold,
                     const int point_count) {
    int closest_point = -1;
    double min_diameter = std::numeric_limits<double>::max();
    
    // Try each unclustered point as a candidate
    for (int candidate = 0; candidate < point_count; ++candidate) {
        // Skip if already clustered or already in this cluster
        if (clustered[candidate] || in_cluster[candidate]) continue;
        
        // Calculate the maximum distance from candidate to all cluster members
        double max_dist = 0.0;
        for (size_t i = 0; i < cluster_members.size(); ++i) {
            const int member = cluster_members[i];
            const double dist = distance(points[candidate], points[member]);
            max_dist = std::max(max_dist, dist);
        }
        
        // If adding this point keeps diameter below threshold and is better than current best
        if (max_dist < threshold && max_dist < min_diameter) {
            min_diameter = max_dist;
            closest_point = candidate;
        }
    }
    
    return closest_point;
}

// Generate a candidate cluster starting from a seed point
// Returns the cardinality (size) of the cluster
int generateCandidateCluster(const int seed_point,
                              const std::vector<bool>& clustered,
                              const std::vector<Point>& points,
                              const double threshold,
                              const int point_count,
                              std::vector<int>* cluster_members = nullptr) {
    std::vector<bool> in_cluster(point_count, false);
    std::vector<int> members;
    
    // Add seed point
    in_cluster[seed_point] = true;
    members.push_back(seed_point);
    
    // Iteratively add closest points
    while (static_cast<int>(members.size()) < point_count) {
        // Find closest point to current cluster that maintains diameter < threshold
        const int closest = findClosestPoint(members, clustered, in_cluster, points, 
                                             threshold, point_count);
        
        if (closest < 0) break; // No more points can be added
        
        in_cluster[closest] = true;
        members.push_back(closest);
    }
    
    // Copy members if requested
    if (cluster_members) {
        *cluster_members = members;
    }
    
    return static_cast<int>(members.size());
}

// Main QT clustering algorithm
std::vector<Cluster> qtClustering(const std::vector<Point>& points,
                                  const double threshold) {
    const int N = static_cast<int>(points.size());
    std::vector<bool> clustered(N, false);
    std::vector<int> unclustered_indices;
    std::vector<Cluster> clusters;
    
    // Initialize unclustered indices
    for (int i = 0; i < N; ++i) {
        unclustered_indices.push_back(i);
    }
    
    // Main clustering loop
    while (!unclustered_indices.empty()) {
        int max_cardinality = -1;
        int best_seed = -1;
        std::vector<int> best_cluster_members;
        
        // Try each unclustered point as a seed
        for (size_t i = 0; i < unclustered_indices.size(); ++i) {
            const int seed = unclustered_indices[i];
            if (clustered[seed]) continue;
            
            std::vector<int> candidate_members;
            const int cardinality = generateCandidateCluster(seed, clustered, points, 
                                                       threshold, N, 
                                                       &candidate_members);
            
            if (cardinality > max_cardinality) {
                max_cardinality = cardinality;
                best_seed = seed;
                best_cluster_members = candidate_members;
            }
        }
        
        // If we found a cluster, add it
        if (best_seed >= 0 && max_cardinality > 0) {
            Cluster cluster;
            cluster.seed_point = best_seed;
            cluster.members = best_cluster_members;
            clusters.push_back(cluster);
            
            // Mark all members as clustered
            for (size_t i = 0; i < best_cluster_members.size(); ++i) {
                clustered[best_cluster_members[i]] = true;
            }
            
            // Remove clustered points from unclustered list
            unclustered_indices.erase(
                std::remove_if(unclustered_indices.begin(), unclustered_indices.end(),
                              [&clustered](int idx) { return clustered[idx]; }),
                unclustered_indices.end()
            );
        } else {
            // No more clusters can be formed
            break;
        }
    }
    
    return clusters;
}

// Validation: check that clusters satisfy the QT clustering properties
bool validateClusters(const std::vector<Cluster>& clusters,
                     const std::vector<Point>& points,
                     const double threshold) {
    bool valid = true;
    
    printf("Validating clusters:\n");
    
    // Check each cluster
    for (size_t c = 0; c < clusters.size(); ++c) {
        const auto& cluster = clusters[c];
        double max_diameter = 0.0;
        
        // Check diameter (max distance between any two points)
        for (size_t i = 0; i < cluster.members.size(); ++i) {
            for (size_t j = i + 1; j < cluster.members.size(); ++j) {
                const double dist = distance(points[cluster.members[i]], 
                                           points[cluster.members[j]]);
                max_diameter = std::max(max_diameter, dist);
            }
        }
        
        if (c < 10) { // Print first 10 clusters
            printf("  Cluster %zu: size=%zu, seed=%d, diameter=%.4f\n", 
                   c, cluster.members.size(), cluster.seed_point, max_diameter);
        }
        
        // Validate diameter is within threshold
        if (max_diameter > threshold * 1.001) { // Allow small numerical error
            printf("ERROR: Cluster %zu has diameter %.4f > threshold %.4f\n", 
                   c, max_diameter, threshold);
            valid = false;
        }
    }
    
    // Check for duplicate memberships
    std::vector<int> membership(points.size(), -1);
    for (size_t c = 0; c < clusters.size(); ++c) {
        for (size_t i = 0; i < clusters[c].members.size(); ++i) {
            const int member = clusters[c].members[i];
            if (membership[member] >= 0) {
                printf("ERROR: Point %d appears in multiple clusters (%d and %zu)\n",
                       member, membership[member], c);
                valid = false;
            }
            membership[member] = static_cast<int>(c);
        }
    }
    
    // Count clustered points
    int clustered_count = 0;
    for (size_t i = 0; i < membership.size(); ++i) {
        if (membership[i] >= 0) clustered_count++;
    }
    
    printf("Total points: %zu, Clustered: %d, Unclustered: %zu\n",
           points.size(), clustered_count, points.size() - clustered_count);
    
    return valid;
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// Print results for floating-point arrays
template <typename T>
void print_results(const std::vector<T>& data, const char* name = "Result") {
    if (data.empty()) {
        printf("=== RESULTS ===\n");
        printf("Name: %s\n", name);
        printf("Elements: 0\n");
        printf("=== END RESULTS ===\n");
        return;
    }

    const size_t N = data.size();
    
    // Compute statistics
    T sum = kahan_sum(data);
    T min_val = data[0];
    T max_val = data[0];
    for (const auto& val : data) {
        min_val = std::min(min_val, val);
        max_val = std::max(max_val, val);
    }
    
    // Compute hash
    hash h;
    for (const auto& val : data) {
        h.add(val);
    }
    
    // Sample indices: 0, N/4, N/2, 3N/4, N-1
    size_t indices[5] = {0, N/4, N/2, 3*N/4, N-1};
    
    printf("=== RESULTS ===\n");
    printf("Name: %s\n", name);
    printf("Elements: %zu\n", N);
    printf("Sum: %.17e\n", static_cast<double>(sum));
    printf("Min: %.17e\n", static_cast<double>(min_val));
    printf("Max: %.17e\n", static_cast<double>(max_val));
    for (int i = 0; i < 5; ++i) {
        printf("Sample[%zu]: %.17e\n", indices[i], static_cast<double>(data[indices[i]]));
    }
    printf("Hash: %016" PRIx64 "\n", h.get());
    printf("=== END RESULTS ===\n");
}

CUDA_ARCHITECTURES
------------------

.. versionadded:: 3.18

List of architectures to generate device code for.

An architecture can be suffixed by either ``-real`` or ``-virtual`` to specify
the kind of architecture to generate code for.
If no suffix is given then code is generated for both real and virtual
architectures.

A non-empty false value (e.g. ``OFF``) disables adding architectures.
This is intended to support packagers and rare cases where full control
over the passed flags is required.

This property is initialized by the value of the ``CMAKE_CUDA_ARCHITECTURES``
variable if it is set when a target is created.

The ``CUDA_ARCHITECTURES`` target property must be set to a non-empty value on targets
that compile CUDA sources, or it is an error.  See policy ``CMP0104``.

The ``CUDA_ARCHITECTURES`` may be set to one of the following special values:

``all``
  .. versionadded:: 3.23

  Compile for all supported major and minor real architectures,
  and the highest major virtual architecture.

``all-major``
  .. versionadded:: 3.23

  Compile for all supported major real architectures, and the highest
  major virtual architecture.

``native``
  .. versionadded:: 3.24

  Compile for the architecture(s) of the host's GPU(s).

__device__ double squaredDistance(Point a, Point b) {
    const double dx = a.x - b.x;
    const double dy = a.y - b.y;
    const double yy = __dmul_rn(dy, dy);
#ifdef __FMA__
    // Match the original host build'"'s contraction of dx * dx + dy * dy.
    return __fma_rn(dx, dx, yy);
#else
    return __dadd_rn(__dmul_rn(dx, dx), yy);
#endif
}

// Find the largest squared distance whose rounded square root passes the
// original strict comparison. This lets every edge test skip the square root.
double squaredCutoff(double threshold) {
    if ("'!(threshold > 0.0)) return -1.0;

    uint64_t low = 0;
    uint64_t high = UINT64_C(0x7ff0000000000000); // Positive infinity.
    while (high - low > 1) {
        const uint64_t mid = low + (high - low) / 2;
        double squared;
        std::memcpy(&squared, &mid, sizeof(squared));
        if (std::sqrt(squared) < threshold) {
            low = mid;
        } else {
            high = mid;
        }
    }
    double cutoff;
    std::memcpy(&cutoff, &low, sizeof(cutoff));
    return cutoff;
}

// Build rows in point-index order so a smaller neighbor slot wins ties.
template <bool writeNeighbors>
__global__ void buildNeighborhoods(const Point* points, int n, double cutoff,
                                   int seedCount, const size_t* offsets,
                                   size_t* degrees, int* neighbors, int* maxDegree) {
    const int seed = blockIdx.x * warpsPerBlock + threadIdx.x / warpSizeCuda;
    const int lane = threadIdx.x % warpSizeCuda;
    if (seed >= seedCount) return;

    const Point origin = points[seed];
    int degree = 0;
    for (size_t base = 0; base < static_cast<size_t>(n); base += warpSizeCuda) {
        const size_t candidate = base + lane;
        const bool fits = candidate < static_cast<size_t>(n) &&
                          candidate != static_cast<size_t>(seed) &&
                          fmax(0.0, squaredDistance(points[candidate], origin)) <= cutoff;
        const unsigned mask = __ballot_sync(fullWarp, fits);
        if constexpr (writeNeighbors) {
            if (fits) {
                const int slot = degree + __popc(mask & ((1u << lane) - 1));
                neighbors[offsets[seed] + slot] = static_cast<int>(candidate);
            }
        }
        degree += __popc(mask);
    }

    if constexpr (!writeNeighbors) {
        if (lane == 0) {
            degrees[seed] = degree;
            if (degree > registerNeighbors) atomicMax(maxDegree, degree);
            if (seed == 0) degrees[n] = 0;
        }
    }
}

constexpr int noNeighbor = std::numeric_limits<int>::max();

template <int slots>
__device__ int growCandidate(const Point* points, const int* neighbors,
                             const int* clustered, int* selected, double* scratch,
                             int degree, int lane, Point newest, double cutoff) {
    // A candidate'"'s farthest distance only needs an update against the member
    // just added. A point above the cutoff can never reenter the candidate set.
    double bounds[slots == 0 ? 1 : slots];
    int ids[slots == 0 ? 1 : slots];
    const double rejected = CUDART_INF;
    const int laneSlots = slots == 0 ? (degree - lane + warpSizeCuda - 1) / warpSizeCuda : slots;

    #pragma unroll
    for (int i = 0; i < laneSlots; ++i) {
        const int slot = lane + i * warpSizeCuda;
        const int point = slot < degree ? neighbors[slot] : -1;
        const double initial = point >= 0 && "'!clustered[point] ? 0.0 : rejected;
        if constexpr (slots == 0) {
            scratch[slot] = initial;
        } else {
            ids[i] = point;
            bounds[i] = initial;
        }
    }

    int count = 1;
    for (;;) {
        double bestDistance = rejected;
        int bestSlot = noNeighbor;

        #pragma unroll
        for (int i = 0; i < laneSlots; ++i) {
            const int slot = lane + i * warpSizeCuda;
            const double oldBound = slots == 0 ? scratch[slot] : bounds[i];
            if (oldBound != rejected) {
                const int point = slots == 0 ? neighbors[slot] : ids[i];
                const double bound = fmax(oldBound, squaredDistance(points[point], newest));
                const double updated = bound <= cutoff ? bound : rejected;
                if constexpr (slots == 0) {
                    scratch[slot] = updated;
                } else {
                    bounds[i] = updated;
                }

                // Compare rounded distances, as distinct squared distances may
                // round to the same square root and tie in the original search.
                const double candidateDistance = __dsqrt_rn(updated);
                if (candidateDistance < bestDistance) {
                    bestDistance = candidateDistance;
                    bestSlot = slot;
                }
            }
        }

        for (int offset = warpSizeCuda / 2; offset > 0; offset /= 2) {
            const double otherDistance = __shfl_down_sync(fullWarp, bestDistance, offset);
            const int otherSlot = __shfl_down_sync(fullWarp, bestSlot, offset);
            if (otherDistance < bestDistance ||
                (otherDistance == bestDistance && otherSlot < bestSlot)) {
                bestDistance = otherDistance;
                bestSlot = otherSlot;
            }
        }
        const int winner = __shfl_sync(fullWarp, bestSlot, 0);
        if (winner == noNeighbor) return count;

        const int member = neighbors[winner];
        if (lane == 0) selected[count - 1] = member;
        ++count;
        if (count > degree) return count;

        if constexpr (slots == 0) {
            if (lane == winner % warpSizeCuda) scratch[winner] = rejected;
        } else {
            #pragma unroll
            for (int i = 0; i < slots; ++i) {
                if (lane + i * warpSizeCuda == winner) bounds[i] = rejected;
            }
        }
        newest = points[member];
    }
}

using CandidateKey = unsigned long long;

struct ClusteringState {
    CandidateKey best;
    int remaining;
    int clusterCount;
};

__global__ void evaluateSeeds(const Point* points, int seedCount, double cutoff,
                              const size_t* offsets, const int* neighbors, int* selected,
                              double* scratch, int* sizes, const int* clustered,
                              ClusteringState* state) {
    if (state->remaining == 0) return;

    const int warp = threadIdx.x / warpSizeCuda;
    const int lane = threadIdx.x % warpSizeCuda;
    const int seed = blockIdx.x * warpsPerBlock + warp;
    CandidateKey key = 0;
    if (seed < seedCount && !clustered[seed]) {
        const size_t begin = offsets[seed];
        const int degree = offsets[seed + 1] - begin;
        int count = sizes[seed];
        bool dirty = count == 0;
        // Removing an unselected point cannot change a greedy choice. Only a
        // cached candidate that lost a member needs to be generated again.
        for (int base = 0; !dirty && base < count - 1; base += warpSizeCuda) {
            const int slot = base + lane;
            dirty = __any_sync(fullWarp, slot < count - 1 && clustered[selected[begin + slot]]);
        }
        if (dirty) {
            const Point origin = points[seed];
            if (degree == 0) {
                count = 1;
            } else if (degree <= 32) {
                count = growCandidate<1>(points, neighbors + begin, clustered, selected + begin,
                                         nullptr, degree, lane, origin, cutoff);
            } else if (degree <= 64) {
                count = growCandidate<2>(points, neighbors + begin, clustered, selected + begin,
                                         nullptr, degree, lane, origin, cutoff);
            } else if (degree <= registerNeighbors) {
                count = growCandidate<4>(points, neighbors + begin, clustered, selected + begin,
                                         nullptr, degree, lane, origin, cutoff);
            } else {
                count = growCandidate<0>(points, neighbors + begin, clustered, selected + begin,
                                         scratch + begin, degree, lane, origin, cutoff);
            }
            if (lane == 0) sizes[seed] = count;
        }
        // Maximize cardinality, favoring the lowest seed index on a tie.
        key = (static_cast<CandidateKey>(count) << 32) | (fullWarp - seed);
    }

    __shared__ CandidateKey winners[warpsPerBlock];
    if (lane == 0) winners[warp] = key;
    __syncthreads();
    if (threadIdx.x == 0) {
        CandidateKey best = 0;
        for (int i = 0; i < warpsPerBlock; ++i) best = max(best, winners[i]);
        if (best != 0) atomicMax(&state->best, best);
    }
}

__global__ void commitCluster(int n, const size_t* offsets, const int* selected,
                              int* clustered, int* members, int* ends, ClusteringState* state) {
    __shared__ ClusteringState round;
    if (threadIdx.x == 0) round = *state;
    __syncthreads();
    if (round.remaining == 0) return;

    const int count = round.best >> 32;
    const int seed = fullWarp - static_cast<unsigned>(round.best);
    const int start = n - round.remaining;
    if (count == 1) {
        // Once the largest candidate is a singleton, all remaining points
        // form singleton clusters in ascending point order.
        using Scan = cub::BlockScan<int, blockSize>;
        __shared__ typename Scan::TempStorage scan;
        int written = 0;
        for (size_t base = 0; base < static_cast<size_t>(n); base += blockSize) {
            const size_t point = base + threadIdx.x;
            const bool active = point < static_cast<size_t>(n) && !clustered[point];
            int prefix, total;
            Scan(scan).ExclusiveSum(static_cast<int>(active), prefix, total);
            if (active) {
                members[start + written + prefix] = static_cast<int>(point);
                ends[round.clusterCount + written + prefix] = start + written + prefix + 1;
            }
            written += total;
            __syncthreads();
        }
        if (threadIdx.x == 0) {
            *state = {0, 0, round.clusterCount + round.remaining};
        }
        return;
    }

    const size_t begin = offsets[seed];
    for (int i = threadIdx.x; i < count; i += blockSize) {
        const int point = i == 0 ? seed : selected[begin + i - 1];
        members[start + i] = point;
        clustered[point] = 1;
    }
    if (threadIdx.x == 0) {
        ends[round.clusterCount] = start + count;
        *state = {0, round.remaining - count, round.clusterCount + 1};
    }
}

int seedBlocks(int count) {
    return (count - 1) / warpsPerBlock + 1;
}
