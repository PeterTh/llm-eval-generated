// Generate pseudo-random values for matrix initialization
constexpr double getPseudoRndValue(const size_t N, const size_t i, const size_t j) noexcept {
    return (((i + 1) * (i + j + 1) * 1299709) % (N * N)) / static_cast<double>(N * N);
}

void initMatrix(std::vector<double>& mat, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            mat[i * N + j] = getPseudoRndValue(N, i, j);
        }
    }
}

void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < N; ++k) {
                sum += A[i * N + k] * B[k * N + j];
            }
            C[i * N + j] = sum;
        }
    }
}

class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// Compute independent 64 x 64 output tiles. Each thread retains a 4 x 4
// microtile in registers and consumes shared input panels in increasing k.
__global__ void multiply(const double* a, const double* b, double* c,
                         int m, int n, int width) {
    __shared__ double lhs[16][65], rhs[16][64];
    const int x = threadIdx.x, y = threadIdx.y, lane = 16*y + x;
    const size_t col = size_t(blockIdx.x)*64, row = size_t(blockIdx.y)*64;
    double acc[4][4] = {};
    const int paired = n - (n % 2);
    for (size_t k = 0; k < paired; k += 16) {
        #pragma unroll
        for (int t = 0; t < 4; ++t) {
            const int p = lane + t*256;
            lhs[p%16][p/16] = row+p/16 < m && k+p%16 < paired
                ? a[(row+p/16)*n + k+p%16] : 0.0;
            rhs[p/64][p%64] = k+p/64 < paired && col+p%64 < width
                ? b[(k+p/64)*width + col+p%64] : 0.0;
        }
        __syncthreads();
        #pragma unroll
        for (int t = 0; t < 16; ++t) {
            #pragma unroll
            for (int i = 0; i < 4; ++i) {
                #pragma unroll
                for (int j = 0; j < 4; ++j) {
                    double aij = lhs[t][y+16*i], bij = rhs[t][x+16*j];
                    acc[i][j] = __dadd_rn(acc[i][j], __dmul_rn(aij, bij));
                }
            }
        }
        __syncthreads();
    }
    for (int i = 0; i < 4; ++i) {
        for (int j = 0; j < 4; ++j) {
            const size_t r = row+y+16*i, s = col+x+16*j;
            if (r < m && s < width) {
                double sum = acc[i][j];
                // The native vectorized reference contracts its odd scalar tail.
                if (n % 2) sum = __fma_rn(a[r*n+paired], b[size_t(paired)*width+s], sum);
                c[r*width+s] = sum;
            }
        }
    }
}

// location, splitting large messages to stay within MPI's int count limit.
static void report(const std::vector<double>& packed, int n, int rank, int ranks,
                   const int dims[2]) {
    std::vector<double> result(rank == 0 ? size_t(n)*n : 0);
    for (int source = rank == 0 ? 0 : rank; source < ranks; ++source) {
        Span r = partition(n, dims[0], source / dims[1]);
        Span c = partition(n, dims[1], source % dims[1]);
        if (c.count > 0) {
            for (int offset = 0; offset < r.count;) {
                int rows = std::min(r.count - offset, INT_MAX / c.count);
                size_t pos = size_t(offset) * c.count;
                if (rank != 0) {
                    MPI_Send(packed.data()+pos, rows*c.count, MPI_DOUBLE, 0, 0, MPI_COMM_WORLD);
                } else {
                    double* dest = result.data() + size_t(r.first+offset)*n + c.first;
                    if (source == 0) {
                        for (int i = 0; i < rows; ++i)
                            std::copy_n(packed.data()+pos+size_t(i)*c.count, c.count, dest+size_t(i)*n);
                    } else {
                        MPI_Datatype rect;
                        MPI_Type_vector(rows, c.count, n, MPI_DOUBLE, &rect);
                        MPI_Type_commit(&rect);
                        MPI_Recv(dest, 1, rect, source, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
                        MPI_Type_free(&rect);
                    }
                }
                offset += rows;
            }
        }
        if (rank != 0) break;
    }
    if (rank == 0) print_results(result, "MatrixC");
}

// Compute independent 64 x 64 output tiles. Each thread retains a 4 x 4
// microtile in registers and consumes shared input panels in increasing k.
template <bool exactProducts>
__global__ void multiply(const double *a, const double *b, double *c, int m, int n, int width) {
    __shared__ double lhs[16][65], rhs[16][64];
    const int x = threadIdx.x, y = threadIdx.y, lane = 16 * y + x;
    const size_t col = size_t(blockIdx.x) * 64, row = size_t(blockIdx.y) * 64;
    double acc[4][4] = {};
    const int paired = n - (n % 2);
    for (size_t k = 0; k < paired; k += 16) {
#pragma unroll
        for (int t = 0; t < 4; ++t) {
            const int p = lane + t * 256;
            lhs[p % 16][p / 16] =
                row + p / 16 < m && k + p % 16 < paired ? a[(row + p / 16) * n + k + p % 16] : 0.0;
            rhs[p / 64][p % 64] = k + p / 64 < paired && col + p % 64 < width
                                      ? b[(k + p / 64) * width + col + p % 64]
                                      : 0.0;
        }
        __syncthreads();
#pragma unroll
        for (int t = 0; t < 16; ++t) {
#pragma unroll
            for (int i = 0; i < 4; ++i) {
#pragma unroll
                for (int j = 0; j < 4; ++j) {
                    double aij = lhs[t][y + 16 * i], bij = rhs[t][x + 16 * j];
                    if constexpr (exactProducts)
                        acc[i][j] = __fma_rn(aij, bij, acc[i][j]);
                    else
                        acc[i][j] = __dadd_rn(acc[i][j], __dmul_rn(aij, bij));
                }
            }
        }
        __syncthreads();
    }
    for (int i = 0; i < 4; ++i) {
        for (int j = 0; j < 4; ++j) {
            const size_t r = row + y + 16 * i, s = col + x + 16 * j;
            if (r < m && s < width) {
                double sum = acc[i][j];
                // The native vectorized reference contracts its odd scalar tail.
                if (n % 2)
                    sum = __fma_rn(a[r * n + paired], b[size_t(paired) * width + s], sum);
                c[r * width + s] = sum;
            }
        }
    }
}

// Each sender holds a packed rectangle. Receive into its final row-major
// location, splitting large messages to stay within MPI's int count limit.
static void report(const std::vector<double> &packed, int n, int rank, int ranks, const int dims[2]) {
    std::vector<double> result(rank == 0 ? size_t(n) * n : 0);
    const int end = rank == 0 ? ranks : rank + 1;
    for (int source = rank; source < end; ++source) {
        Span r = partition(n, dims[0], source / dims[1]);
        Span c = partition(n, dims[1], source % dims[1]);
        if (c.count > 0) {
            for (int offset = 0; offset < r.count;) {
                int rows = std::min(r.count - offset, INT_MAX / c.count);
                size_t pos = size_t(offset) * c.count;
                if (rank != 0) {
                    MPI_Send(packed.data() + pos, rows * c.count, MPI_DOUBLE, 0, 0, MPI_COMM_WORLD);
                } else {
                    double *dest = result.data() + size_t(r.first + offset) * n + c.first;
                    if (source == 0) {
                        for (int i = 0; i < rows; ++i)
                            std::copy_n(packed.data() + pos + size_t(i) * c.count, c.count,
                                        dest + size_t(i) * n);
                    } else {
                        MPI_Datatype rect;
                        MPI_Type_vector(rows, c.count, n, MPI_DOUBLE, &rect);
                        MPI_Type_commit(&rect);
                        MPI_Recv(dest, 1, rect, source, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
                        MPI_Type_free(&rect);
                    }
                }
                offset += rows;
            }
        }
    }
    if (rank == 0)
        print_results(result, "MatrixC");
}
