// 3D index calculation
inline constexpr size_t idx3(const size_t x, const size_t y, const size_t z, const size_t nx, const size_t ny) noexcept {
    return z * (nx * ny) + y * nx + x;
}

void initializeGrid(std::vector<Real>& grid, const size_t nx, const size_t ny, const size_t nz) {
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {

    }
}

// 7-point stencil computation
void stencilIteration(const std::vector<Real>& input, 
                      std::vector<Real>& output,
                      const size_t nx, const size_t ny, const size_t nz) {
    // Process interior points (not on boundaries)
    for (size_t z = 1; z < nz - 1; ++z) {
        for (size_t y = 1; y < ny - 1; ++y) {
            for (size_t x = 1; x < nx - 1; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                
                const Real center = input[idx];
                const Real left = input[idx3(x-1, y, z, nx, ny)];
                const Real right = input[idx3(x+1, y, z, nx, ny)];
                const Real front = input[idx3(x, y-1, z, nx, ny)];
                const Real back = input[idx3(x, y+1, z, nx, ny)];
                const Real bottom = input[idx3(x, y, z-1, nx, ny)];
                const Real top = input[idx3(x, y, z+1, nx, ny)];
                
                // Simple averaging stencil
                output[idx] = (center + left + right + front + back + bottom + top) / 7.0;
            }
        }
    }
    
    // Copy boundary values
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                if (x == 0 || x == nx-1 || y == 0 || y == ny-1 || z == 0 || z == nz-1) {
                    const size_t idx = idx3(x, y, z, nx, ny);
                    output[idx] = input[idx];
                }
            }
        }
    }
}

                        size_t zBegin, size_t zEnd) {
    const size_t x = static_cast<size_t>(blockIdx.x) * blockDim.x + threadIdx.x + 1;
    const size_t y = static_cast<size_t>(blockIdx.y) * blockDim.y + threadIdx.y + 1;
    if (x >= nx - 1 || y >= ny - 1) return;
    const size_t plane = nx * ny;
    for (size_t z = zBegin + blockIdx.z; z < zEnd; z += gridDim.z) {
        const size_t globalZ = globalStart + z - 1;
        if (globalZ == 0 || globalZ + 1 == nz) continue;
        const size_t p = z * plane + y * nx + x;
        const Real sum = in[p] + in[p - 1] + in[p + 1] + in[p - nx] +
                         in[p + nx] + in[p - plane] + in[p + plane];
        out[p] = __ddiv_rn(sum, 7.0);
    }
}

static void launchStencil(const Real* in, Real* out, size_t nx, size_t ny, size_t nz,
                          size_t globalStart, size_t begin, size_t end, cudaStream_t stream) {
    if (begin >= end) return;
    const dim3 block(32, 8);
    const dim3 grid(static_cast<unsigned>((nx - 2 + block.x - 1) / block.x),
                    static_cast<unsigned>((ny - 2 + block.y - 1) / block.y),
                    static_cast<unsigned>(std::min<size_t>(end - begin, 65535)));
    stencil<<<grid, block, 0, stream>>>(in, out, nx, ny, nz, globalStart, begin, end);
    CUDA(cudaGetLastError());
}

Implemented the hybrid benchmark in the existing files. MPI distributes Z slabs across ranks, CUDA updates each slab on a GPU, and OpenMP parallelizes host initialization and validation. Halo transfers use pinned host memory and overlap with interior GPU computation.

The result hashes matched the original in single- and multi-rank tests, including thin slabs, more ranks than Z planes, grids with no interior cells, and zero, odd, and even iteration counts. Validation also passed for a tall grid that exercises CUDA’s grid-dimension limit.

The build requires MPI, OpenMP, and CUDA. On this machine, CUDA 12 required selecting GCC 12 as its host compiler during CMake configuration.
