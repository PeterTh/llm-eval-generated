// Generate pseudo-random values for matrix initialization
constexpr double getPseudoRndValue(const size_t N, const size_t i, const size_t j) noexcept {
    return (((i + 1) * (i + j + 1) * 1299709) % (N * N)) / static_cast<double>(N * N);
}

void initMatrix(std::vector<double>& mat, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            mat[i * N + j] = getPseudoRndValue(N, i, j);
        }
    }
}

void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < N; ++k) {
                sum += A[i * N + k] * B[k * N + j];
            }
            C[i * N + j] = sum;
        }
    }
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N) {
    // Four rows share each load from B. The eight independent column sums
    // are vectorized, while each sum still visits k in its original order.
    const size_t rowBlocks = (N + 3) / 4;
    const double work = static_cast<double>(N) * N * N;
    const int threads = static_cast<int>(std::max(1.0, std::min({
        static_cast<double>(omp_get_max_threads()), static_cast<double>(rowBlocks),
        std::ceil(work / 16000000.0)})));
    // Pack each eight-column panel so its k traversal is contiguous. This
    // avoids cache and TLB misses from repeatedly stepping by an entire row.
    const size_t panels = N / 8;
    std::vector<double> packedB(panels * N * 8);
    #pragma omp parallel num_threads(threads)
    {
    #pragma omp for schedule(static)
    for (size_t panel = 0; panel < panels; ++panel) {
        for (size_t k = 0; k < N; ++k) {
            #pragma omp simd
            for (size_t x = 0; x < 8; ++x)
                packedB[(panel * N + k) * 8 + x] = B[k * N + panel * 8 + x];
        }

        const size_t rows = std::min(size_t{4}, N - i);
