// Simple Cholesky decomposition (sequential, unblocked algorithm)
// Decomposes positive definite matrix A into L * L^T where L is lower triangular

bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
    // A is stored in row-major order
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j <= i; ++j) {
            double sum = 0.0;
            
            if (i == j) {
                // Diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[j * n + k] * A[j * n + k];
                }
                const double val = A[j * n + j] - sum;
                if (val <= 0.0) {
                    // Matrix is not positive definite
                    printf("Error: Matrix is not positive definite at diagonal element %zu\n", j);
                    return false;
                }
                A[j * n + j] = sqrt(val);
            } else {
                // Off-diagonal element
                for (size_t k = 0; k < j; ++k) {
                    sum += A[i * n + k] * A[j * n + k];
                }
                A[i * n + j] = (A[i * n + j] - sum) / A[j * n + j];
            }
        }
        
        // Zero out upper triangular part
        for (size_t j = i + 1; j < n; ++j) {
            A[i * n + j] = 0.0;
        }
    }
    
    return true;
}

// Generate a symmetric positive definite matrix
void generatePositiveDefiniteMatrix(std::vector<double>& A, const size_t n) {
    // Method: Create A = B * B^T where B is random
    // This guarantees A is positive semi-definite
    // Then add identity to make it strictly positive definite
    
    std::vector<double> B(n * n);
    unsigned int seed = 42;
    
    // Generate random matrix B
    for (size_t i = 0; i < n * n; ++i) {
        B[i] = (rand_r(&seed) / (double)RAND_MAX) - 0.5;
    }
    
    // Compute A = B * B^T
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j < n; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < n; ++k) {
                sum += B[i * n + k] * B[j * n + k];
            }
            A[i * n + j] = sum;
        }
    }
    
    // Add diagonal dominance to ensure positive definiteness
    for (size_t i = 0; i < n; ++i) {
        A[i * n + i] += n;
    }
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// A right-looking blocked Cholesky factorization. Each completed panel is
// applied to the lower triangular trailing matrix before the next panel.
bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
    constexpr size_t blockSize = 64;
    double* const matrix = A.data();
    bool success = true;

    #pragma omp parallel shared(success)
    {
        for (size_t start = 0; start < n; start += blockSize) {
            const size_t end = std::min(start + blockSize, n);

            // The diagonal block is small and must be factored in order.
            #pragma omp single
            {
                for (size_t j = start; j < end; ++j) {
                    double sum = 0.0;
                    for (size_t k = start; k < j; ++k) {
                        sum += matrix[j * n + k] * matrix[j * n + k];
                    }
                    const double val = matrix[j * n + j] - sum;
                    if (val <= 0.0) {
                        printf("Error: Matrix is not positive definite at diagonal element %zu\n", j);
                        success = false;
                        break;
                    }
                    matrix[j * n + j] = sqrt(val);
                    for (size_t i = j + 1; i < end; ++i) {
                        double dot = 0.0;
                        for (size_t k = start; k < j; ++k) {
                            dot += matrix[i * n + k] * matrix[j * n + k];
                        }
                        matrix[i * n + j] = (matrix[i * n + j] - dot) / matrix[j * n + j];
                    }
                }
            }
            // The implicit single barrier makes the status and block visible.
            if (!success) break;

            // Solve the panel below the diagonal block, independently by row.
            #pragma omp for schedule(static)
            for (size_t i = end; i < n; ++i) {
                for (size_t j = start; j < end; ++j) {
                    double dot = 0.0;
                    #pragma omp simd reduction(+:dot)
                    for (size_t k = start; k < j; ++k) {
                        dot += matrix[i * n + k] * matrix[j * n + k];
                    }
                    matrix[i * n + j] = (matrix[i * n + j] - dot) / matrix[j * n + j];
                }
            }

            // Rows have increasing lengths, so distribute them cyclically.
            // Four simultaneous dot products reuse the current panel row.
            #pragma omp for schedule(static, 1)
            for (size_t i = end; i < n; ++i) {
                const double* const row = matrix + i * n;
                size_t j = end;
                for (; j + 3 <= i; j += 4) {
                    const double* const row0 = matrix + j * n;
                    const double* const row1 = row0 + n;
                    const double* const row2 = row1 + n;
                    const double* const row3 = row2 + n;
                    double sum0 = 0.0, sum1 = 0.0, sum2 = 0.0, sum3 = 0.0;
                    #pragma omp simd reduction(+:sum0,sum1,sum2,sum3)
                    for (size_t k = start; k < end; ++k) {
                        const double x = row[k];
                        sum0 += x * row0[k];
                        sum1 += x * row1[k];
                        sum2 += x * row2[k];
                        sum3 += x * row3[k];
                    }
                    matrix[i * n + j] -= sum0;
                    matrix[i * n + j + 1] -= sum1;
                    matrix[i * n + j + 2] -= sum2;
                    matrix[i * n + j + 3] -= sum3;
                }
                for (; j <= i; ++j) {
                    double sum = 0.0;
                    #pragma omp simd reduction(+:sum)
                    for (size_t k = start; k < end; ++k) {
                        sum += row[k] * matrix[j * n + k];
                    }
                    matrix[i * n + j] -= sum;
                }
            }
        }

        if (success) {
            #pragma omp for schedule(static)
            for (size_t i = 0; i < n; ++i) {
                std::fill(matrix + i * n + i + 1, matrix + (i + 1) * n, 0.0);
            }
        }
    }
    return success;
}

// Generate a symmetric positive definite matrix using the original random
// sequence. Each pair of rows has the same dot product in either order.
void generatePositiveDefiniteMatrix(std::vector<double>& A, const size_t n) {
    std::vector<double> B(n * n);
    unsigned int seed = 42;
    for (size_t i = 0; i < n * n; ++i) {
        B[i] = (rand_r(&seed) / (double)RAND_MAX) - 0.5;
    }

    #pragma omp parallel for schedule(static, 1)
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j <= i; ++j) {
            double sum = 0.0;
            #pragma omp simd reduction(+:sum)
            for (size_t k = 0; k < n; ++k) {
                sum += B[i * n + k] * B[j * n + k];
            }
            if (i == j) sum += n;
            A[i * n + j] = sum;
            if (i != j) A[j * n + i] = sum;
        }
    }
}

bool validateCholesky(const std::vector<double>& L, const std::vector<double>& A_orig, const size_t n) {
    double maxError = 0.0;
    double relError = 0.0;

    // L * L^T is symmetric, but compare both entries of A_orig independently.
    #pragma omp parallel for schedule(static, 1) reduction(max:maxError,relError)
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = 0; j <= i; ++j) {
            double sum = 0.0;
            #pragma omp simd reduction(+:sum)
            for (size_t k = 0; k < n; ++k) {
                sum += L[i * n + k] * L[j * n + k];
            }
            const double error = fabs(sum - A_orig[i * n + j]);
            maxError = std::max(maxError, error);
            relError = std::max(relError, error / (fabs(A_orig[i * n + j]) + 1e-10));
            if (i != j) {
                const double otherError = fabs(sum - A_orig[j * n + i]);
                maxError = std::max(maxError, otherError);
                relError = std::max(relError, otherError / (fabs(A_orig[j * n + i]) + 1e-10));
            }
        }
    }

    printf("Max absolute error: %.10e\n", maxError);
    printf("Max relative error: %.10e\n", relError);
    if (relError > 1e-6) {
        printf("Validation failed: relative error too large\n");
        return false;
    }
    return true;
}
