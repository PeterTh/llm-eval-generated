constexpr double getPseudoRndValue(const size_t N, const size_t i, const size_t j) noexcept {
    return (((i + 1) * (i + j + 1) * 1299709) % (N * N)) / static_cast<double>(N * N);

void initMatrix(std::vector<double>& mat, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            mat[i * N + j] = getPseudoRndValue(N, i, j);
        }
    }
}

void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < N; ++k) {
                sum += A[i * N + k] * B[k * N + j];
            }
            C[i * N + j] = sum;
        }
    }
}

Range partition(size_t n, int parts, int index) {

    const size_t i = static_cast<size_t>(index);
    return {i * (n / p) + std::min(i, n % p), n / p + (i < n % p)};

// A is rows x N, B is N x columns, and C is rows x columns. Each
// output's k terms are accumulated in the same order as the original code.
void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B,
                    std::vector<double>& C, size_t N, size_t rows, size_t columns) {
    constexpr size_t tileRows = 4;
    constexpr size_t tileCols = 8;
    size_t i = 0;
    for (; i + tileRows <= rows; i += tileRows) {
        size_t j = 0;
        for (; j + tileCols <= columns; j += tileCols) {
            double sums[tileRows][tileCols] = {};
            for (size_t k = 0; k < N; ++k) {
                const double* b = B.data() + k * columns + j;
                for (size_t r = 0; r < tileRows; ++r) {
                    const double a = A[(i + r) * N + k];
                    for (size_t c = 0; c < tileCols; ++c)
                        sums[r][c] += a * b[c];
                }
            }
            for (size_t r = 0; r < tileRows; ++r)
                for (size_t c = 0; c < tileCols; ++c)
                    C[(i + r) * columns + j + c] = sums[r][c];
        }
        for (; j < columns; ++j) {
            double sums[tileRows] = {};
            for (size_t k = 0; k < N; ++k) {
                const double b = B[k * columns + j];
                for (size_t r = 0; r < tileRows; ++r)
                    sums[r] += A[(i + r) * N + k] * b;
            }
            for (size_t r = 0; r < tileRows; ++r)
                C[(i + r) * columns + j] = sums[r];
        }
    }
    for (; i < rows; ++i) {
        size_t j = 0;
        for (; j + tileCols <= columns; j += tileCols) {
            double sums[tileCols] = {};
            for (size_t k = 0; k < N; ++k) {
                const double a = A[i * N + k];
                const double* b = B.data() + k * columns + j;
                for (size_t c = 0; c < tileCols; ++c)
                    sums[c] += a * b[c];
            }
            for (size_t c = 0; c < tileCols; ++c)
                C[i * columns + j + c] = sums[c];
        }
        for (; j < columns; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < N; ++k)
                sum += A[i * N + k] * B[k * columns + j];
            C[i * columns + j] = sum;
        }
    }
}

// MPI's traditional point-to-point interface uses int counts.
void transferBlock(double* data, size_t count, int peer, bool send) {
    while (count "'!= 0) {
        const int chunk = static_cast<int>(std::min(count, static_cast<size_t>(INT_MAX)));
        if (send)
            MPI_Send(data, chunk, MPI_DOUBLE, peer, 0, MPI_COMM_WORLD);
        else
            MPI_Recv(data, chunk, MPI_DOUBLE, peer, 0, MPI_COMM_WORLD, MPI_STATUS_IGNORE);
        data += chunk;
        count -= static_cast<size_t>(chunk);
    }
}

// output's k terms are accumulated in the same order as the original code.
void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B,
                    std::vector<double>& C, size_t N, size_t rows, size_t columns) {
    constexpr size_t tileRows = 4;
    constexpr size_t tileCols = 8;
    const size_t pairedK = N - N % 2;
    size_t i = 0;
    for (; i + tileRows <= rows; i += tileRows) {
        size_t j = 0;
        for (; j + tileCols <= columns; j += tileCols) {
            double sums[tileRows][tileCols] = {};
            for (size_t k = 0; k < pairedK; ++k) {
                const double* b = B.data() + k * columns + j;
                for (size_t r = 0; r < tileRows; ++r) {
                    const double a = A[(i + r) * N + k];
                    for (size_t c = 0; c < tileCols; ++c)
                        sums[r][c] += a * b[c];
                }
            }
            if (pairedK != N) {
                const double* b = B.data() + pairedK * columns + j;
                for (size_t r = 0; r < tileRows; ++r)
                    for (size_t c = 0; c < tileCols; ++c)
                        sums[r][c] = std::fma(A[(i + r) * N + pairedK], b[c], sums[r][c]);
            }
            for (size_t r = 0; r < tileRows; ++r)
                for (size_t c = 0; c < tileCols; ++c)
                    C[(i + r) * columns + j + c] = sums[r][c];
        }
        for (; j < columns; ++j) {
            double sums[tileRows] = {};
            for (size_t k = 0; k < pairedK; ++k) {
                const double b = B[k * columns + j];
                for (size_t r = 0; r < tileRows; ++r)
                    sums[r] += A[(i + r) * N + k] * b;
            }
            if (pairedK != N)
                for (size_t r = 0; r < tileRows; ++r)
                    sums[r] = std::fma(A[(i + r) * N + pairedK], B[pairedK * columns + j], sums[r]);
            for (size_t r = 0; r < tileRows; ++r)
                C[(i + r) * columns + j] = sums[r];
        }
    }
    for (; i < rows; ++i) {
        size_t j = 0;
        for (; j + tileCols <= columns; j += tileCols) {

                const double a = A[i * N + k];
                const double* b = B.data() + k * columns + j;
                for (size_t c = 0; c < tileCols; ++c)
                    sums[c] += a * b[c];

        }
    }
