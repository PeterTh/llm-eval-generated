// 3D index calculation
inline constexpr size_t idx3(const size_t x, const size_t y, const size_t z, const size_t nx, const size_t ny) noexcept {
    return z * (nx * ny) + y * nx + x;
}

// Compute Laplacian with clamped boundary conditions
double computeLaplacian(const std::vector<double>& c, const size_t nx, const size_t ny, const size_t nz,
                        const double dx, const double dy, const double dz, const size_t x, const size_t y, const size_t z) {
    const size_t xp = (x < nx - 1) ? x + 1 : x;
    const size_t yp = (y < ny - 1) ? y + 1 : y;
    const size_t zp = (z < nz - 1) ? z + 1 : z;
    const size_t xn = (x > 0) ? x - 1 : 0;
    const size_t yn = (y > 0) ? y - 1 : 0;
    const size_t zn = (z > 0) ? z - 1 : 0;
    
    const double cxx = (c[idx3(xp, y, z, nx, ny)] + c[idx3(xn, y, z, nx, ny)] - 
                  2.0 * c[idx3(x, y, z, nx, ny)]) / (dx * dx);
    const double cyy = (c[idx3(x, yp, z, nx, ny)] + c[idx3(x, yn, z, nx, ny)] - 
                  2.0 * c[idx3(x, y, z, nx, ny)]) / (dy * dy);
    const double czz = (c[idx3(x, y, zp, nx, ny)] + c[idx3(x, y, zn, nx, ny)] - 
                  2.0 * c[idx3(x, y, z, nx, ny)]) / (dz * dz);
    
    return cxx + cyy + czz;
}

// Compute chemical potential
void computeChemicalPotential(const std::vector<double>& c, std::vector<double>& mu,
                              const size_t nx, const size_t ny, const size_t nz,
                              const double dx, const double dy, const double dz,
                              const double gamma, const double e_AA, const double e_BB, const double e_AB) {

            }
        }
    }
}

// Cahn-Hilliard update step
void cahnHilliardUpdate(std::vector<double>& cnew, const std::vector<double>& cold,
                        const std::vector<double>& mu,
                        const size_t nx, const size_t ny, const size_t nz,
                        const double D, const double dt, const double dx, const double dy, const double dz) {
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                cnew[idx] = cold[idx] + dt * D * 
                           computeLaplacian(mu, nx, ny, nz, dx, dy, dz, x, y, z);
            }
        }
    }
}

// Initialize concentration field
void initializeConcentration(std::vector<double>& c, const size_t nx, const size_t ny, const size_t nz) {
    const size_t vol = nx * ny * nz;
    
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                // Generate pseudo-random value in [-1, 1]
                const size_t linear_id = z * (nx * ny) + y * nx + x;
                const double pseudo = ((((linear_id + 1) * 1299709) % vol) / static_cast<double>(vol));
                c[idx] = -1.0 + 2.0 * pseudo;
            }
        }
    }
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// dx = dy = dz = 1 in the benchmark. Keeping the original operation order
// preserves the serial floating-point results, including at clamped boundaries.
static inline double laplacian(const double* a, size_t i, size_t x, size_t y,
                               const Slab& s) {
    const size_t xp = x + (x + 1 < s.nx);
    const size_t xn = x - (x > 0);
    const size_t row = i - x;
    const size_t yp = i + (y + 1 < s.ny ? s.nx : 0);
    const size_t yn = i - (y > 0 ? s.nx : 0);
    const double center = a[i];
    const double cxx = a[row + xp] + a[row + xn] - 2.0 * center;
    const double cyy = a[yp] + a[yn] - 2.0 * center;
    const double czz = a[i + s.plane] + a[i - s.plane] - 2.0 * center;
    return cxx + cyy + czz;
}

static void chemicalPotential(const std::vector<double>& c, std::vector<double>& mu,
                              const Slab& s, size_t begin, size_t end) {
    constexpr double e_AA = -(2.0 / 9.0), e_BB = -(2.0 / 9.0), e_AB = 2.0 / 9.0;
    const double* src = c.data();
    double* dst = mu.data();
    for (size_t z = begin; z < end; ++z)
        for (size_t y = 0; y < s.ny; ++y)
            for (size_t x = 0; x < s.nx; ++x) {
                const size_t i = z * s.plane + y * s.nx + x;
                const double cv = src[i];
                dst[i] = 4.5 * ((cv + 1.0) * e_AA + (cv - 1.0) * e_BB - 2.0 * cv * e_AB)
                       + 3.0 * cv + cv * cv * cv - 0.5 * laplacian(src, i, x, y, s);
            }
}

static void update(std::vector<double>& next, const std::vector<double>& old,
                   const std::vector<double>& mu, const Slab& s, size_t begin, size_t end) {
    const double* src = mu.data();

                const size_t i = z * s.plane + y * s.nx + x;
                dst[i] = c[i] + 0.01 * laplacian(src, i, x, y, s);
            }
}

static void stencil(const std::vector<double>& input, const std::vector<double>& old,
                    std::vector<double>& output, const Slab& s, size_t begin, size_t end) {
    constexpr double e_AA = -(2.0 / 9.0), e_BB = -(2.0 / 9.0), e_AB = 2.0 / 9.0;
    const double* __restrict__ src = input.data();
    const double* __restrict__ c = old.data();
    double* __restrict__ dst = output.data();
    for (size_t z = begin; z < end; ++z) {
        for (size_t y = 0; y < s.ny; ++y) {
            const size_t row = z * s.plane + y * s.nx;
            const size_t yp = row + (y + 1 < s.ny ? s.nx : 0);
            const size_t yn = row - (y > 0 ? s.nx : 0);
            auto cell = [&](size_t x, size_t xp, size_t xn) {
                const size_t i = row + x;
                const double center = src[i];
                const double cxx = src[row + xp] + src[row + xn] - 2.0 * center;
                const double cyy = src[yp + x] + src[yn + x] - 2.0 * center;
                const double czz = src[i + s.plane] + src[i - s.plane] - 2.0 * center;
                const double lap = cxx + cyy + czz;
                if constexpr (Chemical) {
                    dst[i] = 4.5 * ((center + 1.0) * e_AA + (center - 1.0) * e_BB - 2.0 * center * e_AB)

                }
            };
            cell(0, s.nx > 1 ? 1 : 0, 0);
            for (size_t x = 1; x + 1 < s.nx; ++x) cell(x, x + 1, x - 1);

    }
}
