cmake_minimum_required(VERSION 3.18)
# Supply optimized machine code for common cluster GPUs and PTX for newer GPUs.
# A caller can override this with -DCMAKE_CUDA_ARCHITECTURES=...
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(ROOMSIM_DEFAULT_CUDA_ARCHITECTURES ON)
endif()
project(roomsim_benchmark LANGUAGES CXX CUDA)
if(ROOMSIM_DEFAULT_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 75-real)
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_LESS 13.0)
        list(APPEND CMAKE_CUDA_ARCHITECTURES 60-real 70-real)
    endif()
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 11.0)
        list(APPEND CMAKE_CUDA_ARCHITECTURES 80-real)
    endif()
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 11.1)
        list(APPEND CMAKE_CUDA_ARCHITECTURES 86-real 86-virtual)
    elseif(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 11.0)
        list(APPEND CMAKE_CUDA_ARCHITECTURES 80-virtual)
    else()
        list(APPEND CMAKE_CUDA_ARCHITECTURES 75-virtual)
    endif()
    if(CMAKE_CUDA_COMPILER_VERSION VERSION_GREATER_EQUAL 11.8)
        list(APPEND CMAKE_CUDA_ARCHITECTURES 90-real)
    endif()
    set(CMAKE_CUDA_ARCHITECTURES "${CMAKE_CUDA_ARCHITECTURES}" CACHE STRING "CUDA architectures" FORCE)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED COMPONENTS CXX)

set_source_files_properties(roomsim.cpp PROPERTIES LANGUAGE CUDA)
add_executable(roomsim roomsim.cpp)
target_link_libraries(roomsim PRIVATE MPI::MPI_CXX OpenMP::OpenMP_CXX)
target_compile_options(roomsim PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-O3;--fmad=false;-Xcompiler=-fopenmp>)
