cmake_minimum_required(VERSION 3.24)

# Select the local GPU before CMake initializes the CUDA compiler.  Doing this
# afterwards makes CMake retain nvcc's conservative compatibility default.
if (NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES native)
endif()

project(floydwarshall_benchmark LANGUAGES CXX CUDA)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

add_executable(floydwarshall floydwarshall.cpp)
set_source_files_properties(floydwarshall.cpp PROPERTIES LANGUAGE CUDA)

# Enable aggressive optimization for both device and host code generated by nvcc.
target_compile_options(floydwarshall PRIVATE
    $<$<COMPILE_LANGUAGE:CUDA>:-O3>
    $<$<COMPILE_LANGUAGE:CUDA>:-Xcompiler=-O3,-march=native,-Wall,-Wextra>
)

# CUDA is the unconditional implementation path.
find_package(CUDAToolkit REQUIRED)
target_link_libraries(floydwarshall PRIVATE CUDA::cudart m)
