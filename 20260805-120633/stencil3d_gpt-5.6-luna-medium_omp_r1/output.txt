// 3D index calculation
inline constexpr size_t idx3(const size_t x, const size_t y, const size_t z, const size_t nx, const size_t ny) noexcept {
    return z * (nx * ny) + y * nx + x;
}

void initializeGrid(std::vector<Real>& grid, const size_t nx, const size_t ny, const size_t nz) {
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                grid[idx] = (idx % 19) * 1.0;
            }
        }
    }
}

// 7-point stencil computation
void stencilIteration(const std::vector<Real>& input, 
                      std::vector<Real>& output,
                      const size_t nx, const size_t ny, const size_t nz) {
    // Process interior points (not on boundaries)
    for (size_t z = 1; z < nz - 1; ++z) {
        for (size_t y = 1; y < ny - 1; ++y) {
            for (size_t x = 1; x < nx - 1; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                
                const Real center = input[idx];
                const Real left = input[idx3(x-1, y, z, nx, ny)];
                const Real right = input[idx3(x+1, y, z, nx, ny)];
                const Real front = input[idx3(x, y-1, z, nx, ny)];
                const Real back = input[idx3(x, y+1, z, nx, ny)];
                const Real bottom = input[idx3(x, y, z-1, nx, ny)];
                const Real top = input[idx3(x, y, z+1, nx, ny)];
                
                // Simple averaging stencil
                output[idx] = (center + left + right + front + back + bottom + top) / 7.0;
            }
        }
    }
    
    // Copy boundary values
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                if (x == 0 || x == nx-1 || y == 0 || y == ny-1 || z == 0 || z == nz-1) {
                    const size_t idx = idx3(x, y, z, nx, ny);
                    output[idx] = input[idx];
                }
            }
        }
    }
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

    35	// 7-point stencil computation
    36	void stencilIteration(const std::vector<Real>& input, 
    37	                      std::vector<Real>& output,
    38	                      const size_t nx, const size_t ny, const size_t nz) {
    39	    const size_t planeSize = nx * ny;
    40	
    41	    // Process interior points (not on boundaries)
    42	    #pragma omp parallel for collapse(2) schedule(static)
    43	    for (size_t z = 1; z < nz - 1; ++z) {
    44	        for (size_t y = 1; y < ny - 1; ++y) {
    45	            const size_t rowStart = z * planeSize + y * nx;
    46	            const size_t previousRow = rowStart - nx;
    47	            const size_t nextRow = rowStart + nx;
    48	            const size_t previousPlane = rowStart - planeSize;
    49	            const size_t nextPlane = rowStart + planeSize;
    50	
    51	            #pragma omp simd
    52	            for (size_t x = 1; x < nx - 1; ++x) {
    53	                const size_t index = rowStart + x;
    54	                
    55	                // Simple averaging stencil
    56	                output[index] = (input[index] + input[index - 1] + input[index + 1]
    57	                    + input[previousRow + x] + input[nextRow + x]
    58	                    + input[previousPlane + x] + input[nextPlane + x]) / 7.0;
    59	            }
    60	        }
    61	    }
    62	    
    63	    // Copy boundary values
    64	    // A single parallel loop gives every output element one writer, including
    65	    // edges and corners, while keeping this infrequently executed work simple.
    66	    #pragma omp parallel for collapse(2) schedule(static)
    67	    for (size_t z = 0; z < nz; ++z) {
    68	        for (size_t y = 0; y < ny; ++y) {
    69	            const size_t rowStart = z * planeSize + y * nx;
    70	            #pragma omp simd
    71	            for (size_t x = 0; x < nx; ++x) {
    72	                if (x == 0 || x + 1 == nx || y == 0 || y + 1 == ny || z == 0 || z + 1 == nz) {
    73	                    output[rowStart + x] = input[rowStart + x];
    74	                }
    75	            }
    76	        }
    77	    }
    78	}
