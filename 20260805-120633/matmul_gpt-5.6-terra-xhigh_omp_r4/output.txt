// Generate pseudo-random values for matrix initialization
constexpr double getPseudoRndValue(const size_t N, const size_t i, const size_t j) noexcept {
    return (((i + 1) * (i + j + 1) * 1299709) % (N * N)) / static_cast<double>(N * N);
}

void initMatrix(std::vector<double>& mat, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            mat[i * N + j] = getPseudoRndValue(N, i, j);
        }
    }
}

void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < N; ++k) {
                sum += A[i * N + k] * B[k * N + j];
            }
            C[i * N + j] = sum;
        }
    }
}

// Simple validation: compute a single element and compare
bool validateResult(const std::vector<double>& A, const std::vector<double>& B,
                   const std::vector<double>& C, const size_t N) {
    // Check a few random positions
    constexpr size_t checkPoints[] = {0, 1, 2, 3, 4};
    
    for (size_t pi = 0; pi < 5; ++pi) {
        for (size_t pj = 0; pj < 5; ++pj) {
            const size_t i = checkPoints[pi] % N;
            const size_t j = checkPoints[pj] % N;
            
            double expected = 0.0;
            for (size_t k = 0; k < N; ++k) {
                expected += A[i * N + k] * B[k * N + j];
            }
            
            const double actual = C[i * N + j];
            const double relError = std::abs((actual - expected) / (expected + 1e-10));
            
            if (relError > 1e-6) {
                printf("Validation failed at (%zu, %zu): expected %.10f, got %.10f (error: %.10e)\n",
                       i, j, expected, actual, relError);
                return false;
            }
        }
    }
    
    return true;
}

// Generate pseudo-random values for matrix initialization
constexpr double getPseudoRndValue(const size_t N, const size_t i, const size_t j) noexcept {
    return (((i + 1) * (i + j + 1) * 1299709) % (N * N)) / static_cast<double>(N * N);
}

void initMatrix(std::vector<double>& mat, const size_t N) {
    #pragma omp parallel for schedule(static)
    for (std::ptrdiff_t row = 0; row < static_cast<std::ptrdiff_t>(N); ++row) {
        const size_t i = static_cast<size_t>(row);
        for (size_t j = 0; j < N; ++j) {
            mat[i * N + j] = getPseudoRndValue(N, i, j);
        }
    }
}

void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N) {
    // A 32 x 32 tile of doubles is 8 KiB.  Keeping a C tile in cache while
    // streaming rows of B avoids the strided access in the original i/j/k loop.
    constexpr size_t blockSize = 32;
    const double* const a = A.data();
    const double* const b = B.data();
    double* const c = C.data();

    // Each iteration owns one C tile, so no synchronization is needed.  The
    // k blocks are traversed in increasing order to retain the original
    // accumulation order for every output element.
    #pragma omp parallel for collapse(2) schedule(static)
    for (std::ptrdiff_t rowBlock = 0;
         rowBlock < static_cast<std::ptrdiff_t>(N);
         rowBlock += static_cast<std::ptrdiff_t>(blockSize)) {
        for (std::ptrdiff_t columnBlock = 0;
             columnBlock < static_cast<std::ptrdiff_t>(N);
             columnBlock += static_cast<std::ptrdiff_t>(blockSize)) {
            const size_t ii = static_cast<size_t>(rowBlock);
            const size_t jj = static_cast<size_t>(columnBlock);
            const size_t iEnd = std::min(ii + blockSize, N);
            const size_t jEnd = std::min(jj + blockSize, N);
            const size_t tileWidth = jEnd - jj;

            for (size_t i = ii; i < iEnd; ++i) {
                double* const cRow = c + i * N + jj;

                for (size_t j = 0; j < tileWidth; ++j) {
                    cRow[j] = 0.0;
                }

                for (size_t kk = 0; kk < N; kk += blockSize) {
                    const size_t kEnd = std::min(kk + blockSize, N);

                    for (size_t k = kk; k < kEnd; ++k) {
                        const double aValue = a[i * N + k];
                        const double* const bRow = b + k * N + jj;

                        #pragma omp simd
                        for (size_t j = 0; j < tileWidth; ++j) {
                            cRow[j] += aValue * bRow[j];
                        }
                    }
                }
            }
        }
    }
}

bool validateResult(const std::vector<double>& A, const std::vector<double>& B,

    constexpr size_t checkPoints[] = {0, 1, 2, 3, 4};
    
    for (size_t pi = 0; pi < 5; ++pi) {
        for (size_t pj = 0; pj < 5; ++pj) {
            const size_t i = checkPoints[pi] % N;
            const size_t j = checkPoints[pj] % N;
            
            double expected = 0.0;
            for (size_t k = 0; k < N; ++k) {
                expected += A[i * N + k] * B[k * N + j];
            }
            
            const double actual = C[i * N + j];
            const double relError = std::abs((actual - expected) / (expected + 1e-10));

                printf("Validation failed at (%zu, %zu): expected %.10f, got %.10f (error: %.10e)\n",

                return false;
            }
        }
    }
