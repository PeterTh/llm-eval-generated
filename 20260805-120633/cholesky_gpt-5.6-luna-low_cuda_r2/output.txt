 // Simple Cholesky decomposition (sequential, unblocked algorithm)
 // Decomposes positive definite matrix A into L * L^T where L is lower triangular
 
+__global__ void cholKernel(double *a, int n, int k) {
+    int i = k + blockIdx.x * blockDim.x + threadIdx.x;
+    if (i >= n) return;
+    for (int j = k; j < n; ++j) {
+        if (j > i) break;
+        double sum = 0.0;
+        for (int q = 0; q < j; ++q) sum += a[i*n+q] * a[j*n+q];
+        if (i == j) a[i*n+i] = sqrt(a[i*n+i] - sum);
+        else a[i*n+j] = (a[i*n+j] - sum) / a[j*n+j];
+    }
+    for (int j=i+1; j<n; ++j) a[i*n+j]=0.0;
+}
+
 bool choleskyDecomposition(std::vector<double>& A, const size_t n) {
+    double *d=nullptr;
+    if (cudaMalloc(&d,n*n*sizeof(double)) != cudaSuccess) return false;
+    if (cudaMemcpy(d,A.data(),n*n*sizeof(double),cudaMemcpyHostToDevice) != cudaSuccess) { cudaFree(d); return false; }
+    // Each launch advances the dependency frontier by one column; rows within
+    // that frontier execute concurrently on the GPU.
+    for (int k=0; k<(int)n; ++k) {
+        cholKernel<<<(n-k+255)/256,256>>>(d,(int)n,k);

+        if (cudaDeviceSynchronize()!=cudaSuccess) { cudaFree(d); return false; }
+    }
+    bool ok = cudaMemcpy(A.data(),d,n*n*sizeof(double),cudaMemcpyDeviceToHost)==cudaSuccess;
+    cudaFree(d); return ok;
+}
