// Generate pseudo-random values for matrix initialization
constexpr double getPseudoRndValue(const size_t N, const size_t i, const size_t j) noexcept {
    return (((i + 1) * (i + j + 1) * 1299709) % (N * N)) / static_cast<double>(N * N);
}

void initMatrix(std::vector<double>& mat, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            mat[i * N + j] = getPseudoRndValue(N, i, j);
        }
    }
}

void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N) {
    for (size_t i = 0; i < N; ++i) {
        for (size_t j = 0; j < N; ++j) {
            double sum = 0.0;
            for (size_t k = 0; k < N; ++k) {
                sum += A[i * N + k] * B[k * N + j];
            }
            C[i * N + j] = sum;
        }
    }
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

Implemented MPI distributed-memory parallelization in `matmul/`.

- `A` and `C` are balanced contiguous row partitions across MPI ranks.
- `B` is deterministically generated on every rank, avoiding communication overhead.
- Multiplication uses cache-friendly blocked `i-k-j` loops.
- Timing reports the slowest rank’s compute time; output/validation occur on rank 0.
- Results are gathered only for `-v` or `-r`.
- CMake now requires and links MPI while retaining the `matmul` executable name.

Verified with MPI runs including 3 ranks (`N=17`) and more ranks than rows (4 ranks, `N=2`); validation passed.
