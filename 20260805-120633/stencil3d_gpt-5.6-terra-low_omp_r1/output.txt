// 3D index calculation
inline constexpr size_t idx3(const size_t x, const size_t y, const size_t z, const size_t nx, const size_t ny) noexcept {
    return z * (nx * ny) + y * nx + x;
}

void initializeGrid(std::vector<Real>& grid, const size_t nx, const size_t ny, const size_t nz) {
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                grid[idx] = (idx % 19) * 1.0;
            }
        }
    }
}

// 7-point stencil computation
void stencilIteration(const std::vector<Real>& input, 
                      std::vector<Real>& output,
                      const size_t nx, const size_t ny, const size_t nz) {
    // Process interior points (not on boundaries)
    for (size_t z = 1; z < nz - 1; ++z) {
        for (size_t y = 1; y < ny - 1; ++y) {
            for (size_t x = 1; x < nx - 1; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                
                const Real center = input[idx];
                const Real left = input[idx3(x-1, y, z, nx, ny)];
                const Real right = input[idx3(x+1, y, z, nx, ny)];
                const Real front = input[idx3(x, y-1, z, nx, ny)];
                const Real back = input[idx3(x, y+1, z, nx, ny)];
                const Real bottom = input[idx3(x, y, z-1, nx, ny)];
                const Real top = input[idx3(x, y, z+1, nx, ny)];
                
                // Simple averaging stencil
                output[idx] = (center + left + right + front + back + bottom + top) / 7.0;
            }
        }
    }
    
    // Copy boundary values
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                if (x == 0 || x == nx-1 || y == 0 || y == ny-1 || z == 0 || z == nz-1) {
                    const size_t idx = idx3(x, y, z, nx, ny);
                    output[idx] = input[idx];
                }
            }
        }
    }
}

        }
    }
    
    // 2. Values should be reasonable (bounded)

    }
    
    printf("Value range: [%.6f, %.6f]\n", minVal, maxVal);
    
    // After averaging, values should be somewhat bounded
    if (maxVal > 1e6 || minVal < -1e6) {
        printf("Validation failed: values out of expected range\n");
        return false;
    }
    
    // 3. Boundary values should not change significantly
    // (they are copied, so they should be close to initial values)
    
    return true;
}

// 7-point stencil computation
void stencilIteration(const std::vector<Real>& input, 
                      std::vector<Real>& output,
                      const size_t nx, const size_t ny, const size_t nz) {
    const size_t planeSize = nx * ny;

    // Process interior points (not on boundaries)
    #pragma omp parallel for collapse(2) schedule(static)
    for (size_t z = 1; z < nz - 1; ++z) {
        for (size_t y = 1; y < ny - 1; ++y) {
            const size_t rowStart = z * planeSize + y * nx;
            for (size_t x = 1; x < nx - 1; ++x) {
                const size_t idx = rowStart + x;

                // Simple averaging stencil.  The X dimension is contiguous,
                // so each worker processes complete, cache-friendly rows.
                output[idx] = (input[idx] + input[idx - 1] + input[idx + 1]
                    + input[idx - nx] + input[idx + nx]
                    + input[idx - planeSize] + input[idx + planeSize]) / 7.0;
            }
        }
    }
    
    // Copy boundary values
    // Write non-overlapping faces directly.  This avoids a second pass over
    // every interior cell just to test whether it is a boundary cell.
    #pragma omp parallel
    {
        #pragma omp for schedule(static)
        for (size_t idx = 0; idx < planeSize; ++idx) {
            output[idx] = input[idx];
            output[(nz - 1) * planeSize + idx] = input[(nz - 1) * planeSize + idx];
        }

        #pragma omp for schedule(static)
        for (size_t z = 1; z < nz - 1; ++z) {
            const size_t planeStart = z * planeSize;
            const size_t firstRow = planeStart;
            const size_t lastRow = planeStart + (ny - 1) * nx;
            for (size_t x = 0; x < nx; ++x) {
                output[firstRow + x] = input[firstRow + x];
                output[lastRow + x] = input[lastRow + x];
            }
        }

        #pragma omp for collapse(2) schedule(static)
        for (size_t z = 1; z < nz - 1; ++z) {
            for (size_t y = 1; y < ny - 1; ++y) {
                const size_t rowStart = z * planeSize + y * nx;
                output[rowStart] = input[rowStart];
                output[rowStart + nx - 1] = input[rowStart + nx - 1];
            }
        }
    }
}

bool validateResult(const std::vector<Real>& grid, [[maybe_unused]] const size_t nx, [[maybe_unused]] const size_t ny, [[maybe_unused]] const size_t nz) {
    // Simple sanity checks
    
    // 1. No NaN or Inf values
    Real minVal = grid[0];
    Real maxVal = grid[0];
    bool finite = true;
    #pragma omp parallel for reduction(min:minVal) reduction(max:maxVal) reduction(&:finite) schedule(static)
    for (size_t i = 0; i < grid.size(); ++i) {
        const Real val = grid[i];
        finite = finite && !std::isnan(val) && !std::isinf(val);
        minVal = std::min(minVal, val);
        maxVal = std::max(maxVal, val);
    }

    if (!finite) {
        printf("Validation failed: found NaN or Inf value\n");
        return false;
    }
    
    // 2. Values should be reasonable (bounded)
    
    printf("Value range: [%.6f, %.6f]\n", minVal, maxVal);
    
    // After averaging, values should be somewhat bounded

        printf("Validation failed: values out of expected range\n");
        return false;
    }
    
    // 3. Boundary values should not change significantly

    return true;
