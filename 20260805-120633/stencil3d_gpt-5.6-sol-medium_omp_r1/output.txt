// 3D index calculation
inline constexpr size_t idx3(const size_t x, const size_t y, const size_t z, const size_t nx, const size_t ny) noexcept {
    return z * (nx * ny) + y * nx + x;
}

void initializeGrid(std::vector<Real>& grid, const size_t nx, const size_t ny, const size_t nz) {
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                grid[idx] = (idx % 19) * 1.0;
            }
        }
    }
}

// 7-point stencil computation
void stencilIteration(const std::vector<Real>& input, 
                      std::vector<Real>& output,
                      const size_t nx, const size_t ny, const size_t nz) {
    // Process interior points (not on boundaries)
    for (size_t z = 1; z < nz - 1; ++z) {
        for (size_t y = 1; y < ny - 1; ++y) {
            for (size_t x = 1; x < nx - 1; ++x) {
                const size_t idx = idx3(x, y, z, nx, ny);
                
                const Real center = input[idx];
                const Real left = input[idx3(x-1, y, z, nx, ny)];
                const Real right = input[idx3(x+1, y, z, nx, ny)];
                const Real front = input[idx3(x, y-1, z, nx, ny)];
                const Real back = input[idx3(x, y+1, z, nx, ny)];
                const Real bottom = input[idx3(x, y, z-1, nx, ny)];
                const Real top = input[idx3(x, y, z+1, nx, ny)];
                
                // Simple averaging stencil
                output[idx] = (center + left + right + front + back + bottom + top) / 7.0;
            }
        }
    }
    
    // Copy boundary values
    for (size_t z = 0; z < nz; ++z) {
        for (size_t y = 0; y < ny; ++y) {
            for (size_t x = 0; x < nx; ++x) {
                if (x == 0 || x == nx-1 || y == 0 || y == ny-1 || z == 0 || z == nz-1) {
                    const size_t idx = idx3(x, y, z, nx, ny);
                    output[idx] = input[idx];
                }
            }
        }
    }
}

void initializeGrids(std::vector<Real>& grid1, std::vector<Real>& grid2) {
    const size_t gridSize = grid1.size();

    // Initializing both buffers once makes their fixed boundary cells identical.
    // The stencil therefore does not need to recopy the six faces every step.
#pragma omp parallel for schedule(static)
    for (size_t i = 0; i < gridSize; ++i) {
        const Real value = static_cast<Real>(i % 19);
        grid1[i] = value;
        grid2[i] = value;
    }
}

// Run all time steps inside one persistent team.  Keeping x as the innermost
// dimension gives unit-stride accesses, while collapsing z/y exposes enough
// rows to scale even when the grid has relatively few planes.
void runStencil(std::vector<Real>& grid1, std::vector<Real>& grid2,
                const size_t nx, const size_t ny, const size_t nz,
                const int iterations) {
    const size_t plane = nx * ny;
    Real* input = grid1.data();
    Real* output = grid2.data();

#pragma omp parallel shared(input, output)
    {
        for (int iter = 0; iter < iterations; ++iter) {
            const Real* __restrict in = input;
            Real* __restrict out = output;

#pragma omp for collapse(2) schedule(static)
            for (size_t z = 1; z < nz - 1; ++z) {
                for (size_t y = 1; y < ny - 1; ++y) {
                    const size_t row = z * plane + y * nx;
#pragma omp simd
                    for (size_t x = 1; x < nx - 1; ++x) {
                        const size_t i = row + x;
                        out[i] = (in[i] + in[i - 1] + in[i + 1]
                                  + in[i - nx] + in[i + nx]
                                  + in[i - plane] + in[i + plane]) / 7.0;
                    }
                }
            }

            // The omp-for barrier completes the current time step.  The
            // single-region barrier then publishes the swapped buffers before
            // every thread starts the next one.
#pragma omp single
            {
                std::swap(input, output);
            }
        }
    }
}
