// Standard normal cumulative distribution function
inline double cumulativeNormal(const double x) noexcept {
    return 0.5 * (1.0 + std::erf(x * M_SQRT1_2));
}

// Standard normal probability density function
inline double normalPDF(const double x) noexcept {
    return M_1_SQRTPI * M_SQRT1_2 * exp(-0.5 * x * x);
}

// Black-Scholes formula for European options
double blackScholes(const OptionInput& option) noexcept {
    const double S = option.spot;
    const double K = option.strike;
    const double r = option.r;
    const double q = option.q;
    const double T = option.t;
    const double sigma = option.vol;
    
    if (T <= 0.0 || sigma <= 0.0) {
        return 0.0;
    }
    
    const double d1 = (log(S / K) + (r - q + 0.5 * sigma * sigma) * T) / (sigma * sqrt(T));
    const double d2 = d1 - sigma * sqrt(T);
    
    const double Nd1 = cumulativeNormal(d1);
    const double Nd2 = cumulativeNormal(d2);
    const double discount = exp(-r * T);
    
    double price;
    if (option.type == CALL) {
        price = S * exp(-q * T) * Nd1 - K * discount * Nd2;
    } else { // PUT
        price = K * discount * cumulativeNormal(-d2) - S * exp(-q * T) * cumulativeNormal(-d1);
    }
    
    return price;
}

// Generate a larger set of options by scaling the test set
void generateOptions(std::vector<OptionInput>& options, const size_t numOptions) {
    constexpr auto testOptions = getTestOptions();
    options.resize(numOptions);
    
    for (size_t i = 0; i < numOptions; ++i) {
        // Cycle through test options and vary parameters slightly
        const OptionInput& base = testOptions[i % testOptions.size()];
        options[i] = base;
        
        // Add some variation for larger datasets
        const double factor = 1.0 + 0.1 * (i / static_cast<double>(testOptions.size()));
        options[i].spot *= factor;
        options[i].strike *= factor;
    }
}

class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// Standard normal cumulative distribution function
inline double cumulativeNormal(const double x) noexcept {
    return 0.5 * (1.0 + std::erf(x * M_SQRT1_2));
}

// Black-Scholes formula for European options
double blackScholes(const OptionInput& option) noexcept {
    const double S = option.spot;
    const double K = option.strike;
    const double r = option.r;
    const double q = option.q;
    const double T = option.t;
    const double sigma = option.vol;
    
    if (T <= 0.0 || sigma <= 0.0) {
        return 0.0;
    }
    
    const double d1 = (log(S / K) + (r - q + 0.5 * sigma * sigma) * T) / (sigma * sqrt(T));
    const double d2 = d1 - sigma * sqrt(T);
    
    const double Nd1 = cumulativeNormal(d1);
    const double Nd2 = cumulativeNormal(d2);
    const double discount = exp(-r * T);
    
    double price;
    if (option.type == CALL) {
        price = S * exp(-q * T) * Nd1 - K * discount * Nd2;
    } else { // PUT
        price = K * discount * cumulativeNormal(-d2) - S * exp(-q * T) * cumulativeNormal(-d1);
    }
    
    return price;
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// Standard normal cumulative distribution function
__host__ __device__ inline double cumulativeNormal(const double x) noexcept {
    return 0.5 * (1.0 + erf(x * M_SQRT1_2));
}

// Standard normal probability density function
inline double normalPDF(const double x) noexcept {
    return M_1_SQRTPI * M_SQRT1_2 * exp(-0.5 * x * x);
}

// Black-Scholes formula for European options
__host__ __device__ double blackScholes(const OptionInput& option) noexcept {
    const double S = option.spot;
    const double K = option.strike;
    const double r = option.r;
    const double q = option.q;
    const double T = option.t;
    const double sigma = option.vol;
    
    if (T <= 0.0 || sigma <= 0.0) {
        return 0.0;
    }
    
    const double d1 = (log(S / K) + (r - q + 0.5 * sigma * sigma) * T) / (sigma * sqrt(T));
    const double d2 = d1 - sigma * sqrt(T);
    
    const double Nd1 = cumulativeNormal(d1);
    const double Nd2 = cumulativeNormal(d2);
    const double discount = exp(-r * T);
    
    double price;
    if (option.type == CALL) {
        price = S * exp(-q * T) * Nd1 - K * discount * Nd2;
    } else { // PUT
        price = K * discount * cumulativeNormal(-d2) - S * exp(-q * T) * cumulativeNormal(-d1);
    }
    
    return price;
}

    printf("Validation: %s\n", validate ? "enabled" : "disabled");
    
    // Generate options
    std::vector<OptionInput> options;
    generateOptions(options, numOptions);
    
    // Allocate results
    std::vector<double> results(numOptions);
    
    // Stage data once. Transfers and allocation are intentionally outside the
    // computation timer, matching the original benchmark's pricing-only timer.
    OptionInput* deviceOptions = nullptr;
    double* deviceResults = nullptr;
    if (numOptions != 0) {
        checkCuda(cudaMalloc(&deviceOptions, numOptions * sizeof(OptionInput)),
                  "allocating device options");
        checkCuda(cudaMalloc(&deviceResults, numOptions * sizeof(double)),
                  "allocating device results");
        checkCuda(cudaMemcpy(deviceOptions, options.data(),
                             numOptions * sizeof(OptionInput), cudaMemcpyHostToDevice),
                  "copying options to device");
    }

    // Price options
    printf("Pricing options...\n");
    float elapsedMs = 0.0f;
    if (numOptions != 0) {
        cudaEvent_t start;
        cudaEvent_t end;
        checkCuda(cudaEventCreate(&start), "creating start event");
        checkCuda(cudaEventCreate(&end), "creating end event");

        constexpr unsigned threadsPerBlock = 256;
        int device = 0;
        cudaDeviceProp properties{};
        checkCuda(cudaGetDevice(&device), "getting CUDA device");
        checkCuda(cudaGetDeviceProperties(&properties, device), "querying CUDA device");
        const size_t neededBlocks =
            (numOptions + threadsPerBlock - 1) / threadsPerBlock;
        // Enough resident work to saturate the GPU without excessive launch
        // overhead; the grid-stride loop handles the remainder.
        const size_t saturationBlocks =
            static_cast<size_t>(properties.multiProcessorCount) * 32;
        const unsigned blocks = static_cast<unsigned>(
            std::min(neededBlocks, saturationBlocks));

        checkCuda(cudaEventRecord(start), "recording start event");
        blackScholesKernel<<<blocks, threadsPerBlock>>>(deviceOptions, deviceResults,
                                                       numOptions);
        checkCuda(cudaGetLastError(), "launching Black-Scholes kernel");
        checkCuda(cudaEventRecord(end), "recording end event");
        checkCuda(cudaEventSynchronize(end), "waiting for Black-Scholes kernel");
        checkCuda(cudaEventElapsedTime(&elapsedMs, start, end), "timing kernel");

        checkCuda(cudaMemcpy(results.data(), deviceResults,
                             numOptions * sizeof(double), cudaMemcpyDeviceToHost),
                  "copying results to host");
        checkCuda(cudaEventDestroy(start), "destroying start event");
        checkCuda(cudaEventDestroy(end), "destroying end event");
        checkCuda(cudaFree(deviceOptions), "freeing device options");
        checkCuda(cudaFree(deviceResults), "freeing device results");
    }

    printf("Computation time: %.3f ms\n", static_cast<double>(elapsedMs));
    printf("Options per second: %.0f\n",
           elapsedMs > 0.0f ? numOptions * 1000.0 / elapsedMs : 0.0);
    
    // Print results for external validation
    if (printResults) {
        print_results(results, "OptionPrices");
    }
    
    // Validation
    if (validate) {
        printf("Validating results...\n");
        bool valid = validateResults(options, results);
        
        if (valid) {
            printf("Validation: PASSED\n");
            return 0;
        } else {
            printf("Validation: FAILED\n");
            return 1;
        }
    }
    
    return 0;
