void randomizeBodies(std::vector<Body>& bodies, unsigned int seed = 42) {
    for (auto& body : bodies) {
        body.pos.x = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.pos.y = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.pos.z = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.vel.x = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.vel.y = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.vel.z = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;

}

void computeForces(std::vector<Body>& bodies) {
    const size_t n = bodies.size();
    
    for (size_t i = 0; i < n; ++i) {
        double Fx = 0.0, Fy = 0.0, Fz = 0.0;
        
        for (size_t j = 0; j < n; ++j) {
            const double dx = bodies[j].pos.x - bodies[i].pos.x;
            const double dy = bodies[j].pos.y - bodies[i].pos.y;
            const double dz = bodies[j].pos.z - bodies[i].pos.z;
            const double distSqr = dx * dx + dy * dy + dz * dz + SOFTENING;
            const double invDist = 1.0 / std::sqrt(distSqr);
            const double invDist3 = invDist * invDist * invDist;
            
            Fx += dx * invDist3;
            Fy += dy * invDist3;
            Fz += dz * invDist3;
        }
        
        bodies[i].vel.x += DT * Fx;
        bodies[i].vel.y += DT * Fy;
        bodies[i].vel.z += DT * Fz;
    }
}

void integrateBodies(std::vector<Body>& bodies) {
    for (auto& body : bodies) {
        body.pos.x += body.vel.x * DT;
        body.pos.y += body.vel.y * DT;
        body.pos.z += body.vel.z * DT;
    }
}

double computeTotalEnergy(const std::vector<Body>& bodies) {
    double energy = 0.0;
    const size_t n = bodies.size();
    
    // Kinetic energy (assuming unit mass)
    for (const auto& body : bodies) {
        energy += 0.5 * (body.vel.x * body.vel.x + 
                        body.vel.y * body.vel.y + 
                        body.vel.z * body.vel.z);
    }
    
    // Potential energy (assuming unit mass for all bodies)
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = i + 1; j < n; ++j) {
            const double dx = bodies[j].pos.x - bodies[i].pos.x;
            const double dy = bodies[j].pos.y - bodies[i].pos.y;
            const double dz = bodies[j].pos.z - bodies[i].pos.z;
            const double dist = std::sqrt(dx * dx + dy * dy + dz * dz + SOFTENING);
            energy -= 1.0 / dist;
        }
    }
    
    return energy;
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

constexpr double DT = 0.01;

struct Vec3 {
    double x, y, z;
    constexpr Vec3(const double x = 0, const double y = 0, const double z = 0) noexcept : x(x), y(y), z(z) {}
};

    Vec3 pos;
    Vec3 vel;

void randomizeBodies(std::vector<Body>& bodies, unsigned int seed = 42) {
    for (auto& body : bodies) {
        body.pos.x = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.pos.y = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.pos.z = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.vel.x = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.vel.y = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
        body.vel.z = 2.0 * (rand_r(&seed) / (double)RAND_MAX) - 1.0;
    }
}

// Each thread owns one body. Tiles reuse source positions across the block;
// separate position buffers make force evaluation and integration safe to fuse.
constexpr int TILE_SIZE = 32;

void checkCuda(cudaError_t result, const char* operation) {
    if (result != cudaSuccess) {
        fprintf(stderr, "CUDA error in %s: %s\n", operation, cudaGetErrorString(result));
        std::exit(EXIT_FAILURE);
    }
}

__global__ void advanceBodies(const double4* __restrict__ positions,
                              double4* __restrict__ nextPositions,
                              double4* __restrict__ velocities, int n) {
    __shared__ double sourceX[TILE_SIZE];
    __shared__ double sourceY[TILE_SIZE];
    __shared__ double sourceZ[TILE_SIZE];
    const int lane = threadIdx.x;
    const size_t i = static_cast<size_t>(blockIdx.x) * TILE_SIZE + lane;
    const bool active = i < static_cast<size_t>(n);
    const double4 pos = active ? positions[i] : make_double4(0, 0, 0, 0);
    double fx = 0.0, fy = 0.0, fz = 0.0;

    for (size_t base = 0; base < static_cast<size_t>(n); base += TILE_SIZE) {
        const size_t j = base + lane;
        if (j < static_cast<size_t>(n)) {
            const double4 source = positions[j];
            sourceX[lane] = source.x;
            sourceY[lane] = source.y;
            sourceZ[lane] = source.z;
        }
        __syncthreads();
        const int count = static_cast<int>(static_cast<size_t>(n) - base < TILE_SIZE
            ? static_cast<size_t>(n) - base : TILE_SIZE);
        if (active) {
            #pragma unroll 8
            for (int k = 0; k < count; ++k) {
                const double dx = sourceX[k] - pos.x;
                const double dy = sourceY[k] - pos.y;
                const double dz = sourceZ[k] - pos.z;
                const double distSqr = dx * dx + dy * dy + dz * dz + SOFTENING;
                const double invDist = 1.0 / sqrt(distSqr);
                const double invDist3 = invDist * invDist * invDist;
                fx += dx * invDist3;
                fy += dy * invDist3;
                fz += dz * invDist3;
            }
        }
        // All threads, including those past n in the last block, participate.
        __syncthreads();
    }
    if (active) {
        double4 vel = velocities[i];
        vel.x += DT * fx;
        vel.y += DT * fy;
        vel.z += DT * fz;
        velocities[i] = vel;
        nextPositions[i] = make_double4(pos.x + vel.x * DT,
                                        pos.y + vel.y * DT,
                                        pos.z + vel.z * DT, 0.0);
    }
}

class GpuSimulation {
    double4* positions = nullptr;
    double4* nextPositions = nullptr;
    double4* velocities = nullptr;
    int n;

public:
    explicit GpuSimulation(const std::vector<Body>& bodies)
        : n(static_cast<int>(bodies.size())) {
        // Initialize CUDA even for an empty simulation; there is no CPU fallback.
        checkCuda(cudaFree(nullptr), "initialize CUDA");
        if (n == 0) return;
        cudaFuncAttributes attributes;
        checkCuda(cudaFuncGetAttributes(&attributes, advanceBodies), "load timestep kernel");
        const size_t bytes = static_cast<size_t>(n) * sizeof(double4);
        checkCuda(cudaMalloc(&positions, bytes), "allocate positions");
        checkCuda(cudaMalloc(&nextPositions, bytes), "allocate next positions");
        checkCuda(cudaMalloc(&velocities, bytes), "allocate velocities");
        std::vector<double4> hostPositions(n), hostVelocities(n);
        for (int i = 0; i < n; ++i) {
            const auto& body = bodies[i];
            hostPositions[i] = make_double4(body.pos.x, body.pos.y, body.pos.z, 0.0);
            hostVelocities[i] = make_double4(body.vel.x, body.vel.y, body.vel.z, 0.0);
        }
        checkCuda(cudaMemcpy(positions, hostPositions.data(), bytes, cudaMemcpyHostToDevice),
                  "upload positions");
        checkCuda(cudaMemcpy(velocities, hostVelocities.data(), bytes, cudaMemcpyHostToDevice),
                  "upload velocities");
    }

    GpuSimulation(const GpuSimulation&) = delete;
    GpuSimulation& operator=(const GpuSimulation&) = delete;

    ~GpuSimulation() {
        cudaFree(positions);
        cudaFree(nextPositions);
        cudaFree(velocities);
    }

    void run(int steps) {
        if (n == 0 || steps <= 0) return;
        const unsigned int blocks = (static_cast<size_t>(n) + TILE_SIZE - 1) / TILE_SIZE;
        for (int step = 0; step < steps; ++step) {
            advanceBodies<<<blocks, TILE_SIZE>>>(positions, nextPositions, velocities, n);
            checkCuda(cudaGetLastError(), "launch timestep");
            std::swap(positions, nextPositions);
        }
        checkCuda(cudaDeviceSynchronize(), "complete simulation");
    }

    void download(std::vector<Body>& bodies) const {
        if (n == 0) return;
        const size_t bytes = static_cast<size_t>(n) * sizeof(double4);
        std::vector<double4> hostPositions(n), hostVelocities(n);
        checkCuda(cudaMemcpy(hostPositions.data(), positions, bytes, cudaMemcpyDeviceToHost),
                  "download positions");
        checkCuda(cudaMemcpy(hostVelocities.data(), velocities, bytes, cudaMemcpyDeviceToHost),
                  "download velocities");
        for (int i = 0; i < n; ++i) {
            bodies[i].pos = Vec3(hostPositions[i].x, hostPositions[i].y, hostPositions[i].z);
            bodies[i].vel = Vec3(hostVelocities[i].x, hostVelocities[i].y, hostVelocities[i].z);
        }
    }
};

double computeTotalEnergy(const std::vector<Body>& bodies) {
    double energy = 0.0;
    const size_t n = bodies.size();
    
    // Kinetic energy (assuming unit mass)
    for (const auto& body : bodies) {
        energy += 0.5 * (body.vel.x * body.vel.x + 
                        body.vel.y * body.vel.y + 
                        body.vel.z * body.vel.z);
    }
    
    // Potential energy (assuming unit mass for all bodies)
    for (size_t i = 0; i < n; ++i) {
        for (size_t j = i + 1; j < n; ++j) {
            const double dx = bodies[j].pos.x - bodies[i].pos.x;
            const double dy = bodies[j].pos.y - bodies[i].pos.y;
            const double dz = bodies[j].pos.z - bodies[i].pos.z;
            const double dist = std::sqrt(dx * dx + dy * dy + dz * dz + SOFTENING);
            energy -= 1.0 / dist;
        }
    }
    
    return energy;
}

// Validate that simulation produces finite, reasonable values
bool validateSimulation(const std::vector<Body>& bodies) {
    for (const auto& body : bodies) {
        // Check for NaN or Inf values
        if (!std::isfinite(body.pos.x) || !std::isfinite(body.pos.y) || !std::isfinite(body.pos.z) ||
            !std::isfinite(body.vel.x) || !std::isfinite(body.vel.y) || !std::isfinite(body.vel.z)) {
            printf("Validation failed: found NaN or Inf value in body state\n");
            return false;
        }
        
        // Check for extreme values (bodies shouldn't fly off to infinity)
        const double maxPos = 1e6;
        const double maxVel = 1e6;
        if (std::abs(body.pos.x) > maxPos || std::abs(body.pos.y) > maxPos || std::abs(body.pos.z) > maxPos) {
            printf("Validation failed: body position exceeds reasonable bounds\n");
            return false;
        }
        if (std::abs(body.vel.x) > maxVel || std::abs(body.vel.y) > maxVel || std::abs(body.vel.z) > maxVel) {
            printf("Validation failed: body velocity exceeds reasonable bounds\n");
            return false;
        }
    }
    return true;
}
