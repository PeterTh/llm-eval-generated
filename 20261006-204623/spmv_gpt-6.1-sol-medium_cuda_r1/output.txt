// ****************************************************************************
// Function: fill
//
// Purpose:
//   Initialize array with random values
//
// Arguments:
//   A: pointer to the array to initialize
//   n: number of elements in the array
//   maxVal: specifies range of random values [0, maxVal]
//
// ****************************************************************************
void fill(double* A, const index_t n, const double maxVal) {
    for (index_t i = 0; i < n; ++i) {
        A[i] = maxVal * (rand() / (static_cast<double>(RAND_MAX) + 1.0));
    }
}

// ****************************************************************************
// Function: initRandomMatrix
//
// Purpose:
//   Assigns random positions to a given number of elements in a square
//   matrix. The function encodes these positions in compressed sparse
//   row (CSR) format.
//
// Arguments:
//   cols:          array for column indexes of elements (size should be = n)
//   rowDelimiters: array of size dim+1 holding indices to rows;
//                  last element is the index one past the last element
//   n:             number of nonzero elements
//   dim:           number of rows/columns in the matrix
//
// ****************************************************************************
void initRandomMatrix(index_t* cols, index_t* rowDelimiters, const index_t n, const index_t dim) {
    index_t nnzAssigned = 0;

    // Figure out the probability that a nonzero should be assigned
    double prob = static_cast<double>(n) / (static_cast<double>(dim) * static_cast<double>(dim));

    // Seed random number generator
    srand(8675309);

    // Randomly decide whether entry i,j gets a value, but ensure n values are assigned
    bool fillRemaining = false;
    for (index_t i = 0; i < dim; ++i) {
        rowDelimiters[i] = nnzAssigned;
        for (index_t j = 0; j < dim; ++j) {
            index_t numEntriesLeft = (dim * dim) - ((i * dim) + j);
            index_t needToAssign = n - nnzAssigned;
            if (numEntriesLeft <= needToAssign) {
                fillRemaining = true;
            }
            double randVal = static_cast<double>(rand()) / RAND_MAX;
            if ((nnzAssigned < n && randVal <= prob) || fillRemaining) {
                // Assign (i,j) a value
                cols[nnzAssigned] = j;
                nnzAssigned++;
            }
        }
    }
    // Convention: put the number of non-zeroes at the end of the row delimiters array
    rowDelimiters[dim] = n;
}

// ****************************************************************************
// Function: spmvCpu
//
// Purpose:
//   Computes sparse matrix-vector multiplication using CSR format
//
// Arguments:
//   val: array holding the non-zero values for the matrix
//   cols: array of column indices for each element
//   rowDelimiters: array of size dim+1 holding indices to rows;
//                  last element is the index one past the last element
//   vec: dense vector of size dim to be used for multiplication
//   dim: number of rows/columns in the matrix
//   out: output - result from the spmv calculation
//
// ****************************************************************************
void spmvCpu(const double* val, const index_t* cols, const index_t* rowDelimiters,
             const double* vec, const index_t dim, double* out) {
    for (index_t i = 0; i < dim; ++i) {
        double t = 0.0;
        for (index_t j = rowDelimiters[i]; j < rowDelimiters[i + 1]; ++j) {
            const auto col = cols[j];
            t += val[j] * vec[col];
        }
        out[i] = t;
    }
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

+// Adjacent lanes read adjacent CSR entries. Short rows use subwarps so that
+// sparse matrices do not waste an entire warp on each row.
+template <unsigned Width>
+__global__ void spmvRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                         const index_t* __restrict__ rows, const double* __restrict__ vec,
+                         index_t dim, double* __restrict__ out) {
+    const unsigned lane = threadIdx.x % Width;
+    const size_t row = (size_t(blockIdx.x) * blockDim.x + threadIdx.x) / Width;
+    double sum = 0.0;
+    if (row < dim) {
+        const uint64_t end = rows[row + 1];
+        for (uint64_t j = uint64_t(rows[row]) + lane; j < end; j += Width)
+            sum += val[j] * __ldg(vec + cols[j]);
+    }
+    // All lanes, including padded rows, participate in the shuffle.
+    for (unsigned offset = Width / 2; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset, Width);
+    if (lane == 0 && row < dim) out[row] = sum;
+}

+// Long rows use a block per row, increasing parallelism for dense matrices.
+__global__ void spmvLongRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                             const index_t* __restrict__ rows, const double* __restrict__ vec,
+                             double* __restrict__ out) {
+    constexpr unsigned Threads = 128;
+    __shared__ double partial[Threads / 32];
+    const unsigned row = blockIdx.x;
+    double sum = 0.0;
+    for (uint64_t j = uint64_t(rows[row]) + threadIdx.x; j < rows[row + 1]; j += Threads)
+        sum += val[j] * __ldg(vec + cols[j]);
+    for (unsigned offset = 16; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset);
+    if (threadIdx.x % 32 == 0) partial[threadIdx.x / 32] = sum;
+    __syncthreads();
+    if (threadIdx.x < 32) {
+        sum = threadIdx.x < Threads / 32 ? partial[threadIdx.x] : 0.0;
+        for (unsigned offset = 16; offset; offset /= 2)
+            sum += __shfl_down_sync(0xffffffffu, sum, offset);
+        if (threadIdx.x == 0) out[row] = sum;
+    }
+}

+                         index_t dim, double* __restrict__ out) {

+    double sum = 0.0;

+// Adjacent lanes read adjacent CSR entries. Short rows use subwarps so that
+// sparse matrices do not waste an entire warp on each row.
+template <unsigned Width>
+__global__ void spmvRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                         const index_t* __restrict__ rows, const double* __restrict__ vec,
+                         index_t dim, double* __restrict__ out) {
+    const unsigned lane = threadIdx.x % Width;
+    const size_t row = (size_t(blockIdx.x) * blockDim.x + threadIdx.x) / Width;
+    double sum = 0.0;
+    if (row < dim) {
+        const uint64_t end = rows[row + 1];
+        for (uint64_t j = uint64_t(rows[row]) + lane; j < end; j += Width)
+            sum += val[j] * __ldg(vec + cols[j]);
+    }
+    // All lanes, including padded rows, participate in the shuffle.
+    for (unsigned offset = Width / 2; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset, Width);
+    if (lane == 0 && row < dim) out[row] = sum;
+}

+// Long rows use a block per row, increasing parallelism for dense matrices.
+__global__ void spmvLongRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                             const index_t* __restrict__ rows, const double* __restrict__ vec,
+                             double* __restrict__ out) {
+    constexpr unsigned Threads = 128;
+    __shared__ double partial[Threads / 32];
+    const unsigned row = blockIdx.x;
+    double sum = 0.0;
+    for (uint64_t j = uint64_t(rows[row]) + threadIdx.x; j < rows[row + 1]; j += Threads)
+        sum += val[j] * __ldg(vec + cols[j]);
+    for (unsigned offset = 16; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset);
+    if (threadIdx.x % 32 == 0) partial[threadIdx.x / 32] = sum;
+    __syncthreads();
+    if (threadIdx.x < 32) {
+        sum = threadIdx.x < Threads / 32 ? partial[threadIdx.x] : 0.0;
+        for (unsigned offset = 16; offset; offset /= 2)
+            sum += __shfl_down_sync(0xffffffffu, sum, offset);
+        if (threadIdx.x == 0) out[row] = sum;
+    }

+// Adjacent lanes read adjacent CSR entries. Short rows use subwarps so that
+// sparse matrices do not waste an entire warp on each row.
+template <unsigned Width>
+__global__ void spmvRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                         const index_t* __restrict__ rows, const double* __restrict__ vec,
+                         index_t dim, double* __restrict__ out) {
+    const unsigned lane = threadIdx.x % Width;
+    const size_t row = (size_t(blockIdx.x) * blockDim.x + threadIdx.x) / Width;
+    double sum = 0.0;
+    if (row < dim) {
+        const uint64_t end = rows[row + 1];
+        for (uint64_t j = uint64_t(rows[row]) + lane; j < end; j += Width)
+            sum += val[j] * __ldg(vec + cols[j]);
+    }
+    // All lanes, including padded rows, participate in the shuffle.
+    for (unsigned offset = Width / 2; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset, Width);
+    if (lane == 0 && row < dim) out[row] = sum;
+}

+// Long rows use a block per row, increasing parallelism for dense matrices.
+__global__ void spmvLongRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                             const index_t* __restrict__ rows, const double* __restrict__ vec,
+                             double* __restrict__ out) {
+    constexpr unsigned Threads = 128;
+    __shared__ double partial[Threads / 32];
+    const unsigned row = blockIdx.x;
+    double sum = 0.0;
+    for (uint64_t j = uint64_t(rows[row]) + threadIdx.x; j < rows[row + 1]; j += Threads)
+        sum += val[j] * __ldg(vec + cols[j]);
+    for (unsigned offset = 16; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset);
+    if (threadIdx.x % 32 == 0) partial[threadIdx.x / 32] = sum;
+    __syncthreads();
+    if (threadIdx.x < 32) {
+        sum = threadIdx.x < Threads / 32 ? partial[threadIdx.x] : 0.0;
+        for (unsigned offset = 16; offset; offset /= 2)

+// Adjacent lanes read adjacent CSR entries. Short rows use subwarps so that

+template <unsigned Width>
+__global__ void spmvRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                         const index_t* __restrict__ rows, const double* __restrict__ vec,
+                         index_t dim, double* __restrict__ out) {
+    const unsigned lane = threadIdx.x % Width;
+    const size_t row = (size_t(blockIdx.x) * blockDim.x + threadIdx.x) / Width;
+    double sum = 0.0;
+    if (row < dim) {
+        const uint64_t end = rows[row + 1];
+        for (uint64_t j = uint64_t(rows[row]) + lane; j < end; j += Width)
+            sum += val[j] * __ldg(vec + cols[j]);
+    }
+    // All lanes, including padded rows, participate in the shuffle.
+    for (unsigned offset = Width / 2; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset, Width);

+__global__ void spmvLongRows(const double* __restrict__ val, const index_t* __restrict__ cols,

+    __shared__ double partial[Threads / 32];
+    const unsigned row = blockIdx.x;
+    double sum = 0.0;
+    for (uint64_t j = uint64_t(rows[row]) + threadIdx.x; j < rows[row + 1]; j += Threads)

+__global__ void spmvRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                         const index_t* __restrict__ rows, const double* __restrict__ vec,

+            sum += val[j] * __ldg(vec + cols[j]);

                         const index_t* __restrict__ rows, const double* __restrict__ vec,
                         index_t dim, double* __restrict__ out) {
    const unsigned lane = threadIdx.x % Width;
    const size_t row = (size_t(blockIdx.x) * blockDim.x + threadIdx.x) / Width;
    double sum = 0.0;
    if (row < dim) {
        const uint64_t end = rows[row + 1];
        for (uint64_t j = uint64_t(rows[row]) + lane; j < end; j += Width)
            sum += val[j] * __ldg(vec + cols[j]);
    }
    // All lanes, including padded rows, participate in the shuffle.
    for (unsigned offset = Width / 2; offset; offset /= 2)
        sum += __shfl_down_sync(0xffffffffu, sum, offset, Width);
    if (lane == 0 && row < dim) out[row] = sum;
}

// Long rows use a block per row, increasing parallelism for dense matrices.
__global__ void spmvLongRows(const double* __restrict__ val, const index_t* __restrict__ cols,
                             const index_t* __restrict__ rows, const double* __restrict__ vec,
                             double* __restrict__ out) {
    constexpr unsigned Threads = 128;
    __shared__ double partial[Threads / 32];
    const unsigned row = blockIdx.x;
    double sum = 0.0;
    for (uint64_t j = uint64_t(rows[row]) + threadIdx.x; j < rows[row + 1]; j += Threads)
        sum += val[j] * __ldg(vec + cols[j]);
    for (unsigned offset = 16; offset; offset /= 2)
        sum += __shfl_down_sync(0xffffffffu, sum, offset);
    if (threadIdx.x % 32 == 0) partial[threadIdx.x / 32] = sum;
    __syncthreads();
    if (threadIdx.x < 32) {
        sum = threadIdx.x < Threads / 32 ? partial[threadIdx.x] : 0.0;
        for (unsigned offset = 16; offset; offset /= 2)
            sum += __shfl_down_sync(0xffffffffu, sum, offset);
        if (threadIdx.x == 0) out[row] = sum;
    }
}

double spmvGpu(const std::vector<double>& val, const std::vector<index_t>& cols,
               const std::vector<index_t>& rows, const std::vector<double>& vec,
               index_t iterations, std::vector<double>& out) {
    DeviceBuffer<double> d_val(val.size()), d_vec(vec.size()), d_out(out.size());
    DeviceBuffer<index_t> d_cols(cols.size()), d_rows(rows.size());
    d_val.upload(val);
    d_cols.upload(cols);
    d_rows.upload(rows);
    d_vec.upload(vec);
    const index_t dim = static_cast<index_t>(vec.size());
    const size_t average = dim ? val.size() / dim : 0;
    const unsigned width = average <= 2 ? 2 : average <= 4 ? 4 :
                           average <= 8 ? 8 : average <= 16 ? 16 : 32;
    // Preserve the original zero-iteration output (a zero-initialized vector).
    if (!iterations || !dim) return 0.0;
    cudaStream_t stream;
    checkCuda(cudaStreamCreateWithFlags(&stream, cudaStreamNonBlocking));
    auto launch = [&]() {
        if (average >= 256) {
            spmvLongRows<<<dim, 128, 0, stream>>>(d_val.data, d_cols.data, d_rows.data, d_vec.data, d_out.data);
        } else {
            const unsigned blocks = static_cast<unsigned>((size_t(dim) * width + 127) / 128);
            switch (width) {
                case 2: spmvRows<2><<<blocks, 128, 0, stream>>>(d_val.data, d_cols.data, d_rows.data, d_vec.data, dim, d_out.data); break;
                case 4: spmvRows<4><<<blocks, 128, 0, stream>>>(d_val.data, d_cols.data, d_rows.data, d_vec.data, dim, d_out.data); break;
                case 8: spmvRows<8><<<blocks, 128, 0, stream>>>(d_val.data, d_cols.data, d_rows.data, d_vec.data, dim, d_out.data); break;
                case 16: spmvRows<16><<<blocks, 128, 0, stream>>>(d_val.data, d_cols.data, d_rows.data, d_vec.data, dim, d_out.data); break;
                default: spmvRows<32><<<blocks, 128, 0, stream>>>(d_val.data, d_cols.data, d_rows.data, d_vec.data, dim, d_out.data); break;
            }
        }
    };
    launch();
    checkCuda(cudaGetLastError());
    checkCuda(cudaStreamSynchronize(stream));
    // Replay bounded batches to amortize host launch overhead without growing
    // graph storage with the requested iteration count.
    const index_t batch = iterations < 128 ? iterations : 128;
    cudaGraph_t graph = nullptr;
    cudaGraphExec_t executableGraph = nullptr;
    if (batch > 1) {
        checkCuda(cudaStreamBeginCapture(stream, cudaStreamCaptureModeThreadLocal));
        for (index_t iter = 0; iter < batch; ++iter) launch();
        checkCuda(cudaStreamEndCapture(stream, &graph));
        checkCuda(cudaGraphInstantiate(&executableGraph, graph, nullptr, nullptr, 0));
        checkCuda(cudaGraphUpload(executableGraph, stream));
        checkCuda(cudaStreamSynchronize(stream));
    }
    cudaEvent_t start, stop;
    checkCuda(cudaEventCreate(&start));
    checkCuda(cudaEventCreate(&stop));
    checkCuda(cudaEventRecord(start, stream));
    index_t remaining = iterations;
    if (executableGraph) {
        while (remaining >= batch) {
            checkCuda(cudaGraphLaunch(executableGraph, stream));
            remaining -= batch;
        }
    }
    for (index_t iter = 0; iter < remaining; ++iter) launch();
    checkCuda(cudaGetLastError());
    checkCuda(cudaEventRecord(stop, stream));
    checkCuda(cudaEventSynchronize(stop));
    float milliseconds = 0.0f;
    checkCuda(cudaEventElapsedTime(&milliseconds, start, stop));
    checkCuda(cudaMemcpy(out.data(), d_out.data, out.size() * sizeof(double), cudaMemcpyDeviceToHost));
    checkCuda(cudaEventDestroy(start));
    checkCuda(cudaEventDestroy(stop));
    if (executableGraph) checkCuda(cudaGraphExecDestroy(executableGraph));
    if (graph) checkCuda(cudaGraphDestroy(graph));
    checkCuda(cudaStreamDestroy(stream));
    return milliseconds;
}

// ****************************************************************************
// Function: verifyResults
//
// Purpose:
//   Verifies correctness of results by comparing to reference solution
//
// Arguments:
//   reference: array holding the reference result vector
//   result: array holding the result vector to verify
//   size: number of elements per vector
//
// ****************************************************************************
bool verifyResults(const double* reference, const double* result, const index_t size) {
    for (index_t i = 0; i < size; ++i) {
        const double ref = reference[i];
        const double res = result[i];
        if (std::abs(ref) < 1e-10) {
            // For very small values, check absolute error
            if (std::abs(res) > MAX_RELATIVE_ERROR) {
                printf("Validation failed at index %u: reference %.10e, got %.10e\n", i, ref, res);
                return false;
            }
        } else {
            // Check relative error
            const double relError = std::abs((res - ref) / ref);
            if (relError > MAX_RELATIVE_ERROR) {
                printf("Validation failed at index %u: reference %.10e, got %.10e (rel error: %.10e)\n",
                       i, ref, res, relError);
                return false;
            }
        }
    }
    return true;
}

+// Adjacent lanes read adjacent CSR entries. Short rows use subwarps so that
+// sparse matrices do not waste an entire warp on each row.
+template <unsigned Width>
+__global__ void spmvRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                         const index_t* __restrict__ rows, const double* __restrict__ vec,
+                         index_t dim, double* __restrict__ out) {
+    const unsigned lane = threadIdx.x % Width;
+    const size_t row = (size_t(blockIdx.x) * blockDim.x + threadIdx.x) / Width;
+    double sum = 0.0;
+    if (row < dim) {
+        const uint64_t end = rows[row + 1];
+        for (uint64_t j = uint64_t(rows[row]) + lane; j < end; j += Width)
+            sum += val[j] * __ldg(vec + cols[j]);
+    }
+    // All lanes, including padded rows, participate in the shuffle.
+    for (unsigned offset = Width / 2; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset, Width);
+    if (lane == 0 && row < dim) out[row] = sum;
+}
+
+// Long rows use a block per row, increasing parallelism for dense matrices.
+__global__ void spmvLongRows(const double* __restrict__ val, const index_t* __restrict__ cols,
+                             const index_t* __restrict__ rows, const double* __restrict__ vec,
+                             double* __restrict__ out) {
+    constexpr unsigned Threads = 128;
+    __shared__ double partial[Threads / 32];
+    const unsigned row = blockIdx.x;
+    double sum = 0.0;
+    for (uint64_t j = uint64_t(rows[row]) + threadIdx.x; j < rows[row + 1]; j += Threads)
+        sum += val[j] * __ldg(vec + cols[j]);
+    for (unsigned offset = 16; offset; offset /= 2)
+        sum += __shfl_down_sync(0xffffffffu, sum, offset);
+    if (threadIdx.x % 32 == 0) partial[threadIdx.x / 32] = sum;
+    __syncthreads();
+    if (threadIdx.x < 32) {
+        sum = threadIdx.x < Threads / 32 ? partial[threadIdx.x] : 0.0;
+        for (unsigned offset = 16; offset; offset /= 2)
+            sum += __shfl_down_sync(0xffffffffu, sum, offset);
+        if (threadIdx.x == 0) out[row] = sum;
+    }
+}
