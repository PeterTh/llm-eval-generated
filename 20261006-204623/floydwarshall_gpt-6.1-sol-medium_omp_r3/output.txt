// Index calculation for flattened 2D array
inline constexpr size_t idx2(const size_t i, const size_t j, const size_t n) noexcept {

void initializeDistanceMatrix(std::vector<unsigned int>& dist, const size_t numNodes, 
                              const unsigned int rangeMin, const unsigned int rangeMax) {

    const double range = static_cast<double>(rangeMax - rangeMin) + 1.0;

    }
}

    }

void floydWarshall(std::vector<unsigned int>& dist, 
                   std::vector<unsigned int>& path, 
                   const size_t numNodes) {
    // Classic Floyd-Warshall algorithm
    // For each intermediate node k
    for (size_t k = 0; k < numNodes; ++k) {
        // For each source node i
        for (size_t i = 0; i < numNodes; ++i) {
            // For each destination node j

                const unsigned int distIJ = dist[idx2(j, i, numNodes)];
                const unsigned int distIK = dist[idx2(k, i, numNodes)];

                if (newDist < distIJ) {

                    path[idx2(j, i, numNodes)] = k;
                }
            }
        }
    }
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

// FNV-1a hash, 64 bit length
class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

    return j * n + i;

inline int matrixThreadCount(const size_t numNodes) {
    // Give each worker at least 16 rows: oversized teams spend more time at
    // the per-pivot barrier than doing useful work on small matrices.
    return static_cast<int>(std::min(static_cast<size_t>(omp_get_max_threads()),
                                    std::max(size_t{1}, numNodes / 16)));
}

void initializeDistanceMatrix(std::vector<unsigned int>& dist, const size_t numNodes, 
                              const unsigned int rangeMin, const unsigned int rangeMax) {
    unsigned int seed = 42;
    const double range = static_cast<double>(rangeMax - rangeMin) + 1.0;
    
    for (size_t i = 0; i < numNodes * numNodes; ++i) {
        dist[i] = rangeMin + (unsigned int)(range * rand_r(&seed) / (double)RAND_MAX);
    }
    
    // Set diagonal to zero (distance from node to itself is 0)
    for (size_t i = 0; i < numNodes; ++i) {
        dist[idx2(i, i, numNodes)] = 0;
    }
}

void initializePathMatrix(std::vector<unsigned int>& path, const size_t numNodes) {
    // Both assignments in the original loop set each entry to its row index.
    // Write contiguous rows once, with no overlapping writes between threads.
    #pragma omp parallel for schedule(static) num_threads(matrixThreadCount(numNodes))
    for (size_t j = 0; j < numNodes; ++j) {
        #pragma omp simd
        for (size_t i = 0; i < numNodes; ++i) {
            path[idx2(i, j, numNodes)] = j;
        }
    }
}

void floydWarshall(std::vector<unsigned int>& dist, 
                   std::vector<unsigned int>& path, 
                   const size_t numNodes) {
    unsigned int* const distances = dist.data();
    unsigned int* const paths = path.data();

    // Reuse one team for all pivots. The worksharing barrier completes every
    // update for k before any thread starts k+1, preserving path tie breaking.
    #pragma omp parallel default(none) shared(distances, paths, numNodes) num_threads(matrixThreadCount(numNodes))
    {
        for (size_t k = 0; k < numNodes; ++k) {
            const unsigned int* const pivot = distances + k * numNodes;
            #pragma omp for schedule(static)
            for (size_t i = 0; i < numNodes; ++i) {
                // The zero diagonal and nonnegative weights make the pivot
                // row invariant. Leave it read-only for all other threads.
                if (i == k) {
                    continue;
                }
                unsigned int* const row = distances + i * numNodes;
                unsigned int* const pathRow = paths + i * numNodes;
                const unsigned int distIK = row[k];

                #pragma omp simd
                for (size_t j = 0; j < numNodes; ++j) {
                    const unsigned int newDist = distIK + pivot[j];
                    if (newDist < row[j]) {
                        row[j] = newDist;
                        pathRow[j] = static_cast<unsigned int>(k);
                    }
                }
            }
        }
    }
}
