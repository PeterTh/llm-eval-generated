// ****************************************************************************
// Function: fill
//
// Purpose:
//   Initialize array with random values
//
// Arguments:
//   A: pointer to the array to initialize
//   n: number of elements in the array
//   maxVal: specifies range of random values [0, maxVal]
//
// ****************************************************************************
void fill(double* A, const index_t n, const double maxVal) {
    for (index_t i = 0; i < n; ++i) {
        A[i] = maxVal * (rand() / (static_cast<double>(RAND_MAX) + 1.0));
    }
}

// ****************************************************************************
// Function: initRandomMatrix
//
// Purpose:
//   Assigns random positions to a given number of elements in a square
//   matrix. The function encodes these positions in compressed sparse
//   row (CSR) format.
//
// Arguments:
//   cols:          array for column indexes of elements (size should be = n)
//   rowDelimiters: array of size dim+1 holding indices to rows;
//                  last element is the index one past the last element
//   n:             number of nonzero elements
//   dim:           number of rows/columns in the matrix
//
// ****************************************************************************
void initRandomMatrix(index_t* cols, index_t* rowDelimiters, const index_t n, const index_t dim) {
    index_t nnzAssigned = 0;

    // Figure out the probability that a nonzero should be assigned
    double prob = static_cast<double>(n) / (static_cast<double>(dim) * static_cast<double>(dim));

    // Seed random number generator
    srand(8675309);

    // Randomly decide whether entry i,j gets a value, but ensure n values are assigned
    bool fillRemaining = false;
    for (index_t i = 0; i < dim; ++i) {
        rowDelimiters[i] = nnzAssigned;
        for (index_t j = 0; j < dim; ++j) {
            index_t numEntriesLeft = (dim * dim) - ((i * dim) + j);
            index_t needToAssign = n - nnzAssigned;
            if (numEntriesLeft <= needToAssign) {
                fillRemaining = true;
            }
            double randVal = static_cast<double>(rand()) / RAND_MAX;
            if ((nnzAssigned < n && randVal <= prob) || fillRemaining) {
                // Assign (i,j) a value
                cols[nnzAssigned] = j;
                nnzAssigned++;
            }
        }
    }
    // Convention: put the number of non-zeroes at the end of the row delimiters array
    rowDelimiters[dim] = n;
}

// ****************************************************************************
// Function: spmvCpu
//
// Purpose:
//   Computes sparse matrix-vector multiplication using CSR format
//
// Arguments:
//   val: array holding the non-zero values for the matrix
//   cols: array of column indices for each element
//   rowDelimiters: array of size dim+1 holding indices to rows;
//                  last element is the index one past the last element
//   vec: dense vector of size dim to be used for multiplication
//   dim: number of rows/columns in the matrix
//   out: output - result from the spmv calculation
//
// ****************************************************************************
void spmvCpu(const double* val, const index_t* cols, const index_t* rowDelimiters,
             const double* vec, const index_t dim, double* out) {
    for (index_t i = 0; i < dim; ++i) {
        double t = 0.0;
        for (index_t j = rowDelimiters[i]; j < rowDelimiters[i + 1]; ++j) {
            const auto col = cols[j];
            t += val[j] * vec[col];
        }
        out[i] = t;
    }
}

class hash {
  public:
	using digest = uint64_t;

	template <typename T>
	void add(const T& value) {
		const unsigned char* bytes = reinterpret_cast<const unsigned char*>(&value);
		for(size_t i = 0; i < sizeof(T); ++i) {
			d = (d ^ bytes[i]) * 0x100000001b3ull;
		}
	}

	digest get() const { return d; }

  private:
	digest d = 0xcbf29ce484222325ull;
};

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

+void spmvOpenmp(const double* val, const index_t* cols, const index_t* rowDelimiters,
+                const double* vec, const index_t dim, double* out,
+                const index_t iterations) {
+    const int numThreads = static_cast<int>(
+        std::min<uint64_t>(omp_get_max_threads(), std::max<uint64_t>(dim, 1)));
+

+        shared(val, cols, rowDelimiters, vec, dim, out, iterations)

    38	// Function: initRandomMatrix
    39	//
    40	// Purpose:
    41	//   Assigns random positions to a given number of elements in a square
    42	//   matrix. The function encodes these positions in compressed sparse
    43	//   row (CSR) format.
    44	//
    45	// Arguments:
    46	//   cols:          array for column indexes of elements (size should be = n)
    47	//   rowDelimiters: array of size dim+1 holding indices to rows;
    48	//                  last element is the index one past the last element
    49	//   n:             number of nonzero elements
    50	//   dim:           number of rows/columns in the matrix
    51	//
    52	// ****************************************************************************
    53	void initRandomMatrix(index_t* cols, index_t* rowDelimiters, const index_t n, const index_t dim) {
    54	    index_t nnzAssigned = 0;
    55	
    56	    // Figure out the probability that a nonzero should be assigned
    57	    double prob = static_cast<double>(n) / (static_cast<double>(dim) * static_cast<double>(dim));
    58	
    59	    // Seed random number generator
    60	    srand(8675309);
    61	
    62	    // Randomly decide whether entry i,j gets a value, but ensure n values are assigned
    63	    bool fillRemaining = false;
    64	    for (index_t i = 0; i < dim; ++i) {
    65	        rowDelimiters[i] = nnzAssigned;
    66	        for (index_t j = 0; j < dim; ++j) {
    67	            index_t numEntriesLeft = (dim * dim) - ((i * dim) + j);
    68	            index_t needToAssign = n - nnzAssigned;
    69	            if (numEntriesLeft <= needToAssign) {
    70	                fillRemaining = true;
    71	            }
    72	            double randVal = static_cast<double>(rand()) / RAND_MAX;
    73	            if ((nnzAssigned < n && randVal <= prob) || fillRemaining) {
    74	                // Assign (i,j) a value
    75	                cols[nnzAssigned] = j;
    76	                nnzAssigned++;
    77	            }
    78	        }
    79	    }

    81	    rowDelimiters[dim] = n;

   100	void spmvCpu(const double* val, const index_t* cols, const index_t* rowDelimiters,

   112	// Keep a single team alive for the entire benchmark. Each thread owns a
   113	// contiguous range of rows, so iterations need no intermediate barriers.
   114	void spmvOpenmp(const double* val, const index_t* cols, const index_t* rowDelimiters,
   115	                const double* vec, const index_t dim, double* out,
   116	                const index_t iterations) {
   117	    const int numThreads = static_cast<int>(
   118	        std::min<uint64_t>(omp_get_max_threads(), std::max<uint64_t>(dim, 1)));
   119	
   120	    #pragma omp parallel num_threads(numThreads) default(none) \
   121	        shared(val, cols, rowDelimiters, vec, dim, out, iterations)
   122	    {
   123	        const uint64_t thread = omp_get_thread_num();
   124	        const uint64_t teamSize = omp_get_num_threads();
   125	        // Include a unit of work per row so empty rows are balanced too.
   126	        const uint64_t work = static_cast<uint64_t>(rowDelimiters[dim]) + dim;
   127	        const auto boundary = [&](uint64_t part) -> index_t {
   128	            const uint64_t target = work * part / teamSize;
   129	            index_t lo = 0;
   130	            index_t hi = dim;
   131	            while (lo < hi) {
   132	                const index_t mid = lo + (hi - lo) / 2;
   133	                if (static_cast<uint64_t>(rowDelimiters[mid]) + mid < target) {
   134	                    lo = mid + 1;
   135	                } else {
   136	                    hi = mid;
   137	                }
   138	            }
   139	            return lo;
   140	        };
   141	        const index_t first = boundary(thread);
   142	        const index_t last = boundary(thread + 1);
   143	
   144	        for (index_t iter = 0; iter < iterations; ++iter) {
   145	            for (index_t i = first; i < last; ++i) {
   146	                double sum = 0.0;
   147	                // Preserve the serial accumulation order within each row.
   148	                for (index_t j = rowDelimiters[i]; j < rowDelimiters[i + 1]; ++j) {
   149	                    sum += val[j] * vec[cols[j]];
   150	                }
   151	                out[i] = sum;
   152	            }
   153	        }
   154	    }
   155	}
