// Standard normal cumulative distribution function
inline double cumulativeNormal(const double x) noexcept {
    return 0.5 * (1.0 + std::erf(x * M_SQRT1_2));
}

// Standard normal probability density function
inline double normalPDF(const double x) noexcept {
    return M_1_SQRTPI * M_SQRT1_2 * exp(-0.5 * x * x);
}

// Black-Scholes formula for European options
double blackScholes(const OptionInput& option) noexcept {
    const double S = option.spot;
    const double K = option.strike;
    const double r = option.r;
    const double q = option.q;
    const double T = option.t;
    const double sigma = option.vol;
    
    if (T <= 0.0 || sigma <= 0.0) {
        return 0.0;
    }
    
    const double d1 = (log(S / K) + (r - q + 0.5 * sigma * sigma) * T) / (sigma * sqrt(T));
    const double d2 = d1 - sigma * sqrt(T);
    
    const double Nd1 = cumulativeNormal(d1);
    const double Nd2 = cumulativeNormal(d2);
    const double discount = exp(-r * T);
    
    double price;
    if (option.type == CALL) {
        price = S * exp(-q * T) * Nd1 - K * discount * Nd2;
    } else { // PUT
        price = K * discount * cumulativeNormal(-d2) - S * exp(-q * T) * cumulativeNormal(-d1);
    }
    
    return price;
}

// Kahan summation for numerically stable sum
template <typename T>
T kahan_sum(const std::vector<T>& data) {
    T sum = 0.0;
    T c = 0.0;
    for (const auto& val : data) {
        T y = val - c;
        T t = sum + y;
        c = (t - sum) - y;
        sum = t;
    }
    return sum;
}

	template <typename T>
	void add(const T& value) {

// Contiguous partitions preserve output order and differ by at most one option.
struct Partition {
    size_t first;
    size_t count;
};

Partition partitionOptions(size_t total, int rank, int ranks) {
    const size_t base = total / static_cast<size_t>(ranks);
    const size_t extra = total % static_cast<size_t>(ranks);
    return {base * static_cast<size_t>(rank) +
                std::min(static_cast<size_t>(rank), extra),
            base + (static_cast<size_t>(rank) < extra ? 1 : 0)};
}

// Gather only the requested prefix. Windowing keeps MPI counts/displacements
// within the MPI int limit even when the full dataset is larger than INT_MAX.
void gatherResults(const std::vector<double>& localResults,
                   std::vector<double>& results, size_t total, size_t requested,
                   int rank, int ranks) {
    const Partition local = partitionOptions(total, rank, ranks);
    std::vector<int> counts(rank == 0 ? ranks : 0);
    std::vector<int> displacements(rank == 0 ? ranks : 0);
    for (size_t first = 0; first < requested;) {
        const size_t length = std::min(requested - first,
                                       static_cast<size_t>(INT_MAX));
        const size_t end = first + length;
        const size_t beginLocal = std::max(first, local.first);
        const size_t endLocal = std::min(end, local.first + local.count);
        const int sendCount = beginLocal < endLocal
            ? static_cast<int>(endLocal - beginLocal) : 0;
        if (rank == 0) {
            for (int source = 0; source < ranks; ++source) {
                const Partition part = partitionOptions(total, source, ranks);
                const size_t begin = std::max(first, part.first);
                const size_t finish = std::min(end, part.first + part.count);
                counts[source] = begin < finish ? static_cast<int>(finish - begin) : 0;
                displacements[source] = counts[source] ? static_cast<int>(begin - first) : 0;
            }
        }
        MPI_Gatherv(sendCount ? localResults.data() + (beginLocal - local.first) : nullptr,
                    sendCount, MPI_DOUBLE,
                    rank == 0 ? results.data() + first : nullptr,
                    counts.data(), displacements.data(), MPI_DOUBLE, 0, MPI_COMM_WORLD);
        first = end;
    }
}

int runBenchmark(int argc, char** argv, int rank, int ranks) {
    size_t numOptions = 10000;
    bool validate = false;
    bool printResults = false;

    // MPI launchers supply the same arguments to every rank.
    for (int i = 1; i < argc; ++i) {
        if (strcmp(argv[i], "-n") == 0 && i + 1 < argc) {
            numOptions = atoll(argv[++i]);
        } else if (strcmp(argv[i], "-v") == 0) {
            validate = true;
        } else if (strcmp(argv[i], "-r") == 0) {
            printResults = true;
        } else if (strcmp(argv[i], "-h") == 0) {
            if (rank == 0) printUsage(argv[0]);
            return 0;
        } else {
            if (rank == 0) {
                printf("Unknown option: %s\n", argv[i]);
                printUsage(argv[0]);
            }
            return 1;
        }
    }

    if (rank == 0) {
        printf("Black-Scholes Option Pricing Benchmark\n");
        printf("Number of options: %zu\n", numOptions);
        printf("Validation: %s\n", validate ? "enabled" : "disabled");
    }

    const Partition local = partitionOptions(numOptions, rank, ranks);
    std::vector<OptionInput> options;
    generateOptions(options, local.count, local.first);
    std::vector<double> localResults(local.count);

    // Full collection is needed only for ordered statistics and hashing (-r).
    // Validation examines only the first ten options in the original benchmark.
    const size_t requested = printResults ? numOptions :
        (validate ? std::min(size_t{10}, numOptions) : 0);
    std::vector<double> results(rank == 0 ? requested : 0);

    if (rank == 0) printf("Pricing options...\n");
    MPI_Barrier(MPI_COMM_WORLD);
    const double start = MPI_Wtime();
    for (size_t i = 0; i < local.count; ++i) {
        localResults[i] = blackScholes(options[i]);
    }
    const double elapsed = MPI_Wtime() - start;
    double duration = 0.0;
    // Report the slowest rank, which determines distributed compute throughput.
    MPI_Reduce(&elapsed, &duration, 1, MPI_DOUBLE, MPI_MAX, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        printf("Computation time: %.3f ms\n", duration * 1000.0);
        printf("Options per second: %.0f\n", numOptions / duration);
    }
    gatherResults(localResults, results, numOptions, requested, rank, ranks);

    int status = 0;
    if (rank == 0) {
        if (printResults) print_results(results, "OptionPrices");
        if (validate) {
            printf("Validating results...\n");
            std::vector<OptionInput> validationOptions;
            generateOptions(validationOptions, std::min(size_t{10}, numOptions));
            const bool valid = validateResults(validationOptions, results);
            printf("Validation: %s\n", valid ? "PASSED" : "FAILED");
            status = valid ? 0 : 1;
        }
    }
    MPI_Bcast(&status, 1, MPI_INT, 0, MPI_COMM_WORLD);
    return status;
}
