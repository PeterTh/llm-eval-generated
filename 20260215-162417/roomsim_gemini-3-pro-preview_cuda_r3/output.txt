HOST_DEVICE bool rayIntersectsBoxDevice(const FlatOctreeNode& node, const Vec3& p1, const Vec3& p2) {
    Vec3 d = (p2 - p1) * 0.5f;
    Vec3 c = p1 + d - node.center;
    Vec3 ad = {std::abs(d.x), std::abs(d.y), std::abs(d.z)};

    if (std::abs(c.x) > node.halfExtent.x + ad.x) return false;
    if (std::abs(c.y) > node.halfExtent.y + ad.y) return false;
    if (std::abs(c.z) > node.halfExtent.z + ad.z) return false;

    if (std::abs(d.y * c.z - d.z * c.y) > node.halfExtent.y * ad.z + node.halfExtent.z * ad.y + EPSILON) return false;
    if (std::abs(d.z * c.x - d.x * c.z) > node.halfExtent.z * ad.x + node.halfExtent.x * ad.z + EPSILON) return false;
    if (std::abs(d.x * c.y - d.y * c.x) > node.halfExtent.x * ad.y + node.halfExtent.y * ad.x + EPSILON) return false;

    return true;
}

HOST_DEVICE bool isRayBlockedDevice(const Vec3& from, const Vec3& to,
                  const FlatOctreeNode* nodes, const int* indices, const Triangle* triangles,
                  size_t srcTriIdx, size_t dstTriIdx) {
    Vec3 dir = to - from;
    val_t rayLen = dir.norm();
    if (rayLen < EPSILON) return true;
    Vec3 dirNorm = dir / rayLen;

    // Stack based traversal
    int stack[64];
    int stackTop = 0;
    stack[0] = 0; // Root index

    while (stackTop >= 0) {
        int nodeIdx = stack[stackTop--];
        const FlatOctreeNode& node = nodes[nodeIdx];

        if (!rayIntersectsBoxDevice(node, from, to)) continue;

        // Leaf
        if (node.triangleIndicesCount > 0) {
            for (int i = 0; i < node.triangleIndicesCount; ++i) {
                int triIdx = indices[node.triangleIndicesStart + i];
                if (triIdx == srcTriIdx || triIdx == dstTriIdx) continue;

                const Triangle& tri = triangles[triIdx];
                val_t dist = rayTriangleIntersect(from, dirNorm, tri.a, tri.b, tri.c);
                if (dist > EPSILON && dist < rayLen - EPSILON) {
                    return true;
                }
            }
        }

        // Children
        for (int i = 0; i < 8; ++i) {
            if (node.children[i] != -1) {
                stack[++stackTop] = node.children[i];
            }
        }
    }
    return false;
}

HOST_DEVICE float deviceRand(uint64_t seed, uint64_t seq) {
    uint64_t x = seed + seq;
    x = (x ^ (x >> 30)) * 0xbf58476d1ce4e5b9ULL;
    x = (x ^ (x >> 27)) * 0x94d049bb133111ebULL;

__global__ void computeTauKernel(int numTriangles, Triangle* triangles, int* tau) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= numTriangles * numTriangles) return;

    int i = idx / numTriangles;

    tau[idx] = static_cast<int>(ceil(dist * INV_WAVE_SPEED)); // use cuda ceil
}

__global__ void computeKijKernel(int numTriangles, Triangle* triangles, FlatOctreeNode* nodes, int* indices, val_t* kij, int seed) {
    int idx = blockIdx.x * blockDim.x + threadIdx.x;
    if (idx >= numTriangles * numTriangles) return;

    int i = idx / numTriangles;
    int j = idx % numTriangles;

    if (i == j) {
        kij[idx] = 0.0f;
        return;
    }

    Triangle triI = triangles[i];
    Triangle triJ = triangles[j];

    if (triI.normal().dot(triJ.normal()) > 0.99f) {
        kij[idx] = 0.0f;
        return;
    }

    val_t val = 0.0f;
    uint64_t seq = idx * NUM_RAYS * 2; // 2 random numbers per ray

    for (int r = 0; r < NUM_RAYS; ++r) {
        Vec3 pI = randomPointInTriangleDevice(triI, seed, seq);
        Vec3 pJ = randomPointInTriangleDevice(triJ, seed, seq);

        if (isRayBlockedDevice(pI, pJ, nodes, indices, triangles, i, j)) continue;

        Vec3 v = pJ - pI;
        val_t distSqr = v.squaredNorm();
        if (distSqr < EPSILON) continue;

        val_t cosPhiI = cosPhiDevice(v, triI.normal());
        val_t cosPhiJ = cosPhiDevice(-v, triJ.normal());

        if (cosPhiI <= ZERO || cosPhiJ <= ZERO) continue;

        val += (cosPhiI * cosPhiJ) / (PI * distSqr);
    }

    kij[idx] = val * INV_NUM_RAYS;
}

__global__ void runSimulationStepKernel(int numTriangles, int t, int numTimesteps, 
                                      val_t* kij, int* tau, val_t* areas, val_t* rho, 
                                      val_t* radE, val_t* radB) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i >= numTriangles) return;

    val_t sumB = ZERO;

    for (int j = 0; j < numTriangles; ++j) {
        if (i == j) continue;

        int idx = i * numTriangles + j;
        int tauij = tau[idx];

        if (t < tauij) continue;

        val_t k = kij[idx];
        if (k <= ZERO) continue;

        int srcTime = t - tauij;
        val_t radJ = radB[srcTime * numTriangles + j];
        if (radJ <= ZERO) continue;

        sumB += min(k * areas[j], ONE) * radJ;
    }

    radB[t * numTriangles + i] = rho[i] * sumB + radE[t * numTriangles + i];
}

__global__ void computeDistancesKernel(int numTriangles, int numTimesteps, int sourceIndex, val_t* radB, val_t* distances) {
    int i = blockIdx.x * blockDim.x + threadIdx.x;
    if (i >= numTriangles) return;

    val_t maxCorr = ZERO;
    int bestT = 0;

    for (int t = 0; t < numTimesteps; ++t) {
        val_t sum = ZERO;
        for (int tt = t; tt < numTimesteps; ++tt) {
            val_t pB = radB[tt * numTriangles + i];
            val_t pS = radB[(tt - t) * numTriangles + sourceIndex];
            sum += pS * pB;
        }
        if (sum > maxCorr) {
            maxCorr = sum;
            bestT = t;
        }
    }
    distances[i] = WAVE_SPEED * static_cast<val_t>(bestT);
}
