void matrixMultiply(const std::vector<double>& A, const std::vector<double>& B, 
                    std::vector<double>& C, const size_t N, const size_t rows_start, const size_t rows_count) {
    std::fill(C.begin(), C.end(), 0.0);
    
    // Block size for cache optimization
    const size_t BLOCK_SIZE = 64; 

    for (size_t ii = 0; ii < rows_count; ii += BLOCK_SIZE) {
        for (size_t kk = 0; kk < N; kk += BLOCK_SIZE) {
            for (size_t jj = 0; jj < N; jj += BLOCK_SIZE) {
                
                size_t i_max = std::min(ii + BLOCK_SIZE, rows_count);
                size_t k_max = std::min(kk + BLOCK_SIZE, N);
